232  auto& rel_full = std::get<I>(db.full);
233  auto& rel_delta = std::get<I>(db.delta);

239  try {
240  auto& idx = const_cast<typename std::decay_t<decltype(rel_full)>::IndexTypeInst&>(
241  rel_full.get_index(spec));
242  idx.clear();
243  } catch (...) {
244  // Index might not exist, skip
245  }
246  }
247  for (const auto& spec : rel_delta.get_all_indexes()) {
248  try {

306 // check if a tuple is in a relation
307 template <CRelationSchema Schema, std::size_t Ver, typename DB>
308 bool is_in_relation(DB& runtime_db, const std::array<std::size_t, Schema::Arity>& encoded_tuple) {
309  // Access relation directly via tuple indexing to avoid get_relation_by_schema visibility issues
310  // Since query.h is included, we can use IndexRelInDBRels from SRDatalog::detail namespace
311  // Note: runtime/query.h has its own detail namespace, so we need to fully qualify
312  using DB_SchemaTuple = typename DB::schema_type;
313  // Use boost::mp11 to find the index directly
314  using SchemaTuple = typename DB::schema_type;
315  constexpr std::size_t rel_index = boost::mp11::mp_find<SchemaTuple, Schema>::value;
316  auto& rel = [&]() -> auto& {
317  if constexpr (Ver == FULL_VER) {
318  return std::get<rel_index>(runtime_db.full);
319  } else if constexpr (Ver == DELTA_VER) {
320  return std::get<rel_index>(runtime_db.delta);
321  } else if constexpr (Ver == NEW_VER) {
322  return std::get<rel_index>(runtime_db.newt);
323  }
324  }();
325  // Construct canonical index spec from schema arity (determined at compile time)
326  // This uses Schema::canonical_index_type which is std::make_index_sequence<Arity>
327  auto canonical_spec = []<int... Cols>(std::integer_sequence<int, Cols...>) {
328  return IndexSpec{{Cols...}};
329  }(std::make_integer_sequence<int, Schema::Arity>{});
330  // Ensure the canonical index exists (canonical index is always [0,1,2,...,arity-1])
331  // This is not "generating at runtime" - the canonical index spec is determined at compile time
332  const auto& idx = rel.ensure_index(canonical_spec).root();
333  // reorder the tuple
334  auto reordered_tuple = reorder_tuple(canonical_spec, encoded_tuple);
335  return idx.prefix(reordered_tuple).valid();
336 }

604  *
605  * @details When iterating over an index, values are in the order of the index columns.
606  * This function converts them back to schema order (0, 1, 2, ...) for insertion.
607  * Uses compile-time template metaprogramming to unroll the reordering.
608  *
609  * @tparam SchemaArity The arity of the schema
610  * @tparam IndexColSeq The integer sequence of index columns (from IndexSpecT)

612  * @return Values reordered to schema order

625  * @brief Recursive TMP helper to iterate over all tuples in an index.
626  *
627  * @details This helper recursively generates nested loops to iterate over
628  * all tuples in a relation index. For arity N, it generates N
629  * nested loops, each iterating over a column using column_intersect.
630  * Values are collected in index order and then reordered to schema order
631  * before insertion.
632  *
633  * @tparam Arity The remaining arity to process
634  * @tparam NodeHandleType The type of the current node handle
635  * @tparam Schema The relation schema (for semiring type)
636  */
637 template <std::size_t Arity, typename NodeHandleType, CRelationSchema Schema>
638 struct IterateInsertFull;

640 // Base case: arity 0, push the row (reorder values from index order to schema order)
641 template <typename NodeHandleType, CRelationSchema Schema>
642 struct IterateInsertFull<0, NodeHandleType, Schema> {
643  template <typename RelType, typename ValuesArray, CIndexSpec IndexSpecT>
644  static void iterate(const NodeHandleType& /* node */, RelType& relation_full, ValuesArray& values,
645  const IndexSpecT& /* index_spec */) {
646  // Reorder values from index order to schema order using compile-time unrolling
647  using IndexColSeq = typename IndexSpecT::column_indexes_type;
648  // Check if reordering is needed (if index order matches schema order [0,1,2,...], no reordering
649  // needed) For now, always reorder to be safe
650  auto schema_ordered_values = inverse_reorder_tuple_ct<Schema::Arity, IndexColSeq>(values);
651  // Convert array to tuple and push row
652  [&schema_ordered_values, &relation_full]<std::size_t... Is>(std::index_sequence<Is...>) {
653  relation_full.push_row({schema_ordered_values[Is]...}, Schema::semiring_type::one());
654  }(std::make_index_sequence<Schema::Arity>{});
655  }

658 // Recursive case: iterate over current column, then recurse
659 template <std::size_t Arity, typename NodeHandleType, CRelationSchema Schema>
660 struct IterateInsertFull {
661  template <typename RelType, typename ValuesArray, CIndexSpec IndexSpecT>
662  static void iterate(const NodeHandleType& node, RelType& relation_full, ValuesArray& values,
663  const IndexSpecT& /* index_spec */) {
664  constexpr std::size_t current_idx = Schema::Arity - Arity;
665  // current_idx is the position in the index order (0, 1, 2, ... for index columns)
666  // We're iterating in index order, so values[current_idx] gets the value for index column
667  // current_idx
668  for (auto val : RelType::column_intersect(node)) {
669  values[current_idx] = val;
670  auto next_node = node.prefix(val);
671  IterateInsertFull<Arity - 1, decltype(next_node), Schema>::iterate(next_node, relation_full,
672  values, IndexSpecT{});
673  }
674  }

698 /**
699  * @brief Functor for executing rules in a fixpoint computation.
700  *
701  * @details This struct executes recursive rules until a fixpoint is reached.
702  * It uses semi-naive evaluation to avoid redundant computation by
703  * only processing newly derived facts in each iteration.
704  *
705  * The execution process:
706  * 1. Extracts rules and incremental relations from the fixpoint
707  * 2. Iteratively executes all rules
708  * 3. After each iteration, computes delta (merges newt into full)
709  * 4. Stops when no new facts are derived (delta == 0)
710  *
711  * Exposes the compiled Plan type as a member type for debugging and clarity.
712  *
713  * @tparam FP A CFixpoint type containing the recursive rules
714  */
715 template <CFixpoint FP>
716 struct ExecuteRulesForFixpoint {
717  // Compile Fixpoint to monadic FixpointPlan (flat instruction list)
718  using Plan = typename CompileFixpointToPlan<FP>::type;
719 
720  // Extract incremental relations from original Fixpoint (needed for delta computation)
721  using IncrementalRelations = typename GetIncrementalRelationsFromFixpoint<FP>::type;
722 
723  // Extract all index specs from Fixpoint rules (needed for canonical index computation)
724  using RulesTuple = typename ExtractRulesFromFixpoint<FP>::type;
725  using AllIndexSpecs = typename detail::CollectAllIndexSpecs<RulesTuple>::type;
726 
727  // Build canonical indexes for FULL_VER for all incremental relations
728  // This is needed because DestinationRelation uses FULL_VER canonical index for deduplication
729  using CanonicalIndexSpecs =
730  typename detail::CollectCanonicalIndexSpecs<IncrementalRelations, AllIndexSpecs>::type;
731 
732  /**
733  * @brief Execute the fixpoint computation.
734  *
735  * @tparam DB The semi-naive database type
736  * @param db The runtime database
737  * @param max_iterations Maximum number of iterations (default: unlimited)
738  * @return Number of iterations executed
739  */
740  template <AST::CSemiNaiveDatabase DB>
741  std::size_t operator()(DB& db, std::size_t max_iterations = SIZE_MAX) const {
742  // std::cout << "Plan: " << get_type_name<Plan>() << std::endl;
743  std::cout << get_type_name<CanonicalIndexSpecs>() << std::endl;
744 
745  mp::mp_for_each<CanonicalIndexSpecs>([&]<typename IndexSpecT>(IndexSpecT) {
746  using Schema = typename IndexSpecT::schema_type;
747  using ColSeq = typename IndexSpecT::column_indexes_type;
748  auto& rel_full = get_relation_by_schema<Schema, FULL_VER>(db);
749  auto canonical_spec = []<int... Cols>(std::integer_sequence<int, Cols...>) {
750  return IndexSpec{{Cols...}};
751  }(ColSeq{});
752  // be careful here
753  rel_full.ensure_index(canonical_spec);
754  });
755 
756  std::size_t iteration = 0;
757  while (iteration < max_iterations) {
758  LOG_INFO << "Iteration " << iteration;
759 
760  // Check DELTA_VER size at the START of iteration (except iteration 0)
761  // After delta computation in previous iteration, NEW_VER was cleared
762  // So if DELTA_VER is empty at the start of this iteration, no new facts were derived
763  // In future we want to make this check happens more like a IR instruction check instead of a
764  // runtime check
765  if (iteration > 0) {
766  std::size_t total_new_facts = 0;
767  mp::mp_for_each<IncrementalRelations>([&]<typename Schema>(Schema) {
768  auto& newt = get_relation_by_schema<Schema, DELTA_VER>(db);
769  total_new_facts += newt.size();
770  });
771 
772  if (total_new_facts == 0) {
773  LOG_INFO << "No new facts in DELTA_VER at start of iteration " << iteration
774  << ", fixpoint reached after " << iteration << " iterations";
775  break;
776  }
777 
778  LOG_INFO << "Found " << total_new_facts << " new facts in DELTA_VER at start of iteration "
779  << iteration;
780  }
781 
782  // Execute the flat instruction list sequentially using interpreter
783  // No MP needed - interpreter just processes instructions in order
784  // When ExecutePipeline is encountered, it executes the pipeline (nested for-loops)
785  // and then continues to the next instruction
786  // ComputeDelta operations are now part of the MIR instructions, so they execute here
787  detail::MIRExecutor<typename Plan::instructions_type>::execute(db, iteration);
788 
789  iteration++;
790  }
791  if (iteration >= max_iterations) {
792  LOG_WARNING << "Reached maximum iterations (" << max_iterations << "), terminating early";
793  }
794 
795  // For debugging: clear all indexes after query execution
796  // detail::clear_all_indexes_for_debug(db);
797 
798  return iteration;
799  }
800 };

802 /**
803  * @brief Functor for executing non-iterative rule sets.
804  *
805  * @details This struct executes a set of non-recursive rules exactly once.
806  * Unlike fixpoint rules, these rules are evaluated only once and
807  * do not iterate until convergence.
808  *
809  * Exposes compile-time types for debugging and clarity:
810  * - RuleList: The list of rules to execute
811  * - IncrementalRelations: Relations that appear in rule heads
812  * - AllIndexSpecs: All index specifications from the rules
813  * - CanonicalIndexSpecs: Canonical index specs for incremental relations
814  * - MIRPlans: Tuple of compiled MIR plans (one per rule)
815  * - FilteredIndexSpecs: Index specs for incremental relations only
816  *
817  * @tparam Rules A CNonIterativeRuleSets type containing the rules
818  */
819 template <CNonIterativeRuleSets Rules>
820 struct ExecuteRulesForNonIterativeRuleSets {
821  // Extract rule list
822  using RuleList = typename GetRulesFromNonIterativeRuleSets<Rules>::type;
823 
824  // Collect all index specs and compute canonical indices (similar to CompileFixpointToPlan)
825  using IncrementalRelations = typename GetIncrementalRelationsFromRules<RuleList>::type;
826  using AllIndexSpecs = typename detail::CollectAllIndexSpecs<RuleList>::type;
827  using CanonicalIndexSpecs =
828  typename detail::CollectCanonicalIndexSpecs<IncrementalRelations, AllIndexSpecs>::type;
829 
830  // Compile MIR plans for each rule (with canonical indices replaced)
831  template <typename Rule>
832  using CompileRuleMIR =
833  typename ReplaceDestinationRelationsInTuple<typename CompileRuleToMIR<Rule>::type,
834  CanonicalIndexSpecs>::type;
835  using MIRPlans = mp::mp_transform<CompileRuleMIR, RuleList>;
836 
837  // Filtered index specs for rebuilding (only incremental relations)
838  using FilteredIndexSpecs =
839  typename detail::FilterIndexSpecsByRelations<AllIndexSpecs, IncrementalRelations>::type;
840 
841  /**
842  * @brief Execute the non-iterative rule sets.
843  *
844  * @tparam DB The semi-naive database type
845  * @param db The runtime database
846  */
847  template <AST::CSemiNaiveDatabase DB>
848  void operator()(DB& db) const {
849  // Build canonical indexes for FULL_VER for all incremental relations BEFORE executing rules
850  // This is needed because DestinationRelation uses FULL_VER canonical index for deduplication
851  // Same logic as fixpoint execution to avoid collision
852  mp::mp_for_each<CanonicalIndexSpecs>([&]<typename IndexSpecT>(IndexSpecT) {
853  using Schema = typename IndexSpecT::schema_type;
854  using ColSeq = typename IndexSpecT::column_indexes_type;
855  auto& rel_full = get_relation_by_schema<Schema, FULL_VER>(db);
856  auto canonical_spec = []<int... Cols>(std::integer_sequence<int, Cols...>) {
857  return IndexSpec{{Cols...}};
858  }(ColSeq{});
859  rel_full.ensure_index(canonical_spec);
860  });
861 
862  // Execute each rule with canonical indices replaced in MIR
863  mp::mp_for_each<RuleList>([&]<typename Rule>(Rule) {
864  using MIR = CompileRuleMIR<Rule>;
865  // std::cout << "Executing rule: " << get_type_name<Rule>() << std::endl;
866 
867  // Create Pipeline from MIR operations with canonical indices
868  // Extract join plan from rule to compute variable positions
869  // Rule::join_plan_type is already the tuple of variables (JoinPlan::vars_type)
870  using GlobalJoinPlan = typename Rule::join_plan_type;
871  // Compute the variable-to-position map
872  using VarPosMap = typename mir::ComputeVarPosMap<GlobalJoinPlan>::type;
873  // Compute the IndexSpecT-to-iterator-position map
874  using IteratorPosMap = typename mir::ComputePipelineIteratorPosMap<MIR>::type;
875  // Create Pipeline
876  using Pipeline = mir::Pipeline<MIR, VarPosMap, IteratorPosMap>;
877 
878  // Execute using the new tuple-based pipeline execution engine
879  execute_pipeline<Pipeline>(db, 0, true); // non_iterative = true
880  });
881 
882  // Rebuild indexes only for IncrementalRelations (relations in rule heads) on FULL_VER
883  mp::mp_for_each<FilteredIndexSpecs>([&]<typename IndexSpecT>(IndexSpecT) {
884  using Schema = typename IndexSpecT::schema_type;
885  using ColSeq = typename IndexSpecT::column_indexes_type;
886  auto& rel = get_relation_by_schema<Schema, FULL_VER>(db);
887  auto runtime_spec = []<int... Cols>(std::integer_sequence<int, Cols...>) {
888  return IndexSpec{{Cols...}};
889  }(ColSeq{});
890  rel.ensure_index(runtime_spec);
891  });
892 
893  // For debugging: clear all indexes after query execution
894  detail::clear_all_indexes_for_debug(db);
895  }
