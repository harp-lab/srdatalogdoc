200  /**
201  * @brief Executes root join with specialized intersection logic.
202  *
203  * Uses `DeviceSortedArrayIndex::intersect` to find matching keys across multiple root sources.
204  * Supports decomposing execution into subgroups (e.g., 8 threads) for better occupancy
205  * if GroupSize allows.
206  */
207  template <int GroupSize, bool SizeOnly, typename Group, typename OutputCtx>
208  __device__ __attribute__((always_inline)) inline static void execute_root_intersection(
209 
210  Group tile, uint32_t warp_id, uint32_t num_warps, uint32_t num_root_keys,
211  state::VarStore<VarPosMap, ValueType, NumVars>& vars,
212  state::HandleStore<IteratorPosMap, HandleType, NumHandles>& handles,
213  state::ProvStore<RelationPosMap, SR, NumRelations>& provs, OutputCtx& output_ctx) {
214 
215  auto root_handles_tuple = get_root_source_handles(handles);
216 
217  using FirstSource = std::tuple_element_t<0, RootSources>;
218  using FirstIndexSpec = typename FirstSource::index_spec_type;
219  using Schema = typename FirstIndexSpec::schema_type;
220  using IndexType = DeviceSortedArrayIndex<typename Schema::semiring_type,
221  typename Schema::attr_ts_type, ValueType, uint32_t>;
222 
223  auto intersect_view = std::apply([&](auto&... hs) { return IndexType::intersect(tile, hs...); },
224  root_handles_tuple);
225 
226  // Use subgroups for better utilization (e.g., 8 threads instead of 32 for each
227  // tuple)
228  constexpr unsigned int NextGroupSize = 8;
229  static_assert(GroupSize % NextGroupSize == 0, "GroupSize must be divisible by NextGroupSize");
230 
231  // Only partition if NextGroupSize < GroupSize (i.e., if we can decompose further)
232  if constexpr (NextGroupSize < GroupSize) {
233  auto sub_tile = cg::tiled_partition<NextGroupSize>(tile);
234  uint32_t subgroup_id = tile.thread_rank() / NextGroupSize;
235  uint32_t num_subgroups = GroupSize / NextGroupSize;
236 
237  uint32_t idx = 0;
238  for (auto it = intersect_view.begin(); it != intersect_view.end(); ++it, ++idx) {
239  // Warp-level distribution first (global distribution)
240  if (idx % num_warps != warp_id)
241  continue;
242 
243  // Subgroup-level distribution (within warp)
244  // Round-robin assignment of items to subgroups
245  uint32_t warp_item_idx = idx / num_warps;
246 
247  if (warp_item_idx % num_subgroups == subgroup_id) {
248  ValueType root_val = it.value();
249  vars.template set<RootVar>(root_val);
250 
251  state::HandleStore<IteratorPosMap, HandleType, NumHandles> handles_copy;
252  handles_copy.copy_from(handles);
253  // Important: Pass sub_tile so collective ops only involve the subgroup
254  update_handles_after_root(handles_copy, root_val, sub_tile);
255 
256  provs.reset();
257 
258  // Recurse using NextGroupSize and sub_tile
259  GPUPipelineExecutor<RemainingOps, VarPosMap, IteratorPosMap, RelationPosMap,
260  NextGroupSize>::template execute<SR>(sub_tile, vars, handles_copy,
261  provs, output_ctx);
262  }
263  }
264  } else {
265  // GroupSize == NextGroupSize, cannot decompose further, process sequentially
266  uint32_t idx = 0;
267  for (auto it = intersect_view.begin(); it != intersect_view.end(); ++it, ++idx) {
268  // Warp-level distribution first (global distribution)
269  if (idx % num_warps != warp_id)
270  continue;
271 
272  ValueType root_val = it.value();
273  vars.template set<RootVar>(root_val);
274 
275  state::HandleStore<IteratorPosMap, HandleType, NumHandles> handles_copy;
276  handles_copy.copy_from(handles);
277  update_handles_after_root(handles_copy, root_val, tile);
278 
279  provs.reset();
280 
281  // Recurse using the same tile (GroupSize is already NextGroupSize)
282  GPUPipelineExecutor<RemainingOps, VarPosMap, IteratorPosMap, RelationPosMap,
283  NextGroupSize>::template execute<SR>(tile, vars, handles_copy, provs,
284  output_ctx);
285  }
286  }

302  __device__ __attribute__((always_inline)) inline static void execute_root_single(
303 
304  Group tile, uint32_t warp_id, uint32_t num_warps, uint32_t num_root_keys,
305  state::VarStore<VarPosMap, ValueType, NumVars>& vars,
306  state::HandleStore<IteratorPosMap, HandleType, NumHandles>& handles,
307  state::ProvStore<RelationPosMap, SR, NumRelations>& provs, OutputCtx& output_ctx) {
308 
309  using SingleSource = std::tuple_element_t<0, RootSources>;
310  using IndexSpec = typename SingleSource::index_spec_type;
311 
312  const auto& root_handle = handles.template get<IndexSpec>();
313  auto root_values = root_handle.values();
314 
315  // Use subgroups strategies
316  constexpr unsigned int NextGroupSize = 8;
317  static_assert(GroupSize % NextGroupSize == 0, "GroupSize must be divisible by NextGroupSize");
318 
319  // Only partition if NextGroupSize < GroupSize (i.e., if we can decompose further)
320  if constexpr (NextGroupSize < GroupSize) {
321  auto sub_tile = cg::tiled_partition<NextGroupSize>(tile);
322  uint32_t subgroup_id = tile.thread_rank() / NextGroupSize;
323  uint32_t num_subgroups = GroupSize / NextGroupSize;
324 
325  uint32_t global_subgroup_id = warp_id * num_subgroups + subgroup_id;
326  uint32_t total_subgroups = num_warps * num_subgroups;
327 
328  for (uint32_t root_idx = global_subgroup_id; root_idx < num_root_keys;
329  root_idx += total_subgroups) {
330  ValueType root_val = root_values[root_idx];
331  vars.template set<RootVar>(root_val);
332 
333  state::HandleStore<IteratorPosMap, HandleType, NumHandles> handles_copy;
334  handles_copy.copy_from(handles);
335  update_handles_after_root(handles_copy, root_val, sub_tile);
336 
337  provs.reset();
338 
339  GPUPipelineExecutor<RemainingOps, VarPosMap, IteratorPosMap, RelationPosMap,
340  NextGroupSize>::template execute<SR>(sub_tile, vars, handles_copy,
341  provs, output_ctx);
342  }
343  } else {
344  // GroupSize == NextGroupSize, cannot decompose further, process sequentially
345  uint32_t global_subgroup_id = warp_id;
346  uint32_t total_subgroups = num_warps;
347 
348  for (uint32_t root_idx = global_subgroup_id; root_idx < num_root_keys;
349  root_idx += total_subgroups) {
350  ValueType root_val = root_values[root_idx];
351  vars.template set<RootVar>(root_val);
352 
353  state::HandleStore<IteratorPosMap, HandleType, NumHandles> handles_copy;
354  handles_copy.copy_from(handles);
355  update_handles_after_root(handles_copy, root_val, tile);
356 
357  provs.reset();
358 
359  GPUPipelineExecutor<RemainingOps, VarPosMap, IteratorPosMap, RelationPosMap,
360  NextGroupSize>::template execute<SR>(tile, vars, handles_copy, provs,
361  output_ctx);
362  }

383  template <typename ToUpdate, std::size_t Idx, typename Group>
384  __device__ static void update_handles_list(
385  state::HandleStore<IteratorPosMap, HandleType, NumHandles>& handles, ValueType val,
386  Group tile) {
387  if constexpr (Idx < std::tuple_size_v<ToUpdate>) {
388  using Source = std::tuple_element_t<Idx, ToUpdate>;
389  using IndexSpec = typename Source::index_spec_type;
390 
391  auto& handle = handles.template get<IndexSpec>();
392  handle = handle.prefix(val, tile);
393 
394  update_handles_list<ToUpdate, Idx + 1>(handles, val, tile);
395  }
