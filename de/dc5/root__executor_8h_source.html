208  /**
209  * @brief Executes root CartesianJoin - iterates over cartesian product of root values.
210  *
211  * @details For CartesianJoin at root level, we iterate over the cartesian product
212  * of root values from all sources. This is the "boot" kernel that starts from root handles.
213  * Uses root_unique_values for partitioning work across warps.
214  */
215  template <int GroupSize, bool SizeOnly, typename Group, typename OutputCtx>
216  __device__ static void execute_root_cartesian(
217  Group tile, uint32_t warp_id, uint32_t num_warps,
218  const ValueType* __restrict__ root_unique_values, uint32_t num_unique_root_keys,
219  uint32_t num_root_keys, state::VarStore<VarPosMap, ValueType, NumVars>& vars,
220  state::HandleStore<HandleType, NumHandles>& handles,
221  state::ProvStore<RelationPosMap, SR, NumRelations>& provs, OutputCtx& output_ctx) {
222 
223  constexpr int next_group_size = GroupSize;
224  constexpr std::size_t num_vars = std::tuple_size_v<RootVarsTuple>;
225  constexpr std::size_t num_sources = num_root_sources;
226 
227  uint32_t lane = tile.thread_rank();
228  auto single_tile = cg::tiled_partition<1>(tile);
229 
230  // Partition by unique root keys from first source (like benchmark and other root executors)
231  // For each unique root value from first source, do cartesian product of remaining sources
232  for (uint32_t first_idx = warp_id; first_idx < num_unique_root_keys; first_idx += num_warps) {
233  ValueType root_val;
234  if (lane == 0) {
235  root_val = root_unique_values[first_idx];
236  }
237  root_val = tile.shfl(root_val, 0);
238 
239  // Navigate first source handle by unique root value
240  auto& first_handle = handles[0];
241  auto navigated_first_handle = first_handle.prefix(root_val, tile, handles.get_view(0));
242  if (!navigated_first_handle.valid())
243  continue;
244 
245  vars.template set<boost::mp11::mp_at_c<RootVarsTuple, 0>>(root_val);
246 
247  if constexpr (num_sources == 1) {
248  // Copy/Project rule: Path(x, y) :- Edge(x, y)
249  // Var0 = x (root), Var1 = y (children)
250  using Var1 = boost::mp11::mp_at_c<RootVarsTuple, 1>;
251  auto values = navigated_first_handle.values(handles.get_view(0));
252  uint32_t total = static_cast<uint32_t>(values.size());
253 
254  // Need to save/restore handles modified in sub-pipelines
255  constexpr std::size_t NextOffset = num_root_sources;
256  using IndicesToUpdate =
257  typename detail::GetIndicesToUpdate<RemainingOps, Var1, NextOffset>::type;
258  constexpr bool has_indices_to_update = (boost::mp11::mp_size<IndicesToUpdate>::value > 0);
259 
260  auto saved_handles_once = [&]() {
261  if constexpr (has_indices_to_update) {
262  return SRDatalog::state::save_handles_to_tuple<IndicesToUpdate>(handles);
263  } else {
264  return std::tuple<>{};
265  }
266  }();
267 
268  for (uint32_t i = lane; i < total; i += tile.size()) {
269  if constexpr (has_indices_to_update) {
270  SRDatalog::state::restore_handles_from_tuple<IndicesToUpdate>(handles,
271  saved_handles_once);
272  }
273  vars.template set<Var1>(values[i]);
274 
275  update_handles_after_root_cartesian(handles, tile);
276  provs.reset();
277  GPUPipelineExecutor<RemainingOps, VarPosMap, NextOffset, RelationPosMap, 1,
278  true>::template execute<SR>(single_tile, vars, handles, provs,
279  output_ctx);
280  }
281  } else {
282  // True Cartesian Product: Path(x, y) :- A(x), B(y)
283  // Get values and provenance from navigated first handle
284  auto first_values = navigated_first_handle.values(handles.get_view(0));
285  auto first_provs = navigated_first_handle.all_prov(handles.get_view(0));
286  uint32_t first_degree = navigated_first_handle.degree();
287 
288  // Get root values and provenance for remaining sources (still at root level)
289  // For 2-arity case, we only have one remaining source
290  if constexpr (num_vars == 2) {
291  auto& second_handle = handles[1];
292  if (!second_handle.valid())
293  continue;
294 
295  auto second_values = second_handle.values(handles.get_view(1));
296  auto second_provs = second_handle.all_prov(handles.get_view(1));
297  uint32_t second_degree = second_handle.degree();
298 
299  if (second_degree == 0)
300  continue;
301 
302  using FirstSource = std::tuple_element_t<0, RootSources>;
303  using FirstIndexSpec = typename FirstSource::index_spec_type;
304  using FirstSchema = typename FirstIndexSpec::schema_type;
305 
306  using VarsList = boost::mp11::mp_rename<RootVarsTuple, boost::mp11::mp_list>;
307  using Var0 = boost::mp11::mp_at_c<VarsList, 0>;
308  using Var1 = boost::mp11::mp_at_c<VarsList, 1>;
309 
310  // Collect indices to update for both variables
311  constexpr std::size_t NextOffset = num_root_sources;
312  using IndicesToUpdate0 =
313  typename detail::GetIndicesToUpdate<RemainingOps, Var0, NextOffset>::type;
314  using IndicesToUpdate1 =
315  typename detail::GetIndicesToUpdate<RemainingOps, Var1, NextOffset>::type;
316  using IndicesToUpdate =
317  boost::mp11::mp_unique<boost::mp11::mp_append<IndicesToUpdate0, IndicesToUpdate1>>;
318  constexpr bool has_indices_to_update = (boost::mp11::mp_size<IndicesToUpdate>::value > 0);
319 
320  uint32_t total = first_degree * second_degree;
321  const bool major_is_1 = (second_degree >= first_degree);
322 
323  auto saved_handles_once = [&]() {
324  if constexpr (has_indices_to_update) {
325  return SRDatalog::state::save_handles_to_tuple<IndicesToUpdate>(handles);
326  } else {
327  return std::tuple<>{};
328  }
329  }();
330 
331  if constexpr (next_group_size < GroupSize) {
332  auto sub_tile = cg::tiled_partition<next_group_size>(tile);
333  uint32_t subgroup_id = tile.thread_rank() / next_group_size;
334  uint32_t num_subgroups = GroupSize / next_group_size;
335  uint32_t global_subgroup_id = warp_id * num_subgroups + subgroup_id;
336  uint32_t total_subgroups = num_warps * num_subgroups;
337 
338  for (uint32_t flat_idx = global_subgroup_id; flat_idx < total;
339  flat_idx += total_subgroups) {
340  if constexpr (has_indices_to_update) {
341  SRDatalog::state::restore_handles_from_tuple<IndicesToUpdate>(handles,
342  saved_handles_once);
343  }
344  uint32_t idx0, idx1;
345  if (major_is_1) {
346  idx0 = flat_idx / second_degree;
347  idx1 = flat_idx % second_degree;
348  } else {
349  idx1 = flat_idx / first_degree;
350  idx0 = flat_idx % first_degree;
351  }
352 
353  vars.template set<Var0>(first_values[idx0]);
354  vars.template set<Var1>(second_values[idx1]);
355  provs.template set<FirstSchema>(sr_mul<SR>(first_provs[idx0], second_provs[idx1]));
356 
357  update_handles_after_root_cartesian(handles, sub_tile);
358  provs.reset();
359  GPUPipelineExecutor<RemainingOps, VarPosMap, NextOffset, RelationPosMap,
360  next_group_size, !mir::HasMoreColumnJoins<RemainingOps>::value>::
361  template execute<SR>(sub_tile, vars, handles, provs, output_ctx);
362  }
363  } else {
364  uint32_t global_subgroup_id = warp_id;
365  uint32_t total_subgroups = num_warps;
366 
367  for (uint32_t flat_idx = global_subgroup_id; flat_idx < total;
368  flat_idx += total_subgroups) {
369  if constexpr (has_indices_to_update) {
370  SRDatalog::state::restore_handles_from_tuple<IndicesToUpdate>(handles,
371  saved_handles_once);
372  }
373  uint32_t idx0 = 0, idx1 = 0;
374  if (major_is_1) {
375  idx0 = flat_idx / second_degree;
376  idx1 = flat_idx % second_degree;
377  } else {
378  idx1 = flat_idx / first_degree;
379  idx0 = flat_idx % first_degree;
380  }
381 
382  vars.template set<Var0>(first_values[idx0]);
383  vars.template set<Var1>(second_values[idx1]);
384  provs.template set<FirstSchema>(sr_mul<SR>(first_provs[idx0], second_provs[idx1]));
385 
386  update_handles_after_root_cartesian(handles, tile);
387  provs.reset();
388  GPUPipelineExecutor<
389  RemainingOps, VarPosMap, NextOffset, RelationPosMap, next_group_size,
390  !mir::HasMoreColumnJoins<RemainingOps>::value>::template execute<SR>(tile, vars,
391  handles,
392  provs,
393  output_ctx);
394  }
395  }
396  } else {
397  // General N-arity case
398  static_assert(num_vars == 2, "CartesianJoin at root > 2 vars not implemented");
399  }
400  }

405  * @brief Executes root join with specialized intersection logic.
406  * Uses root_unique_values for partitioning work across warps.
407  */
408  template <int GroupSize, bool SizeOnly, typename Group, typename OutputCtx>
409  __device__ static void execute_root_intersection(
410  Group tile, uint32_t warp_id, uint32_t num_warps,
411  const ValueType* __restrict__ root_unique_values, uint32_t num_unique_root_keys,
412  uint32_t num_root_keys, state::VarStore<VarPosMap, ValueType, NumVars>& vars,
413  state::HandleStore<HandleType, NumHandles>& handles,
414  state::ProvStore<RelationPosMap, SR, NumRelations>& provs, OutputCtx& output_ctx) {
415 
416  using FirstSource = std::tuple_element_t<0, RootSources>;
417  using FirstIndexSpec = typename FirstSource::index_spec_type;
418  using Schema = typename FirstIndexSpec::schema_type;
419  using IndexType = DeviceSortedArrayIndex<typename Schema::semiring_type,
420  typename Schema::attr_ts_type, ValueType, uint32_t>;
421 
422  uint32_t lane = tile.thread_rank();
423 
424  // Save root handles for handles that will be updated (needed for loop restoration)
425  // For root intersection, multiple sources provide the SAME variable.
426  // Handles which depend on this variable MUST be updated.
427  // AND the root sources themselves must be updated to the navigated positions.
428  using DependentIndices =
429  typename detail::GetIndicesToUpdate<RemainingOps, RootVar, num_root_sources>::type;
430  using RootIndices = boost::mp11::mp_iota_c<num_root_sources>;
431  using IndicesToUpdate =
432  boost::mp11::mp_unique<boost::mp11::mp_append<RootIndices, DependentIndices>>;
433  constexpr bool has_indices_to_update = (boost::mp11::mp_size<IndicesToUpdate>::value > 0);
434 
435  // Save root handles once before the loop (they don't change during the loop)
436  auto saved_root_handles = [&]() {
437  if constexpr (has_indices_to_update) {
438  return SRDatalog::state::save_handles_to_tuple<IndicesToUpdate>(handles);
439  } else {
440  return std::tuple<>{}; // Dummy tuple if no handles to update
441  }
442  }();
443 
444  // Partition by unique root keys (like benchmark)
445  for (uint32_t y_idx = warp_id; y_idx < num_unique_root_keys; y_idx += num_warps) {
446  // Restore handles to root state at start of each iteration
447  if constexpr (has_indices_to_update) {
448  SRDatalog::state::restore_handles_from_tuple<IndicesToUpdate>(handles, saved_root_handles);
449  }
450  ValueType root_val;
451  if (lane == 0) {
452  root_val = root_unique_values[y_idx];
453  }
454  root_val = tile.shfl(root_val, 0);
455 
456  // Navigate to handles for this root value
457  auto source_handles = [&]<std::size_t... Is>(std::index_sequence<Is...>) {
458  return std::make_tuple(handles[Is].prefix(root_val, tile, handles.get_view(Is))...);
459  }(std::make_index_sequence<num_root_sources>{});
460 
461  // Check if all handles are valid (intersection check)
462  bool all_valid = true;
463  [&]<std::size_t... Is>(std::index_sequence<Is...>) {
464  ((all_valid = all_valid && std::get<Is>(source_handles).valid()), ...);
465  }(std::make_index_sequence<num_root_sources>{});
466 
467  if (!all_valid)
468  continue;
469 
470  // For intersection, we need to intersect the values from all navigated handles
471  // But since we already partitioned by unique root keys, we can directly use the root_val
472  // and process the intersection of child values
473  vars.template set<RootVar>(root_val);
474 
475  // Update handles to the navigated positions
476  [&]<std::size_t... Is>(std::index_sequence<Is...>) {
477  auto update_handle = [&](std::size_t idx, auto&& val) { handles[idx] = val; };
478  (update_handle(Is, std::get<Is>(source_handles)), ...);
479  }(std::make_index_sequence<num_root_sources>{});
480 
481  // Use subgroups for better utilization
482  constexpr unsigned int NextGroupSize = 32;
483  static_assert(GroupSize % NextGroupSize == 0, "GroupSize must be divisible by NextGroupSize");
484  constexpr std::size_t NextOffset = num_root_sources;
485 
486  // Debug: Verify that handles for CartesianJoin sources are identified
487  // For TC query: JoinXZ sources (SourcePath_X, SourceEdge_Z) should be at handles[2] and
488  // handles[3] They depend on VarY, so GetIndicesToUpdate should identify them
489  static_assert(boost::mp11::mp_size<IndicesToUpdate>::value >= 0, "IndicesToUpdate computed");
490 
491  // Update handles to sub-positions for deeper pipeline stages
492 
493  if constexpr (NextGroupSize < GroupSize) {
494  auto sub_tile = cg::tiled_partition<NextGroupSize>(tile);
495  update_handles_after_root(handles, root_val, sub_tile);
496  provs.reset();
497  GPUPipelineExecutor<
498  RemainingOps, VarPosMap, NextOffset, RelationPosMap, NextGroupSize,
499  !mir::HasMoreColumnJoins<RemainingOps>::value>::template execute<SR>(sub_tile, vars,
500  handles, provs,
501  output_ctx);
502  } else {
503  update_handles_after_root(handles, root_val, tile);
504  provs.reset();
505  GPUPipelineExecutor<
506  RemainingOps, VarPosMap, NextOffset, RelationPosMap, NextGroupSize,
507  !mir::HasMoreColumnJoins<RemainingOps>::value>::template execute<SR>(tile, vars,
508  handles, provs,
509  output_ctx);
510  }
511  }

515  * @brief Executes root single source join.
516  * Uses root_unique_values for partitioning work across warps.
517  */
518  template <int GroupSize, bool SizeOnly, typename Group, typename OutputCtx>
519  __device__ static void execute_root_single(
520  Group tile, uint32_t warp_id, uint32_t num_warps,
521  const ValueType* __restrict__ root_unique_values, uint32_t num_unique_root_keys,
522  uint32_t num_root_keys, state::VarStore<VarPosMap, ValueType, NumVars>& vars,
523  state::HandleStore<HandleType, NumHandles>& handles,
524  state::ProvStore<RelationPosMap, SR, NumRelations>& provs, OutputCtx& output_ctx) {
525 
526  // Root handles are at 0..N-1. Single source -> index 0
527  auto& root_handle = handles[0];
528  if (!root_handle.valid())
529  return;
530 
531  uint32_t lane = tile.thread_rank();
