template<Semiring SR, ColumnElementTuple AttrTuple, typename ValueType = uint32_t, typename RowIdType = uint32_t, typename IndexType = ValueType>
class SRDatalog::index::HashmapIndex< SR, AttrTuple, ValueType, RowIdType, IndexType >::NodeHandle

Cursor for navigating and querying nodes in a HashmapIndex.

NodeHandle represents a position at a specific depth/layer in the vertical partition. It tracks the prefix path and provides navigation methods.
