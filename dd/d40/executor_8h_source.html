256 /**
257  * @brief The main template metaprogramming interpreter for pipeline execution.
258  *
259  * @details This is the core of the pipeline execution engine. It recursively
260  * executes pipeline operations by pattern matching on the head operation and
261  * dispatching to appropriate specializations. Uses tuple-based state management
262  * with precompiled position maps from the Pipeline.
263  *
264  * Execution model:
265  * 1. Pattern match on the first MIR operation in the plan
266  * 2. Execute that operation, which may:
267  * - Bind new variables (ColumnJoin, LetClause) - updates StateTuple
268  * - Update iterators (ColumnJoin) - updates IteratorTuple
269  * - Write facts (DestinationRelation)
270  * - Continue with the rest of the plan
271  * 3. Recursively execute the remaining operations
272  *
273  * @tparam MIRPlan A std::tuple of the *remaining* MIR operations to execute
274  * @tparam VarPosMap Compile-time hana::map mapping variable types to positions
275  * @tparam IteratorPosMap Compile-time hana::map mapping IndexSpecT types to iterator positions
276  * @tparam OutputFullIteratorPosMap Compile-time hana::map mapping FULL_VER IndexSpecT types to
277  * positions
278  * @tparam StateTuple Runtime tuple of variable values (fixed size, all variables)
279  * @tparam IteratorTuple Runtime tuple of iterators (fixed size, one per unique IndexSpecT)
280  * @tparam OutputFullIteratorTuple Runtime tuple of full index iterators (fixed size, one per
281  * DestinationRelation)
282  */
283 template <typename MIRPlan, typename VarPosMap, typename IteratorPosMap,
284  typename OutputFullIteratorPosMap, typename StateTuple, typename IteratorTuple,
285  typename OutputFullIteratorTuple>
286 struct PipelineExecutor;

320 /**
321  * @brief Specialization for ColumnJoin: executes a multi-way join operation.
322  *
323  * @details This specialization handles the ColumnJoin MIR operation, which
324  * binds a variable by intersecting multiple column sources. The execution
325  * proceeds as:
326  * 1. Get iterators from IteratorTuple using IteratorPosMap
327  * 2. Compute hash intersection of all iterators
328  * 3. For each value in the intersection:
329  * - Set the variable value in StateTuple
330  * - Update iterators in IteratorTuple (for next variable binding)
331  * - Recurse with updated state
332  *
333  * This implements Worst-Case Optimal Join (WCOJ) semantics.
334  *
335  * @tparam VarToBind The variable to bind through this join
336  * @tparam Sources The tuple of ColumnSource operations to intersect
337  * @tparam TailOps The remaining MIR operations to execute after this join
338  * @tparam VarPosMap Compile-time hana::map mapping variable types to positions
339  * @tparam IteratorPosMap Compile-time hana::map mapping IndexSpecT types to iterator positions
340  * @tparam StateTuple Runtime tuple of variable values
341  * @tparam IteratorTuple Runtime tuple of iterators
342  */
343 template <CVar VarToBind, tmp::CTuple Sources, typename... TailOps, typename VarPosMap,
344  typename IteratorPosMap, typename OutputFullIteratorPosMap, typename StateTuple,
345  typename IteratorTuple, typename OutputFullIteratorTuple>
346 struct PipelineExecutor<std::tuple<mir::ColumnJoin<VarToBind, Sources>, TailOps...>, VarPosMap,
347  IteratorPosMap, OutputFullIteratorPosMap, StateTuple, IteratorTuple,
348  OutputFullIteratorTuple> {
349 
350  template <CSemiNaiveDatabase DB>
351  static void execute(DB& db, StateTuple& state, IteratorTuple& iterators,
352  OutputFullIteratorTuple& output_full_iterators, std::size_t iteration,
353  bool non_iterative) {
354  // 1. Get all iterators for the intersection from IteratorTuple
355  auto iter_tuple = [&]<std::size_t... Is>(std::index_sequence<Is...>) {
356  return std::make_tuple(
357  GenerateIterator<std::tuple_element_t<Is, Sources>, IteratorPosMap, IteratorTuple>::get(
358  iterators)...);
359  }(std::make_index_sequence<std::tuple_size_v<Sources>>{});
360 
361  // 2. Get the relation from the first Source to use its IndexTypeInst for column_intersect
362  using FirstSource = std::tuple_element_t<0, Sources>;
363  // Extract IndexSpecT from ColumnSource template parameters
364  // ColumnSource<IndexSpecT, PrefixVars> - extract first template parameter
365  using FirstIndexSpecT = typename detail::ExtractIndexSpecT<FirstSource>::type;
366  using FirstSchema = typename FirstIndexSpecT::schema_type;
367  constexpr std::size_t first_ver = FirstIndexSpecT::kVersion;
368  // Get relation (handle DELTA_VER redirection in iteration 0)
369  auto& rel = [&]() -> auto& {
370  if constexpr (first_ver == DELTA_VER) {
371  if (iteration == 0) {
372  return get_relation_by_schema<FirstSchema, FULL_VER>(db);
373  }
374  return get_relation_by_schema<FirstSchema, DELTA_VER>(db);
375  } else {
376  return get_relation_by_schema<FirstSchema, first_ver>(db);
377  }
378  }();
379 
380  // 3. Call column_intersect using Relation's static method (uses IndexTypeInst)
381  auto intersection = rel.column_intersect(iter_tuple);
382 
383  // 3. This is the 'for' loop
384  for (const auto& new_val : intersection) {
385  // 4. Set the new variable binding in StateTuple
386  state::set_value_in_tuple<VarPosMap, StateTuple, VarToBind>(state, new_val);
387 
388  // 5. Update iterators in IteratorTuple that have VarToBind in their PrefixVars
389  // Extract ColumnSources from TailOps that need updating
390  using ColumnSourcesToUpdate =
391  typename ExtractColumnSourcesWithVarInPrefix<std::tuple<TailOps...>, VarToBind>::type;
392 
393  // Make a shallow copy of the iterator tuple, update the copy, and recurse with it
394  // When recursion returns, the original iterators are unchanged (no restore needed)
395  auto iterators_copy = iterators; // Shallow copy of entire iterator tuple
396 
397  // Update iterators in the copy with new_val as prefix (not from state)
398  mp::mp_for_each<ColumnSourcesToUpdate>([&]<typename Info>(Info) {
399  using IndexSpecT = typename Info::index_spec_type;
400  // Update iterator in the copy with new_val as prefix
401  state::update_iterator_with_prefixed<IteratorPosMap, IteratorTuple, IndexSpecT>(
402  iterators_copy, new_val);
403  });
404 
405  // Recurse with the modified copy - original iterators remain unchanged
406  PipelineExecutor<std::tuple<TailOps...>, VarPosMap, IteratorPosMap, OutputFullIteratorPosMap,
407  StateTuple, IteratorTuple,
408  OutputFullIteratorTuple>::execute(db, state, iterators_copy,
409  output_full_iterators, iteration,
410  non_iterative);
411  }
412  }
413 };

415 // --- Specialization for Aggregate ---
416 template <CVar VarToBind, CIndexSpec Index, tmp::CTuple PrefixVars, typename AggFunc,
417  typename... TailOps, typename VarPosMap, typename IteratorPosMap,
418  typename OutputFullIteratorPosMap, typename StateTuple, typename IteratorTuple,
419  typename OutputFullIteratorTuple>
420 struct PipelineExecutor<
421  std::tuple<mir::Aggregate<VarToBind, Index, PrefixVars, AggFunc>, TailOps...>, VarPosMap,
422  IteratorPosMap, OutputFullIteratorPosMap, StateTuple, IteratorTuple, OutputFullIteratorTuple> {
423  template <CSemiNaiveDatabase DB>
424  static void execute(DB& db, StateTuple& state, IteratorTuple& iterators,
425  OutputFullIteratorTuple& output_full_iterators, std::size_t iteration,
426  bool non_iterative) {
427  // 1. Get the iterator from IteratorTuple using IteratorPosMap
428  using Idx = Index;
429  constexpr auto iter_pos =
430  std::decay_t<decltype(hana::at_key(IteratorPosMap{}, hana::type_c<Idx>))>::value;
431  auto& handle = std::get<iter_pos>(iterators);
432 
433  // 2. Init accumulator
434  using AccType = typename AggFunc::ResultType;
435  AccType acc = AggFunc::init();
436 
437  // 3. Aggregate if handle is valid (handle is already positioned at prefix)
438  // 3. Aggregate
439  // Perform prefix lookups using PrefixVars
440  // using PV = PrefixVars; // PrefixVars is the tuple type
441 
442  // Copy the handle to avoid mutating the persistent iterator directly
443  // (though in this loop structure, we only read from the iterator's current position usually)
444  auto cur_handle = handle;
445  bool prefix_valid = true;
446 
447  // Apply lookups
448  [&]<typename... Vars>(std::tuple<Vars...>) {
449  return (([&]() {
450  if (!prefix_valid)
451  return;
452  if (!cur_handle.valid()) {
453  prefix_valid = false;
454  return;
455  }
456 
457  auto val_encoded = state::get_value_from_tuple<VarPosMap, StateTuple, Vars>(state);
458  cur_handle = cur_handle.prefix(static_cast<std::uint32_t>(val_encoded));
459 
460  if (!cur_handle.valid())
461  prefix_valid = false;
462  }()),
463  ...);
464  }(PrefixVars{});
465 
466  if (prefix_valid && cur_handle.valid()) {
467  // Determine the type of the aggregated column to decode properly
468  using ColSeq = typename Index::column_indexes_type;
469  using Schema = typename Index::schema_type;
470 
471  // Get the last column index from ColSeq (the one being aggregated)
472  constexpr int LastColIdx = []<int... Cols>(std::integer_sequence<int, Cols...>) {
473  int cols[] = {Cols...};
474  return cols[sizeof...(Cols) - 1];
475  }(ColSeq{});
476 
477  using AttrTuple = typename Schema::attr_ts_type;
478  using ValType = std::tuple_element_t<LastColIdx, AttrTuple>;
479 
480  if (cur_handle.is_leaf()) {
481  // Special case: All index columns matched in prefix (Existence check).
482  // For leaf nodes, we might need to check if there are any rows
483  // But typically for aggregation, we want to aggregate over the values
484  // at the aggregation column level, which should be non-leaf
485  ValType val{};
486  acc = AggFunc::apply(acc, val);
487  } else {
488  // Non-leaf node: iterate over values at current depth (the aggregation column)
489  // These are the values we want to aggregate
490  for (const auto& val : cur_handle.values()) {
491  auto decoded_val = decode_from_size_t<ValType>(val);
492  acc = AggFunc::apply(acc, decoded_val);
493  }
494  }
495  }
496 
497  // 4. Bind var (encode back if necessary, but acc might be a new type)
498  // Aggregation result is a new value bound to VarToBind.
499  // If VarToBind is used later, it might need to be encoded?
500  // In LetClause, we do: auto new_val_encoded = encode_to_size_t(new_val);
501  // So we should encode the result too.
502  auto acc_encoded = encode_to_size_t(acc);
503  state::set_value_in_tuple<VarPosMap, StateTuple, VarToBind>(state, acc_encoded);
504 
505  // 5. Recurse
506  PipelineExecutor<std::tuple<TailOps...>, VarPosMap, IteratorPosMap, OutputFullIteratorPosMap,
507  StateTuple, IteratorTuple,
508  OutputFullIteratorTuple>::execute(db, state, iterators, output_full_iterators,
509  iteration, non_iterative);
510  }

739  * @param iteration The current iteration number

743 auto create_root_iterator(DB& db, std::size_t iteration) {

745  using ColSeq = typename IndexSpecT::column_indexes_type;

852  * @brief Executes a pre-compiled Pipeline.
853  *
854  * @details Executes a Pipeline using tuple-based state management with precompiled
855  * position maps. Creates initial state tuples (variables and iterators) and
856  * executes the pipeline operations.
857  *
858  * @tparam Pipeline The Pipeline to execute (must satisfy CPipeline)
859  * @tparam DB The semi-naive database type
860  * @param runtime_db The runtime database
861  * @param iteration The current iteration number (0 for first iteration)
862  * @param non_iterative If true, writes directly to FULL_VER instead of NEW_VER
863  */
864 template <typename Pipeline, CSemiNaiveDatabase DB>
865 void execute_pipeline(DB& runtime_db, std::size_t iteration = 0, bool non_iterative = false) {
866  static_assert(mir::is_pipeline_v<Pipeline>, "Pipeline must satisfy CPipeline");
867 
868  using MIROps = typename Pipeline::mir_ops_type;
869  using VarPosMap = typename Pipeline::var_pos_map_type;
870  using IteratorPosMap = typename Pipeline::iterator_pos_map_type;
871 
872  // 1. Extract all index specs from MIR operations
873  using ExtractedIndexSpecs = typename mir::ExtractIndexSpecsFromMIR<MIROps>::type;
874 
875  // 2. Build all indexes explicitly
876  detail::build_all_indexes_from_mir<ExtractedIndexSpecs>(runtime_db, iteration);
877 
878  // 3. Create initial iterator tuple directly from IteratorPosMap
879  // This ensures the order matches exactly (position 0, 1, 2, ...)
880  auto iterators = detail::create_initial_iterator_tuple<IteratorPosMap, DB>(runtime_db, iteration);
881 
882  // 4. Extract DestinationRelation FULL_VER IndexSpecTs and compute OutputFullIteratorPosMap
883  using DestinationRelationIndexSpecs =
884  typename detail::ExtractDestinationRelationIndexSpecs<MIROps>::type;
885  // Make them unique (each IndexSpecT appears only once)
886  using UniqueDestRelIndexSpecsList = mp::mp_unique<DestinationRelationIndexSpecs>;
887  using UniqueDestRelIndexSpecs = mp::mp_rename<UniqueDestRelIndexSpecsList, std::tuple>;
888  // Create the map from FULL_VER IndexSpecT type to position
889  using OutputFullIteratorPosMap =
890  typename mir::ComputeIndexSpecToIteratorPosMap<UniqueDestRelIndexSpecs>::type;
891 
892  // 5. Create initial output_full_iterators tuple directly from OutputFullIteratorPosMap
893  // This ensures the order matches exactly (position 0, 1, 2, ...)
894  auto output_full_iterators =
895  detail::create_initial_output_full_iterator_tuple<OutputFullIteratorPosMap, DB>(runtime_db);
896 
897  // 6. Create initial state tuple (all variables, fixed size based on join plan)
898  // All values initialized to MAX_INT (unbound variables)
899  constexpr auto state_size = hana::size(VarPosMap{});
900  constexpr std::size_t max_val = std::numeric_limits<std::size_t>::max();
901  auto state = [&]<std::size_t... Is>(std::index_sequence<Is...>) {
902  return std::make_tuple(((void)Is, max_val)...);
903  }(std::make_index_sequence<decltype(state_size)::value>{});
904 
905  // 7. Start the recursive pipeline executor
906  using StateTuple = decltype(state);
907  using IteratorTuple = decltype(iterators);
908  using OutputFullIteratorTuple = decltype(output_full_iterators);
909  detail::PipelineExecutor<MIROps, VarPosMap, IteratorPosMap, OutputFullIteratorPosMap, StateTuple,
910  IteratorTuple, OutputFullIteratorTuple>::execute(runtime_db, state,
911  iterators,
912  output_full_iterators,
913  iteration,
914  non_iterative);
915 }
