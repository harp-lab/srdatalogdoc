2  * @file query.h
3  * @brief Query compilation and index management for Datalog programs.
4  *
5  * @details This file provides template metaprogramming utilities for:
6  * - Compiling high-level AST rules into Mid-level IR (MIR) operations
7  * - Computing required indexes for efficient join execution
8  * - Building runtime indexes from compile-time index specifications

12  * join operations suitable for Worst-Case Optimal Join (WCOJ) algorithms.

14  * @note **C++20/23 Features Used**: This file uses C++20 concepts for type
15  * constraints, `if constexpr` for compile-time branch selection, and

264  * @brief Gets a runtime relation object from a SemiNaiveDatabase using its schema.
265  *
266  * @details Retrieves a reference to a specific relation version (full, delta,
267  * or newt) from a SemiNaiveDatabase based on the compile-time schema and version
268  * number. This bridges the gap between compile-time type information and runtime
269  * data structures.
270  *
271  * @note **C++17/C++20 feature**: Uses `if constexpr` for compile-time branch
272  * selection based on the version parameter. This enables efficient
273  * compile-time dispatch without runtime overhead.
274  *
275  * @tparam Schema The relation schema (compile-time ADT)
276  * @tparam Ver The version number (0=full, 1=delta, 2=newt)
277  * @tparam Database The SemiNaiveDatabase type
278  *
279  * @param rel_tuple Reference to the SemiNaiveDatabase instance
280  * @return Reference to the requested relation version
281  *
282  * @note The version parameter uses numeric constants: 0=FULL_VER, 1=DELTA_VER, 2=NEW_VER
283  */
284 template <CRelationSchema Schema, std::size_t Ver, CSemiNaiveDatabase Database>
285 inline auto& get_relation_by_schema(Database& rel_tuple) {
286  constexpr std::size_t index =
287  detail::IndexRelInDBRels<Schema, typename Database::schema_type>::value;
288  // rel_tuple.verify_distinct_addresses();
289  if constexpr (Ver == FULL_VER) {
290  return std::get<index>(rel_tuple.full);
291  } else if constexpr (Ver == DELTA_VER) {
292  return std::get<index>(rel_tuple.delta);
293  } else if constexpr (Ver == NEW_VER) {
294  return std::get<index>(rel_tuple.newt);
295  } else {
296  static_assert(false, "Invalid version");
297  }
298 }
