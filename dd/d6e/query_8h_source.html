614template <typename Clause, CVar VarToBind, tmp::CTuple BoundVars,
615 tmp::CTuple JoinPlan>
616struct ClauseToColumnSource_impl {
617 using ClauseVars = typename GetClauseVars<Clause>::type;
618 using ClauseTerms = typename GetNormalClauseTerms<Clause>::type;
619 using Schema = typename GetNormalClauseSchema<Clause>::type;
620
621 // 1. Find the prefix vars (those *already* bound)
622 using PrefixVars = typename tmp::TupleSetIntersection<BoundVars, ClauseVars>;
623
624 // 2. Find the free vars
625 using FreeVars = typename tmp::TupleSetDifference<ClauseVars, PrefixVars>;
626
627 // 3. The access order for this *one* clause
628 // using AccessOrder = tmp::TupleConcat<PrefixVars, FreeVars>;
629 // TODO: the access order is based on clause variable order in join plan
630 using AccessOrder = typename tmp::TupleSetIntersection<JoinPlan, ClauseVars>;
631
632 // 4. The column indexes for this access order
633 using ColumnIndexes =
634 typename MapVarsToColumnIndexes<ClauseTerms, AccessOrder>::type;
635
636 // 5. Extract version from clause and create IndexSpec ADT
637 static constexpr std::size_t ClauseVer =
638 GetNormalClauseVersion<Clause>::value;
639 using Index = IndexSpecT<Schema, ColumnIndexes, ClauseVer>;
640
641 // 6. The new ColumnSource MIR op (version is now in IndexSpec)
642 using type = mir::ColumnSource<Index, PrefixVars>;
643};
