362template <CRelationSchema Schema, std::size_t Ver, CSemiNaiveDatabase Database>
363inline auto &get_relation_by_schema(Database &rel_tuple) {
364 constexpr std::size_t index =
365 detail::IndexRelInDBRels<Schema, typename Database::schema_type>::value;
366 // rel_tuple.verify_distinct_addresses();
367 if constexpr (Ver == FULL_VER) {
368 return std::get<index>(rel_tuple.full);
369 } else if constexpr (Ver == DELTA_VER) {
370 return std::get<index>(rel_tuple.delta);
371 } else if constexpr (Ver == NEW_VER) {
372 return std::get<index>(rel_tuple.newt);
373 } else {
374 static_assert(false, "Invalid version");
375 }

652template <typename Clause, CVar VarToBind, tmp::CTuple BoundVars,
653 tmp::CTuple JoinPlan>
654struct ClauseToColumnSource_impl {
655 using ClauseVars = typename GetClauseVars<Clause>::type;
656 using ClauseTerms = typename GetNormalClauseTerms<Clause>::type;
657 using Schema = typename GetNormalClauseSchema<Clause>::type;
658
659 // 1. Find the prefix vars (those *already* bound)
660 using PrefixVars = typename tmp::TupleSetIntersection<BoundVars, ClauseVars>;
661
662 // 2. Find the free vars
663 using FreeVars = typename tmp::TupleSetDifference<ClauseVars, PrefixVars>;
664
665 // 3. The access order for this *one* clause
666 // using AccessOrder = tmp::TupleConcat<PrefixVars, FreeVars>;
667 // TODO: the access order is based on clause variable order in join plan
668 using AccessOrder = typename tmp::TupleSetIntersection<JoinPlan, ClauseVars>;
669
670 // 4. The column indexes for this access order
671 using ColumnIndexes =
672 typename MapVarsToColumnIndexes<ClauseTerms, AccessOrder>::type;
673
674 // 5. Extract version from clause and create IndexSpec ADT
675 static constexpr std::size_t ClauseVer =
676 GetNormalClauseVersion<Clause>::value;
677 using Index = IndexSpecT<Schema, ColumnIndexes, ClauseVer>;
678
679 // 6. The new ColumnSource MIR op (version is now in IndexSpec)
680 using type = mir::ColumnSource<Index, PrefixVars>;
681};
