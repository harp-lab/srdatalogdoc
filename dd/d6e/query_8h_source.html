275template <typename Clause, CVar VarToBind, tmp::CTuple BoundVars,

277struct ClauseToColumnSource_impl {
278 using ClauseVars = typename GetClauseVars<Clause>::type;
279 using ClauseTerms = typename GetNormalClauseTerms<Clause>::type;
280 using Schema = typename GetNormalClauseSchema<Clause>::type;
281
282 // 1. Find the prefix vars (those *already* bound)
283 using PrefixVars = typename tmp::TupleSetIntersection<BoundVars, ClauseVars>;
284
285 // 2. Find the free vars
286 using FreeVars = typename tmp::TupleSetDifference<ClauseVars, PrefixVars>;
287
288 // 3. The access order for this *one* clause
289 // using AccessOrder = tmp::TupleConcat<PrefixVars, FreeVars>;
290 // TODO: the access order is based on clause variable order in join plan
291 using AccessOrder = typename tmp::TupleSetIntersection<JoinPlan, ClauseVars>;
292
293 // 4. The column indexes for this access order
294 using ColumnIndexes =
295 typename MapVarsToColumnIndexes<ClauseTerms, AccessOrder>::type;
296
297 // 5. Extract version from clause and create IndexSpec ADT
298 static constexpr std::size_t ClauseVer =
299 GetNormalClauseVersion<Clause>::value;
300 using Index = IndexSpecT<Schema, ColumnIndexes, ClauseVer>;
301
302 // 6. The new ColumnSource MIR op (version is now in IndexSpec)
303 using type = mir::ColumnSource<Index, PrefixVars>;
304};
