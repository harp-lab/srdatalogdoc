2  * @file system.h
3  * @brief System abstraction layer for memory management and hardware configuration.
4  *
5  * @details This header provides a unified interface for memory management and
6  * system-level abstractions used throughout SRDatalog. Currently, it focuses
7  * on memory resource management using Boost's PMR (Polymorphic Memory Resource)
8  * facilities, but is designed to evolve into a more comprehensive system
9  * abstraction layer.
10  *
11  * @section system_current Current Implementation
12  *
13  * The current implementation provides:
14  * - **Memory Resource Abstractions**: Lightweight aliases around Boost PMR
15  * facilities, allowing the rest of the codebase to depend on a single header
16  * - **PMR-Aware Containers**: Type aliases for containers that support custom
17  * memory allocators (Vector, Map)
18  * - **Default Resource Access**: Helper function to obtain the default memory
19  * resource
20  *
21  * This indirection makes it possible to change the backend allocator in one
22  * place, enabling different memory strategies (arena allocation, memory pooling,
23  * custom allocators) without modifying the rest of the codebase.
24  *
25  * @section system_future Future Evolution
26  *
27  * This module is designed to evolve into a more polymorphic and configurable
28  * system abstraction layer that will support:
29  *
30  * - **Hardware Abstraction**: Runtime selection of execution backends based on
31  * available hardware (CPU, GPU, TPU, etc.)
32  * - **Parallel Framework Support**: Configurable parallel execution frameworks
33  * (OpenMP, TBB, CUDA, SYCL, etc.) selected at runtime based on configuration
34  * and hardware capabilities
35  * - **Memory Strategy Selection**: Runtime configuration of memory allocation
36  * strategies optimized for different hardware and workloads
37  * - **Performance Tuning**: Runtime configuration of performance parameters
38  * (thread counts, batch sizes, cache sizes) based on hardware characteristics
39  *
40  * The current simple aliases will be extended with polymorphic interfaces and
41  * factory patterns that allow runtime selection of implementations based on
42  * configuration files, environment variables, or programmatic configuration.
43  *
44  * @section system_usage Usage
45  *
46  * @code{.cpp}
47  * using namespace SRDatalog;
48  *
49  * // Use default memory resource
50  * Vector<int> vec(default_memory_resource());
51  *
52  * // Use custom memory resource (e.g., arena allocator)
53  * Arena arena;
54  * Vector<int> vec2(&arena);
55  *
56  * // PMR-aware map
57  * Map<std::string, int> my_map(default_memory_resource());
58  * @endcode
59  *
60  * @note The current implementation is intentionally minimal to provide a stable
61  * foundation. Future enhancements will maintain backward compatibility
62  * while adding polymorphic capabilities.
63  *
64  * @see memory_resource for the memory resource type
65  * @see Vector for PMR-aware vector type
66  * @see Map for PMR-aware map type
67  */

202 /**
203  * @brief PMR memory resource that uses Highway's aligned allocation functions.
204  * @details This memory resource provides 64-byte aligned allocations compatible
205  * with SIMD operations. It uses Highway's AllocateAlignedBytes and
206  * FreeAlignedBytes functions, which are target-independent and work
207  * across all SIMD targets.
208  * @note This is particularly useful for Highway VQSort and other SIMD operations
209  * that require aligned memory.
210  */
211 class HighwayAlignedMemoryResource : public memory_resource {
212  public:
213  void* do_allocate(size_t bytes, size_t /*alignment*/) override {
214  // Highway's AllocateAlignedBytes provides proper alignment (typically 64 bytes)
215  // It's target-independent and works across all SIMD targets
216  void* ptr = hwy::AllocateAlignedBytes(bytes, nullptr, nullptr);
217  // if (ptr == nullptr) {
218  // throw std::bad_alloc();
219  // }
220  return ptr;
221  }
222 
223  void do_deallocate(void* p, size_t /*bytes*/, size_t /*alignment*/) override {
224  if (p != nullptr) {
225  hwy::FreeAlignedBytes(p, nullptr, nullptr);

227  }

230  return this == &other;

274 inline void optimize_vector_memory(void* ptr, size_t size_bytes) {
275  // 1. Align pointer to page boundary (downwards)
276  uintptr_t addr = reinterpret_cast<uintptr_t>(ptr);
277  uintptr_t aligned_addr = addr & ~(4095ULL);
278 
279 #ifdef __linux__
280  // 2. Tell Kernel: "I will access this sequentially, please pre-load."
281  // MADV_HUGEPAGE: Use 2MB pages (Reduces TLB misses and Faults)
282  // MADV_SEQUENTIAL: Aggressive read-ahead
283  madvise(reinterpret_cast<void*>(aligned_addr), size_bytes, MADV_HUGEPAGE);
284  madvise(reinterpret_cast<void*>(aligned_addr), size_bytes, MADV_SEQUENTIAL);
285 #endif
286 }
