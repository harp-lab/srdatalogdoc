28 inline DelimiterInfo detect_delimiter(std::ifstream& file, std::string& first_line) {
29  std::string line;
30  DelimiterInfo info;
31 
32  while (std::getline(file, line)) {
33  if (line.empty() || (line.find_first_not_of(" \t\r\n") == std::string::npos)) {
34  continue;
35  }
36  first_line = line;
37 
38  std::size_t tab_count = std::count(line.begin(), line.end(), '\t');
39  std::size_t comma_count = std::count(line.begin(), line.end(), ',');
40 
41  if (tab_count > 0) {
42  info.single_char = '\t';
43  info.is_multi_space = false;
44  return info;
45  }
46  if (comma_count > 0) {
47  info.single_char = ',';
48  info.is_multi_space = false;
49  return info;
50  }
51 
52  std::size_t max_consecutive_spaces = 0;
53  std::size_t current_consecutive = 0;
54  for (char c : line) {
55  if (c == ' ') {
56  current_consecutive++;
57  max_consecutive_spaces = std::max(max_consecutive_spaces, current_consecutive);
58  } else {
59  current_consecutive = 0;
60  }
61  }
62 
63  if (max_consecutive_spaces >= 2) {
64  info.single_char = ' ';
65  info.is_multi_space = true;
66  info.min_spaces = max_consecutive_spaces;
67  return info;
68  }
69 
70  info.single_char = ' ';
71  info.is_multi_space = false;
72  return info;
73  }
74  return info;
75 }

79  * @brief Internal helper function to parse a line into a tuple.
80  *
81  * @details This function parses a single line from a file into a tuple of the
82  * expected type, handling both multi-space and single-character delimiters.
83  *
84  * @tparam AttrTuple The tuple type to parse into
85  * @param line The line to parse
86  * @param delim_info Delimiter information
87  * @param multi_space_regex Regex for multi-space delimiters
88  * @param line_number Current line number (for error messages)
89  * @return The parsed tuple
90  */
91 template <typename AttrTuple>
92 AttrTuple parse_line(const std::string& line, const DelimiterInfo& delim_info,
93  const std::regex& multi_space_regex, std::size_t line_number) {
94  constexpr std::size_t expected_fields = std::tuple_size_v<AttrTuple>;
95  AttrTuple tuple;
96  std::size_t field_count = 0;
97 
98  if (delim_info.is_multi_space) {
99  std::sregex_token_iterator iter(line.begin(), line.end(), multi_space_regex, -1);
100  std::sregex_token_iterator end;
101 
102  std::vector<std::string> tokens;
103  for (; iter != end; ++iter) {
104  std::string token = *iter;
105  auto first_non_ws = token.find_first_not_of(" \t\r\n");
106  if (first_non_ws != std::string::npos) {
107  token.erase(0, first_non_ws);
108  auto last_non_ws = token.find_last_not_of(" \t\r\n");
109  if (last_non_ws != std::string::npos) {
110  token.erase(last_non_ws + 1);
111  tokens.push_back(token);
112  }
113  }
114  }
115 
116  if (tokens.size() != expected_fields) {
117  throw std::runtime_error("Invalid file format at line " + std::to_string(line_number) +
118  ": expected " + std::to_string(expected_fields) +
119  " fields, but found " + std::to_string(tokens.size()));
120  }
121 
122  hana::for_each(hana::make_range(0_c, hana::size_c<expected_fields>), [&](auto i) {
123  field_count++;
124  std::string item = tokens[decltype(i)::value];
125 
126  using ElementType = std::tuple_element_t<decltype(i)::value, AttrTuple>;
127  std::stringstream converter(item);
128  if constexpr (std::is_integral_v<ElementType>) {
129  converter >> std::get<decltype(i)::value>(tuple);
130  if (converter.fail() || !converter.eof()) {
131  throw std::runtime_error("Invalid file format at line " + std::to_string(line_number) +
132  ", field " + std::to_string(field_count) + ": cannot convert '" +
133  item + "' to integer");
134  }
135  } else if constexpr (std::is_floating_point_v<ElementType>) {
136  converter >> std::get<decltype(i)::value>(tuple);
137  if (converter.fail() || !converter.eof()) {
138  throw std::runtime_error("Invalid file format at line " + std::to_string(line_number) +
139  ", field " + std::to_string(field_count) + ": cannot convert '" +
140  item + "' to floating point");
141  }
142  } else {
143  std::get<decltype(i)::value>(tuple) = item;
144  }
145  });
146  } else {
147  std::stringstream ss(line);
148  std::string item;
149 
150  hana::for_each(hana::make_range(0_c, hana::size_c<expected_fields>), [&](auto i) {
151  field_count++;
152  if (!std::getline(ss, item, delim_info.single_char)) {
153  throw std::runtime_error("Invalid file format at line " + std::to_string(line_number) +
154  ": expected " + std::to_string(expected_fields) +
155  " fields, but only found " + std::to_string(field_count - 1));
156  }
157 
158  auto first_non_ws = item.find_first_not_of(" \t\r\n");
159  if (first_non_ws != std::string::npos) {
160  item.erase(0, first_non_ws);
161  auto last_non_ws = item.find_last_not_of(" \t\r\n");
162  if (last_non_ws != std::string::npos) {
163  item.erase(last_non_ws + 1);
164  } else {
165  item.clear();
166  }
167  } else {
168  item.clear();
169  }
170 
171  using ElementType = std::tuple_element_t<decltype(i)::value, AttrTuple>;
172  std::stringstream converter(item);
173  if constexpr (std::is_integral_v<ElementType>) {
174  converter >> std::get<decltype(i)::value>(tuple);
175  if (converter.fail() || !converter.eof()) {
176  throw std::runtime_error("Invalid file format at line " + std::to_string(line_number) +
177  ", field " + std::to_string(field_count) + ": cannot convert '" +
178  item + "' to integer");
179  }
180  } else if constexpr (std::is_floating_point_v<ElementType>) {
181  converter >> std::get<decltype(i)::value>(tuple);
182  if (converter.fail() || !converter.eof()) {
183  throw std::runtime_error("Invalid file format at line " + std::to_string(line_number) +
184  ", field " + std::to_string(field_count) + ": cannot convert '" +
185  item + "' to floating point");
186  }
187  } else {
188  std::get<decltype(i)::value>(tuple) = item;
189  }
190  });
191  }
192 
193  return tuple;
194 }

196 /**
197  * @brief Internal helper function to load file and process each tuple with a callback.
198  *
199  * @details This function handles the common file loading logic: opening the file,
200  * detecting delimiters, and parsing lines. For each parsed tuple, it calls
201  * the provided callback function.
202  *
203  * @tparam AttrTuple The tuple type to parse into
204  * @tparam Callback The callback function type (called with the parsed tuple)
205  * @param file_path The path to the file to load
206  * @param callback Function to call for each parsed tuple
207  */
208 template <typename AttrTuple, typename Callback>
209 void load_file_impl(const std::string& file_path, const Callback& callback) {
210  std::ifstream file(file_path);
211  if (!file.is_open()) {
212  throw std::runtime_error("Could not open file: " + file_path);
213  }
214 
215  std::string first_line;
216  DelimiterInfo delim_info = detect_delimiter(file, first_line);
217 
218  if (first_line.empty()) {
219  return;
220  }
221 
222  file.clear();
223  file.seekg(0, std::ios::beg);
224 
225  std::regex multi_space_regex;
226  if (delim_info.is_multi_space) {
227  std::string pattern = " {" + std::to_string(delim_info.min_spaces) + ",}";
228  multi_space_regex = std::regex(pattern);
229  }
230 
231  std::string line;
232  std::size_t line_number = 0;
233  LOG_INFO << "Loading data from file: " << file_path;
234  LOG_INFO << "Delimiter info: " << delim_info.single_char << " " << delim_info.is_multi_space
235  << " " << delim_info.min_spaces;
236 
237  while (std::getline(file, line)) {
238  line_number++;
239 
240  if (line.empty() || (line.find_first_not_of(" \t\r\n") == std::string::npos)) {
241  continue;
242  }
243 
244  AttrTuple tuple = parse_line<AttrTuple>(line, delim_info, multi_space_regex, line_number);
245  callback(tuple);
246  }
247  LOG_INFO << "Loaded " << line_number << " lines from file: " << file_path;
248 }

253  using AttrTuple = typename Schema::attr_ts_type;
