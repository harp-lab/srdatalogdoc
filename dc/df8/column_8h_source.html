355template <typename T> class DictionaryCodec {
356public:
358 using DictMap = boost::unordered::pmr::unordered_map<T, std::size_t>;
360 using InverseMap = boost::unordered::pmr::unordered_map<std::size_t, T>;
361
364 explicit DictionaryCodec() : next_id_(0) {}
365
376 std::size_t encode(const T &value) {
377 if (auto it = forward_.find(value); it != forward_.end()) {
378 return it->second;
379 }
380 std::size_t id = next_id_++;
381 forward_.emplace(value, id);
382 inverse_.emplace(id, value);
383 return id;
384 }
385
392 const T &decode(std::size_t id) const {
393 auto it = inverse_.find(id);
394 if (it == inverse_.end()) {
395 throw std::runtime_error("Invalid encoded value for dictionary");
396 }
397 return it->second;
398 }
399
406 bool contains(std::size_t id) const { return inverse_.contains(id); }
407
412 std::size_t size() const { return forward_.size(); }
413
417 void clear() {
418 forward_.clear();
419 inverse_.clear();
420 next_id_ = 0;
421 }
422
423private:
424 DictMap forward_;
425 InverseMap inverse_;
426 std::size_t next_id_;
427};
