35 // Helper: Generate map type for layer L (0-indexed)
36 // Layer 0: Map<std::tuple<>, Set<ValueType>> - maps empty tuple to all possible first values
37 // Layer 1: Map<std::tuple<ValueType>, Set<ValueType>> - maps (v0) to all possible second values
38 // Layer L: Map<std::tuple<ValueType, ... (L times)>, Set<ValueType>> - maps prefix of L values to
39 // next values Leaf: Map<std::tuple<ValueType, ... (arity times)>, Set<RowId>> - maps full prefix to
40 // set of row IDs
41 template <std::size_t Layer, typename ValueType, typename RowId, typename IndexType,
42  std::size_t Arity>
43 struct LayerMapType;
44 
45 // Non-leaf layers: map prefix tuple to set of next values (ensures uniqueness)
46 // Layer 0: uses std::tuple<> (empty tuple) as key
47 // Layer L: uses std::tuple<ValueType, ... (L times)> as key
48 // All layers 0 to Arity-1 are non-leaf (path goes through all columns)
49 template <std::size_t Layer, typename ValueType, typename RowId, typename IndexType,
50  std::size_t Arity>
51  requires(Layer < Arity)
52 struct LayerMapType<Layer, ValueType, RowId, IndexType, Arity> {
53  using KeyType =
54  TupleRepeat_t<Layer, ValueType>; // Tuple of Layer ValueTypes (empty tuple for Layer 0)
55  using ValueSetType = Set<IndexType>; // Set ensures uniqueness of values
56  using type = Map<KeyType, ValueSetType>;
57 };
58 
59 // Leaf layer: map full prefix tuple to set of row IDs (separate layer after all columns)
60 // Leaf is at layer Arity (after all non-leaf layers 0 to Arity-1)
61 template <std::size_t Layer, typename ValueType, typename RowId, typename IndexType,
62  std::size_t Arity>
63  requires(Layer == Arity)
64 struct LayerMapType<Layer, ValueType, RowId, IndexType, Arity> {
65  using KeyType = TupleRepeat_t<Arity, ValueType>; // Tuple of Arity ValueTypes
66  using ValueSetType = RowId; // Just store one RowId per unique key (memory efficient)
67  using type = Map<KeyType, ValueSetType>;
68 };
69 
70 template <std::size_t Layer, typename ValueType, typename RowId, typename IndexType,
71  std::size_t Arity>
72 using LayerMapType_t = typename LayerMapType<Layer, ValueType, RowId, IndexType, Arity>::type;

74 // Generate tuple of map types for vertical partitioning
75 // For arity N, we need N layers:
76 // - Layer 0: Map<ValueType, Vector<ValueType>>
77 // - Layer 1: Map<std::tuple<ValueType, ValueType>, Vector<ValueType>>
78 // - ...
79 // - Layer N-1: Map<std::tuple<ValueType, ... (N times)>, RowId>
80 template <std::size_t Arity, typename ValueType, typename RowId, typename IndexType = ValueType>
81 struct VerticalPartitionTypes;
82 
83 // Base case: arity 0 (empty tuple)
84 template <typename ValueType, typename RowId, typename IndexType>
85 struct VerticalPartitionTypes<0, ValueType, RowId, IndexType> {
86  using type = std::tuple<>;
87 };
88 
89 // Generate tuple with one map per layer
90 // For arity N, we have N+1 layers: layers 0 to N-1 are non-leaf, layer N is leaf
91 template <std::size_t Arity, typename ValueType, typename RowId, typename IndexType>
92 struct VerticalPartitionTypes {
93  template <std::size_t... Is>
94  static auto make_map_tuple(std::index_sequence<Is...>) {
95  return std::tuple<LayerMapType_t<Is, ValueType, RowId, IndexType, Arity>...>{};
96  }
97  // Create arity+1 layers: 0 to arity-1 (non-leaf) + arity (leaf)
98  using type = decltype(make_map_tuple(std::make_index_sequence<Arity + 1>{}));
99 };

101 // Special case: arity 1 (layer 0 non-leaf + layer 1 leaf)
102 template <typename ValueType, typename RowId, typename IndexType>
103 struct VerticalPartitionTypes<1, ValueType, RowId, IndexType> {
104  using type = std::tuple<LayerMapType_t<0, ValueType, RowId, IndexType, 1>,
105  LayerMapType_t<1, ValueType, RowId, IndexType, 1>>;
