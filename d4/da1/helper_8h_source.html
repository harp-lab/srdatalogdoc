24 /// @brief Helper to convert a range to a container using the pipe operator
25 /// @details Usage: auto vec = range | to<std::vector>();
26 /// Compatible with std::ranges::to but works with older standard libraries
27 /// @note **C++20 features**: Uses `std::ranges::input_range` concept,
28 /// `std::ranges::sized_range` for optimization, `if constexpr` for
29 /// compile-time conditionals, and `std::ranges::copy` for range copying.
30 template<template<typename...> class Container>
31 struct to_fn {
32  template<std::ranges::input_range R>
33  constexpr auto operator()(R&& r) const {
34  using value_type = std::ranges::range_value_t<R>;
35  Container<value_type> result;
36  if constexpr (std::ranges::sized_range<R>) {
37  result.reserve(std::ranges::size(r));
38  }
39  std::ranges::copy(r, std::back_inserter(result));
40  return result;
41  }
42 };
