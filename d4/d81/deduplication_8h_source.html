24 /**
25  * @brief Deduplicate rows, aggregate provenance, and build root CSR in fused pass.
26  *
27  * @details This function combines three operations that share common computation:
28  * 1. Deduplicate rows (all columns compared)
29  * 2. Aggregate provenance using SR::add for duplicate rows
30  * 3. Build CSR index for root level (first column only)
31  *
32  * ## Algorithm Flow
33  * ```
34  * Input (sorted):
35  * rows: [(A,X) (A,X) (A,Y) (B,X) (B,X)]
36  * prov: [ 1 2 1 3 1 ]
37  *
38  * Step 1-2: Mark heads + positions (full row comparison)
39  * flags: [1 0 1 1 0]
40  * pos: [0 0 1 2 2]
41  * num_unique = 3
42  *
43  * Step 3: Aggregate provenance
44  * agg_prov: [3, 1, 4] (1+2, 1, 3+1)
45  *
46  * Step 4: Compact columns
47  * rows: [(A,X) (A,Y) (B,X)]
48  * prov: [ 3 1 4 ]
49  *
50  * Step 5: Build CSR on first column (after compaction)
51  * csr_flags: [1 0 1] (A != _, A == A, B != A)
52  * csr_values: [A, B]
53  * csr_meta: [(0,2), (2,3)] (A spans [0,2), B spans [2,3))
54  * ```
55  *
56  * ## Buffer Usage
57  *
58  * | Buffer | Size | Lifetime | Used For | Reuse Potential |
59  * |--------------|------------|--------------|-----------------------------|-----------------|
60  * | head_flags | num_rows | Steps 1-4 | Mark unique rows (all cols) | (1) |
61  * | output_pos | num_rows | Steps 2-4 | Compaction positions | (1) |
62  * | segment_ids | num_rows | Step 3 | Segment IDs for reduce | (2) |
63  * | agg_prov | num_unique | Steps 3-end | Aggregated provenance | None (output) |
64  * | csr_flags | num_unique | Step 5 | Mark unique in first col | (3) |
65  * | csr_pos | num_unique | Step 5 | CSR segment positions | (3) |
66  *
67  * ## Future Buffer Reuse Opportunities
68  *
69  * (1) head_flags + output_pos: Could be combined by packing flag into position high bit.
70  * NOT IMPLEMENTED: Clarity > micro-optimization.
71  *
72  * (2) segment_ids: Could reuse output_pos after compaction since segment_ids =
73  * inclusive_scan(head_flags) and output_pos = exclusive_scan(head_flags).
74  * They differ by 1: segment_ids[i] = output_pos[i] + head_flags[i]
75  * NOT IMPLEMENTED: Extra arithmetic vs. one more scan. Profile first.
76  *
77  * (3) csr_flags + csr_pos: Operate on compacted data (num_unique elements).
78  * Could reuse head_flags/output_pos buffers since num_unique <= num_rows.
79  * NOT IMPLEMENTED: Keep separate for clarity and easier debugging.
80  *
81  * @tparam SR Semiring type for provenance aggregation
82  * @tparam ValueType Column element type
83  * @tparam N Number of columns (arity)
84  *
85  * @param[in,out] array Sorted NDDeviceArray, compacted in-place
86  * @param[in,out] provenance Provenance values, aggregated and resized
87  * @param[out] root_unique_values Unique values in first column (for partitioning only, not
88  * correctness)
89  *
90  * @pre array must be sorted lexicographically
91  * @pre provenance.size() == array.num_rows()
92  * @post array.num_rows() == provenance.size() == number of unique rows
93  * @post root_unique_values.size() == unique values in col 0
94  *
95  * @note This function uses array.column_ptr<I>() directly (not through a view) to avoid
96  * dangling pointers. If resize() reallocates, any pointers captured from a view would
97  * become invalid. Getting fresh pointers from the array ensures correctness.
98  */
99 template <Semiring SR, typename ValueType, std::size_t N>
100 void deduplicate_aggregate_and_unique(NDDeviceArray<ValueType, N>& array,
101  DeviceArray<semiring_value_t<SR>>& provenance,
102  DeviceArray<ValueType>& root_unique_values) {
103  //==========================================================================
104  // Trivial cases
105  //==========================================================================
106  if (array.num_rows() == 0) {
107  root_unique_values.reset();
108  return;
109  }
110 
111  if (array.num_rows() == 1) {
112  // Single element: no deduplication needed, directly build CSR
113  // Short-circuit: just copy the single value
114  root_unique_values.resize(1);
115  thrust::copy_n(thrust::device, array.template column_ptr<0>(), 1, root_unique_values.begin());
116  return;
117  }
118 
119  const uint32_t num_rows = static_cast<uint32_t>(array.num_rows());
120 
121  // Critical: Get column pointers directly from array (not from view) to avoid dangling pointers.
122  // If resize() reallocates memory, any pointers captured from a view become invalid.
123  // Using array.column_ptr<I>() ensures we always get fresh pointers to the current storage.
124 
125  //==========================================================================
126  // Key iterator for full-row comparison
127  //==========================================================================
128  auto keys = [&]<std::size_t... I>(std::index_sequence<I...>) {
129  return thrust::make_zip_iterator(thrust::make_tuple(array.template column_ptr<I>()...));
130  }(std::make_index_sequence<N>());
131 
132  //==========================================================================
133  // Step 1: Mark segment heads (full row comparison)
134  // Buffer: head_flags[num_rows] - uint32_t
135  // head_flags[i] = 1 if row[i] != row[i-1], else 0
136  //==========================================================================
137  DeviceArray<uint32_t> head_flags(num_rows);
138 
139  thrust::tabulate(thrust::device, head_flags.begin(), head_flags.end(),
140  [keys] __device__(uint32_t idx) -> uint32_t {
141  if (idx == 0)
142  return 1;
143  return (keys[idx] != keys[idx - 1]) ? 1 : 0;
144  });
145 
146  //==========================================================================
147  // Step 2: Compute output positions via exclusive scan
148  // Buffer: output_pos[num_rows] - uint32_t
149  // output_pos[i] = number of unique rows before index i
150  // Invariant: output_pos[i] <= i (always)
151  //==========================================================================
152  DeviceArray<uint32_t> output_pos(num_rows);
153  thrust::exclusive_scan(rmm::exec_policy(), head_flags.begin(), head_flags.end(),
154  output_pos.begin(), uint32_t{0}, thrust::plus<uint32_t>());
155 
156  // Get unique count by summing flags (count of 1s in head_flags)
157  const uint32_t num_unique = thrust::reduce(thrust::device, head_flags.begin(), head_flags.end(),
158  uint32_t{0}, thrust::plus<uint32_t>());
159 
160  //==========================================================================
161  // Step 3: Aggregate provenance by segment
162  // Buffer: segment_ids[num_rows] - uint32_t (could reuse, see notes above)
163  // Buffer: agg_prov[num_unique] - SR::value_type
164  // segment_ids[i] = 1-indexed segment ID for row i (count of segment heads up to i)
165  //==========================================================================
166  DeviceArray<uint32_t> segment_ids(num_rows);
167 
168  // Count segment heads using arithmetic plus (not boolean OR - we need numeric IDs for grouping)
169  // bool implicitly converts: true→1, false→0, so this counts how many segment heads seen so far
170  thrust::inclusive_scan(rmm::exec_policy(), head_flags.begin(), head_flags.end(),
171  segment_ids.begin(), thrust::plus<uint32_t>());
172 
173  DeviceArray<semiring_value_t<SR>> agg_prov(num_unique);
174 
175  // CRITICAL: Ensure provenance size matches array size before reduce_by_key
176  // If provenance is too small, resize and fill with semiring identity to prevent
177  // out-of-bounds access in reduce_by_key which would cause incorrect aggregation
178  if (provenance.size() != num_rows) {
179  if (provenance.size() < num_rows) {
180  const auto old_size = provenance.size();
181  provenance.resize(num_rows);
182  // Fill new elements with semiring identity (one)
183  if (old_size > 0) {
184  thrust::fill(thrust::device, provenance.begin() + old_size, provenance.end(), sr_one<SR>());
185  } else {
186  thrust::fill(thrust::device, provenance.begin(), provenance.end(), sr_one<SR>());
187  }
188  } else {
189  // Provenance is larger than array - this is a serious error
190  throw std::runtime_error("deduplicate_aggregate_and_unique: Provenance size (" +
191  std::to_string(provenance.size()) + ") > array rows (" +
192  std::to_string(num_rows) + "). This indicates a logic error.");
193  }
194  }
195 
196  {
197  nvtx3::scoped_range r("deduplicate_aggregate_and_build_csr::reduce_by_key");
198  thrust::reduce_by_key(
199  rmm::exec_policy(), segment_ids.begin(), segment_ids.end(), provenance.begin(),
200  thrust::make_discard_iterator(), agg_prov.begin(), thrust::equal_to<uint32_t>(),
201  [] __device__(semiring_value_t<SR> a, semiring_value_t<SR> b) { return SR::add(a, b); });
202  }
203 
204  //==========================================================================
205  // Step 4: Compact columns using gather
206  // 1. Compute gather indices (indices of unique elements)
207  // 2. Gather each column into temp buffer, then copy back
208  // This is robust against race conditions and ensures correct compaction
209  //==========================================================================
210  DeviceArray<uint32_t> gather_indices(num_unique);
211  thrust::copy_if(thrust::device, thrust::counting_iterator<uint32_t>(0),
212  thrust::counting_iterator<uint32_t>(num_rows), head_flags.begin(),
213  gather_indices.begin(), [] __device__(uint32_t flag) { return flag == 1; });
214 
215  // Use safe compaction with temporary buffer and gather
216  // 1. Allocate temp buffer
217  DeviceArray<ValueType> temp_col(num_unique);
218  [&]<std::size_t... I>(std::index_sequence<I...>) {
219  auto compact = [&](auto dim_const) {
220  constexpr std::size_t dim = decltype(dim_const)::value;
221  // Get fresh column pointer directly from array to ensure correct stride
222  ValueType* col = array.template column_ptr<dim>();
223 
224  // 2. Gather to temp buffer
225  thrust::gather(thrust::device, gather_indices.begin(), gather_indices.end(),
226  thrust::device_ptr<ValueType>(col), temp_col.begin());
227 
228  // 3. Copy back to original column
229  thrust::copy(thrust::device, temp_col.begin(), temp_col.end(),
230  thrust::device_ptr<ValueType>(col));
231 
232  // Ensure copy is complete before temp_col is destroyed
233  cudaDeviceSynchronize();
234  };
235  (compact(std::integral_constant<std::size_t, I>{}), ...);
236  }(std::make_index_sequence<N>());
237 
238  // Resize and copy aggregated provenance
239  // Note: Compaction kernels above complete before resize() is called (Thrust synchronizes).
240  // After resize(), array.column_ptr<I>() will return fresh pointers to the current storage,
241  // avoiding any dangling pointer issues from pre-resize views.
242  array.resize(num_unique);
243  provenance.resize(num_unique);
244  thrust::copy_n(thrust::device, agg_prov.begin(), num_unique, provenance.begin());
245 
246  // Synchronize to ensure all operations complete before building CSR
247  cudaDeviceSynchronize();
248 
249  //==========================================================================
250  // Step 5: Build CSR on first column (compacted data)
251  // Use the separated build_csr_from_deduplicated_array function
252  // Note: After resize(), we must ensure any cached pointers are refreshed.
253  // build_csr_from_deduplicated_array will get a fresh column_ptr<0>() from the array,
254  // so it should be safe.
255  //==========================================================================
256  generate_unique(array, root_unique_values);
257 }

259 /**
260  * @brief Build root CSR from sorted, deduplicated data (first column only)
261  *
262  * @details This function builds CSR index on the first column of sorted, deduplicated data.
263  * It identifies unique values in the first column and builds (begin, end) pairs for each.
264  *
265  * This function uses array.column_ptr<0>() directly (not through a view) to avoid dangling
266  * pointers. If the array was recently resized, using a view could result in stale pointers.
267  *
268  * @param array Sorted, deduplicated NDDeviceArray (compacted data)
269  * @param[out] root_unique_values Unique values in first column
270  *
271  * @pre array is sorted lexicographically
272  * @pre array is deduplicated (no duplicate rows)
273  * @post root_unique_values.size() == unique values in column 0
274  */
275 template <typename ValueType, std::size_t N>
276 void generate_unique(const NDDeviceArray<ValueType, N>& array,
277  DeviceArray<ValueType>& root_unique_values) {
278  const uint32_t num_rows = static_cast<uint32_t>(array.num_rows());
279 
280  if (num_rows == 0) {
281  root_unique_values.reset();
282  return;
283  }
284 
285  if (num_rows == 1) {
286  // Single element: directly build CSR
287  root_unique_values.resize(1);
288  thrust::copy_n(thrust::device, array.template column_ptr<0>(), 1, root_unique_values.begin());
289  return;
290  }
291 
292  // Get column pointer directly (fresh pointer, not from view) to ensure correct stride
293  const ValueType* first_col = array.template column_ptr<0>();
294 
295  // Step 1: Unique copy using iterator_unique (or reduce_by_key with discard)
296  // We only need the unique keys, not the counts.
297  // However, thrust::unique_copy is for contiguous duplicates.
298  //
299  // Option A: Use unique_copy with start/end of array.column_ptr<0>()
300  // Option B: Keep reduce_by_key but discard values.
301  //
302  // reduce_by_key is optimized and we already have it. Let's just discard the counts.
303 
304  // Resize to maximum possible size (all unique) first
305  root_unique_values.resize(num_rows);
306 
307  // Use unique_copy to extract unique keys
308  // This is semantically equivalent to reduce_by_key on keys with discard on values,
309  // but potentially more robust or optimized for this specific case.
310  auto result_end = thrust::unique_copy(thrust::device, first_col, first_col + num_rows,
311  root_unique_values.begin());
312 
313  const uint32_t num_csr = result_end - root_unique_values.begin();
314 
315  // Resize outputs to actual number of unique keys
316  root_unique_values.resize(num_csr);
317 }
