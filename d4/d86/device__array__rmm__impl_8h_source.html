40  * @brief Configuration for RMM pool memory resource
41  * @details Pool sizes can be configured via environment variables:
42  * - SRDATALOG_RMM_POOL_INITIAL_SIZE: Initial pool size in bytes (default: 1GB)
43  * - SRDATALOG_RMM_POOL_MAX_SIZE: Maximum pool size in bytes (default: unlimited)
44  * Set to 0 or use environment variable to specify a limit, otherwise unlimited
45  * @note If environment variables are not set, uses default values (unlimited max size)
46  */
47 namespace RMMConfig {
48 // Default pool sizes (can be overridden via environment variables)
49 constexpr std::size_t DEFAULT_INITIAL_SIZE = 1024ULL * 1024 * 1024; // 1GB
50 // Default max size is unlimited (use a very large value that's a multiple of 256)
51 // Using UINT64_MAX rounded down to multiple of 256: 0xFFFFFFFFFFFFFE00
52 // RMM requires max_size to be a multiple of 256 bytes
53 constexpr std::size_t DEFAULT_MAX_SIZE = 0xFFFFFFFFFFFFFE00ULL; // ~18 exabytes, multiple of 256
54 
55 inline std::size_t get_initial_size() {
56  const char* env = std::getenv("SRDATALOG_RMM_POOL_INITIAL_SIZE");
57  if (env != nullptr) {
58  return std::strtoull(env, nullptr, 0);
59  }
60  return DEFAULT_INITIAL_SIZE;
61 }
62 
63 inline std::size_t get_max_size() {
64  const char* env = std::getenv("SRDATALOG_RMM_POOL_MAX_SIZE");
65  if (env != nullptr) {
66  std::size_t value = std::strtoull(env, nullptr, 0);
67  // Allow 0 to mean unlimited (use default unlimited value)
68  if (value == 0) {
69  return DEFAULT_MAX_SIZE;
70  }
71  // Round to nearest multiple of 256 (round down) - RMM requirement
72  return (value / 256) * 256;
73  }
74  return DEFAULT_MAX_SIZE; // Unlimited by default
75 }

77 inline std::optional<std::size_t> get_max_size_optional() {
78  const char* env = std::getenv("SRDATALOG_RMM_POOL_MAX_SIZE");
79  if (env != nullptr) {
80  std::size_t value = std::strtoull(env, nullptr, 0);
81  // Allow 0 or empty to mean unlimited (return std::nullopt)
82  if (value == 0) {
83  return std::nullopt;
84  }
85  // Round to nearest multiple of 256 (round down) - RMM requirement
86  return std::make_optional((value / 256) * 256);
87  }
88  return std::nullopt; // Unlimited by default (no max size limit)
89 }

92 /**
93  * @brief Thread-safe singleton that provides a global GPU pool memory resource
94  * @note CUDA must be initialized before this is called (call init_cuda() first)
95  * @note This function is host-only (allocation happens on host)
96  * @note The pool is also set as the global per-device resource so all RMM allocations use it
97  */
98 inline rmm::mr::device_memory_resource* get_gpu_pool_memory_resource() {
99  static std::unique_ptr<rmm::mr::pool_memory_resource<rmm::mr::cuda_memory_resource>> pool = []() {
100  // Ensure CUDA is initialized before accessing RMM
101  int current_device = -1;
102  cudaError_t err = cudaGetDevice(&current_device);
103  if (err != cudaSuccess) {
104  throw std::runtime_error(
105  "get_gpu_pool_memory_resource: CUDA not initialized. Call init_cuda() first. Error: " +
106  std::string(cudaGetErrorString(err)));
107  }
108 
109  // Create CUDA memory resource as upstream
110  // Note: pool_memory_resource takes ownership via raw pointer, so we use new
111  // and manage the lifetime through the pool itself
112  auto cuda_mr = new rmm::mr::cuda_memory_resource();
113 
114  // Get configured pool sizes (from environment variables or defaults)
115  std::size_t initial_size = RMMConfig::get_initial_size();
116  std::optional<std::size_t> max_size = RMMConfig::get_max_size_optional();
117 
118  // Create pool memory resource with configurable sizes
119  // Pass std::nullopt for max_size to indicate unlimited (no max size limit)
120  // If max_size is nullopt, RMM will allow the pool to grow without limit (up to GPU memory)
121  auto pool_ptr = std::make_unique<rmm::mr::pool_memory_resource<rmm::mr::cuda_memory_resource>>(
122  cuda_mr, initial_size, max_size);
123 
124  // Set as the current device's memory resource so all RMM allocations use this pool
125  // This ensures device_uvector, rmm::device_vector, and other RMM containers
126  // all use the same pool, and Thrust temporary allocations (via rmm::exec_policy) can use it too
127  rmm::mr::set_current_device_resource(pool_ptr.get());
128 
129  return pool_ptr;
130  }();
131  return pool.get();
132 }
