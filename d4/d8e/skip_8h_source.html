20 template <typename T>
21 const T* skip_duplicates_hwy(const T* ptr, const T* end) {
22  const hn::ScalableTag<T> d;
23  const size_t N = hn::Lanes(d);
24 
25  // 1. Snapshot the value we are skipping
26  const T target_val = *ptr;
27  const auto v_target = hn::Set(d, target_val);
28 
29  // 2. SIMD Loop: Consume chunks of size N
30  // We use <= end because we need a full vector's worth of data to load safely.
31  // (If you have a padded buffer, you can be more aggressive here).
32  while (ptr + N <= end) {
33  // LoadU handles unaligned addresses (ptr increments by N, so alignment drifts)
34  const auto v_data = hn::LoadU(d, ptr);
35 
36  // Compare: result is a mask where TRUE means "is a duplicate"
37  const auto mask = hn::Eq(v_data, v_target);
38 
39  // Optimization: If the whole vector matches, skip it all.
40  if (hn::AllTrue(d, mask)) {
41  ptr += N;
42  continue;
43  }
44 
45  // 3. Mismatch Found inside this chunk
46  // We want the first index where they are NOT equal.
47  // Invert the mask: TRUE means "is DIFFERENT"
48  const auto not_mask = hn::Not(mask);
49 
50  // FindFirstTrue returns the index (0..N-1) of the first set bit
51  const size_t offset = hn::FindFirstTrue(d, not_mask);
52  return ptr + offset;
53  }
54 
55  // 4. Tail Cleanup (Scalar)
56  // Handle the last few elements that didn't fit in a SIMD vector
57  while (ptr < end && *ptr == target_val) {
58  ptr++;
59  }
60 
61  return ptr;
62 }
