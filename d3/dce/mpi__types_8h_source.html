172 // 4) Derived datatype helper for trivially copyable T
173 // ===================================================
174 //
175 // Creates a contiguous byte representation for any trivially copyable POD.
176 // Caller owns the resulting MPI_Datatype and must free it with MPI_Type_free.
177 // This is a simple “byte blob” representation (no field alignment semantics).
178 //
179 template<class T>
180 inline MPI_Datatype create_contiguous_datatype() {
181  static_assert(std::is_trivially_copyable_v<T>,
182  "create_contiguous_datatype<T>() requires trivially copyable T.");
183  MPI_Datatype dt = MPI_DATATYPE_NULL;
184  // Represents sizeof(T) bytes; portable across homogeneous nodes.
185  MPI_Type_contiguous(static_cast<int>(sizeof(T)), MPI_BYTE, &dt);
186  MPI_Type_commit(&dt);
187  return dt;
188 }
