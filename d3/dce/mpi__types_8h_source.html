70// WARNING: For size_t/ptrdiff_t and non-direct types (e.g., std::string),

94 else if constexpr (std::is_same_v<T, long double>) return MPI_LONG_DOUBLE;

97 else if constexpr (std::is_same_v<T, bool>) return MPI_C_BOOL;

114// Map an integer type *by size in bytes* using MPI_Type_match_size.
115// Requires MPI to be initialized. Returns MPI_DATATYPE_NULL on failure.
116inline MPI_Datatype mpi_datatype_match_size_integer(int nbytes) {
117 MPI_Datatype dt = MPI_DATATYPE_NULL;
118 if (MPI_Type_match_size(MPI_TYPECLASS_INTEGER, nbytes, &dt) == MPI_SUCCESS)
119 return dt;
120 return MPI_DATATYPE_NULL;
121}

123// Portable mapping for size_t (do NOT guess the underlying type)
124inline MPI_Datatype mpi_datatype_of_size_t() {
125 return mpi_datatype_match_size_integer(static_cast<int>(sizeof(std::size_t)));
126}

171// ===================================================
172// 4) Derived datatype helper for trivially copyable T
173// ===================================================
174//
175// Creates a contiguous byte representation for any trivially copyable POD.
176// Caller owns the resulting MPI_Datatype and must free it with MPI_Type_free.
177// This is a simple “byte blob” representation (no field alignment semantics).
178//
179template<class T>
180inline MPI_Datatype create_contiguous_datatype() {
181 static_assert(std::is_trivially_copyable_v<T>,
182 "create_contiguous_datatype<T>() requires trivially copyable T.");
183 MPI_Datatype dt = MPI_DATATYPE_NULL;
184 // Represents sizeof(T) bytes; portable across homogeneous nodes.
185 MPI_Type_contiguous(static_cast<int>(sizeof(T)), MPI_BYTE, &dt);
186 MPI_Type_commit(&dt);
187 return dt;
188}
