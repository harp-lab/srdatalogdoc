Load data from a file directly into a relation. This function loads data from a file (CSV, TSV, or space-separated) directly into a relation object. It automatically detects the delimiter and parses the file accordingly. Each line in the file becomes a row in the relation with the semiring's one() value as the annotation. After loading, it builds all registered indexes.

struct DelimiterInfo {
char single_char = ' ';
bool is_multi_space = false;
std::size_t min_spaces = 1;
};
inline DelimiterInfo detect_delimiter(std::ifstream& file, std::string& first_line) {
std::string line;
DelimiterInfo info;
while (std::getline(file, line)) {
if (line.empty() || (line.find_first_not_of(" \t\r\n") == std::string::npos)) {
continue;
}
first_line = line;
std::size_t tab_count = std::count(line.begin(), line.end(), '\t');
std::size_t comma_count = std::count(line.begin(), line.end(), ',');
if (tab_count > 0) {
info.single_char = '\t';
info.is_multi_space = false;
return info;
}
if (comma_count > 0) {
info.single_char = ',';
info.is_multi_space = false;
return info;
}
std::size_t max_consecutive_spaces = 0;
std::size_t current_consecutive = 0;
for (char c : line) {
if (c == ' ') {
current_consecutive++;
max_consecutive_spaces = std::max(max_consecutive_spaces, current_consecutive);
} else {
current_consecutive = 0;
}
}
if (max_consecutive_spaces >= 2) {
info.single_char = ' ';
info.is_multi_space = true;
info.min_spaces = max_consecutive_spaces;
return info;
}
info.single_char = ' ';
info.is_multi_space = false;
return info;
}
return info;
}

template <typename AttrTuple>
AttrTuple parse_line(const std::string& line, const DelimiterInfo& delim_info,
const std::regex& multi_space_regex, std::size_t line_number) {
constexpr std::size_t expected_fields = std::tuple_size_v<AttrTuple>;
AttrTuple tuple;
std::size_t field_count = 0;
if (delim_info.is_multi_space) {
std::sregex_token_iterator iter(line.begin(), line.end(), multi_space_regex, -1);
std::sregex_token_iterator end;
std::vector<std::string> tokens;
for (; iter != end; ++iter) {
std::string token = *iter;
auto first_non_ws = token.find_first_not_of(" \t\r\n");
if (first_non_ws != std::string::npos) {
token.erase(0, first_non_ws);
auto last_non_ws = token.find_last_not_of(" \t\r\n");
if (last_non_ws != std::string::npos) {
token.erase(last_non_ws + 1);
tokens.push_back(token);
}
}
}
if (tokens.size() != expected_fields) {
throw std::runtime_error("Invalid file format at line " + std::to_string(line_number) +
": expected " + std::to_string(expected_fields) +
" fields, but found " + std::to_string(tokens.size()));
}
hana::for_each(hana::make_range(0_c, hana::size_c<expected_fields>), [&](auto i) {
field_count++;
std::string item = tokens[decltype(i)::value];
using ElementType = std::tuple_element_t<decltype(i)::value, AttrTuple>;
std::stringstream converter(item);
if constexpr (std::is_integral_v<ElementType>) {
converter >> std::get<decltype(i)::value>(tuple);
if (converter.fail() || !converter.eof()) {
throw std::runtime_error("Invalid file format at line " + std::to_string(line_number) +
", field " + std::to_string(field_count) + ": cannot convert '" +
item + "' to integer");
}
} else if constexpr (std::is_floating_point_v<ElementType>) {
converter >> std::get<decltype(i)::value>(tuple);
if (converter.fail() || !converter.eof()) {
throw std::runtime_error("Invalid file format at line " + std::to_string(line_number) +
", field " + std::to_string(field_count) + ": cannot convert '" +
item + "' to floating point");
}
} else {
std::get<decltype(i)::value>(tuple) = item;
}
});
} else {
std::stringstream ss(line);
std::string item;
hana::for_each(hana::make_range(0_c, hana::size_c<expected_fields>), [&](auto i) {
field_count++;
if (!std::getline(ss, item, delim_info.single_char)) {
throw std::runtime_error("Invalid file format at line " + std::to_string(line_number) +
": expected " + std::to_string(expected_fields) +
" fields, but only found " + std::to_string(field_count - 1));
}
auto first_non_ws = item.find_first_not_of(" \t\r\n");
if (first_non_ws != std::string::npos) {
item.erase(0, first_non_ws);
auto last_non_ws = item.find_last_not_of(" \t\r\n");
if (last_non_ws != std::string::npos) {
item.erase(last_non_ws + 1);
} else {
item.clear();
}
} else {
item.clear();
}
using ElementType = std::tuple_element_t<decltype(i)::value, AttrTuple>;
std::stringstream converter(item);
if constexpr (std::is_integral_v<ElementType>) {
converter >> std::get<decltype(i)::value>(tuple);
if (converter.fail() || !converter.eof()) {
throw std::runtime_error("Invalid file format at line " + std::to_string(line_number) +
", field " + std::to_string(field_count) + ": cannot convert '" +
item + "' to integer");
}
} else if constexpr (std::is_floating_point_v<ElementType>) {
converter >> std::get<decltype(i)::value>(tuple);
if (converter.fail() || !converter.eof()) {
throw std::runtime_error("Invalid file format at line " + std::to_string(line_number) +
", field " + std::to_string(field_count) + ": cannot convert '" +
item + "' to floating point");
}
} else {
std::get<decltype(i)::value>(tuple) = item;
}
});
}
return tuple;
}
template <typename AttrTuple, typename Callback>
void load_file_impl(const std::string& file_path, const Callback& callback) {
std::ifstream file(file_path);
if (!file.is_open()) {
throw std::runtime_error("Could not open file: " + file_path);
}
std::string first_line;
DelimiterInfo delim_info = detect_delimiter(file, first_line);
if (first_line.empty()) {
return;
}
file.clear();
file.seekg(0, std::ios::beg);
std::regex multi_space_regex;
if (delim_info.is_multi_space) {
std::string pattern = " {" + std::to_string(delim_info.min_spaces) + ",}";
multi_space_regex = std::regex(pattern);
}
std::string line;
std::size_t line_number = 0;
LOG_INFO << "Loading data from file: " << file_path;
LOG_INFO << "Delimiter info: " << delim_info.single_char << " " << delim_info.is_multi_space
<< " " << delim_info.min_spaces;
while (std::getline(file, line)) {
line_number++;
if (line.empty() || (line.find_first_not_of(" \t\r\n") == std::string::npos)) {
continue;
}
AttrTuple tuple = parse_line<AttrTuple>(line, delim_info, multi_space_regex, line_number);
callback(tuple);
}
LOG_INFO << "Loaded " << line_number << " lines from file: " << file_path;
}

template <CRelationSchema Schema, typename DB>
void load_from_file(DB& runtime_db, const std::string& file_path) {
using AttrTuple = typename Schema::attr_ts_type;
// Load all data first without building indexes
detail::load_file_impl<AttrTuple>(file_path, [&](const AttrTuple& tuple) {
std::apply(
[&](auto&&... args) {
add_fact<Schema>(runtime_db, Schema::semiring_type::one(), args...);
},
tuple);
});
// Build all indexes after loading is complete
auto& rel = get_relation_by_schema<Schema, FULL_VER>(runtime_db);
rel.build_all_indexes();
}

void load_file(Relation<SR, AttrTuple, IndexType, Policy>& relation, const std::string& file_path) {
// Load all data first without building indexes
detail::load_file_impl<AttrTuple>(

}
