198template <typename Term, typename StateMap>
199constexpr auto get_term_value(const StateMap &state) {
200 if constexpr (CVar<Term>) {
201 // Case 1: Variable - get value from state map using type as key
202 return state[hana::type_c<Term>];
203 } else if constexpr (CConst<Term>) {
204 // Case 2: Constant - use compile-time constant value
205 return Term::value;
206 } else if constexpr (CCppExpr<Term>) {
207 // Case 3: C++ Expression - would evaluate expression
208 static_assert(!std::is_same_v<Term, Term>,
209 "CppExpr not yet supported in head position");
210 return 0; // unreachable
211 } else {
212 // Case 4: Unknown term type - compile error
213 static_assert(!std::is_same_v<Term, Term>, "Unknown term type");
214 return 0; // unreachable
215 }
216}

239template <CVar VarToFind, typename StateMap> struct GetValueFromState {
240 static decltype(auto) get(const StateMap &state) {
241 // Create a type key for the variable
242 constexpr auto var_key = hana::type_c<VarToFind>;
243
244 // Lookup in the map (compile-time check that key exists)
245 constexpr auto has_key = hana::contains(state, var_key);
246 static_assert(decltype(has_key)::value,
247 "Variable not found in bound state. Check that the variable "
248 "is bound before use.");
249
250 // Get the value from the map (O(1) compile-time lookup)
251 return state[var_key];
252 }
253};

537template <CVar VarToBind, tmp::CTuple CapturedVars, auto Func,
538 typename... TailOps, typename StateMap>
539struct MIRExecutor<
540 std::tuple<LetClause<VarToBind, CppExpr<CapturedVars, Func>>, TailOps...>,
541 StateMap> {
542
543 template <CSemiNaiveDatabase DB>
544 static void execute(DB &db, const StateMap &state) {
545 // 1. Get the runtime values for the expression's dependencies
546 // We can now use 'CapturedVars' directly
547 auto dep_vals = GetPrefixValues<CapturedVars, StateMap>::get(state);
548 // 2. Call the lambda
549 // We can now use 'Func' directly
550 auto new_val = std::apply(Func, dep_vals);
551 // 3. Insert the new variable binding into the state map
552 auto new_state =
553 hana::insert(state, hana::make_pair(hana::type_c<VarToBind>, new_val));
554 // 4. Recurse with the new state
555 MIRExecutor<std::tuple<TailOps...>, decltype(new_state)>::execute(
556 db, new_state);
557 }

784 relation_delta.ensure_index(make_canonical_index_spec<Schema>());
