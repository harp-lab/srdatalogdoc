203template <typename Term, typename StateMap>
204constexpr auto get_term_value(const StateMap &state) {
205 if constexpr (CVar<Term>) {
206 // Case 1: Variable - get value from state map using type as key
207 return state[hana::type_c<Term>];
208 } else if constexpr (CConst<Term>) {
209 // Case 2: Constant - use compile-time constant value
210 return Term::value;
211 } else if constexpr (CCppExpr<Term>) {
212 // Case 3: C++ Expression - would evaluate expression
213 static_assert(!std::is_same_v<Term, Term>,
214 "CppExpr not yet supported in head position");
215 return 0; // unreachable
216 } else {
217 // Case 4: Unknown term type - compile error
218 static_assert(!std::is_same_v<Term, Term>, "Unknown term type");
219 return 0; // unreachable
220 }
221}

244template <CVar VarToFind, typename StateMap> struct GetValueFromState {
245 static decltype(auto) get(const StateMap &state) {
246 // Create a type key for the variable
247 constexpr auto var_key = hana::type_c<VarToFind>;
248
249 // Lookup in the map (compile-time check that key exists)
250 constexpr auto has_key = hana::contains(state, var_key);
251 static_assert(decltype(has_key)::value,
252 "Variable not found in bound state. Check that the variable "
253 "is bound before use.");
254
255 // Get the value from the map (O(1) compile-time lookup)
256 return state[var_key];
257 }
258};

503// --- Specialization for ColJoin ---
504template <CVar VarToBind, tmp::CTuple Sources, typename... TailOps,
505 typename StateMap>
506struct MIRExecutor<std::tuple<mir::ColumnJoin<VarToBind, Sources>, TailOps...>,
507 StateMap> {
508
509 template <CSemiNaiveDatabase DB>
510 static void execute(DB &db, const StateMap &state) {
511 // std::cout << "ColumnJoin: " <<
512 // get_type_name<std::tuple<mir::ColumnJoin<VarToBind, Sources>>>() <<
513 // std::endl;
514 // 1. Generate all iterators for the intersection
515 // auto iter_tuple = GenerateIterators<Sources, DB, StateMap>::get(db,
516 // state);
517 auto iter_tuple = [&]<std::size_t... Is>(std::index_sequence<Is...>) {
518 return std::make_tuple(GenerateIterator<std::tuple_element_t<Is, Sources>,
519 DB, StateMap>::get(db, state)...);
520 }(std::make_index_sequence<std::tuple_size_v<Sources>>{});
521
522 // 2. Call hash_intersect(iter1, iter2, ...)
523 auto intersection = hash_intersect(iter_tuple);
524
525 // std::cout << "ColumnJoin: " << var_to_string<VarToBind>()
526 // << ", intersection size check..." << std::endl;
527
528 // 3. This is the 'for' loop
529 for (const auto &new_val : intersection) {
530 // std::cout << "new_val: " << var_to_string<VarToBind>() << " = " <<
531 // new_val
532 // << std::endl;
533
534 // 4. Insert the new variable binding into the state map
535 // Use hana::type_c<VarToBind> as the key and new_val as the value
536 auto new_state = hana::insert(
537 state, hana::make_pair(hana::type_c<VarToBind>, new_val));
538
539 // 5. Recurse on the *rest* of the plan with the new state
540 // std::cout << "TailOps type: " <<
541 // get_type_name<std::tuple<TailOps...>>() << std::endl;
542 MIRExecutor<std::tuple<TailOps...>, decltype(new_state)>::execute(
543 db, new_state);
544 }
545 }

563// --- Specialization for LetClause ---
564template <CVar VarToBind, tmp::CTuple CapturedVars, auto Func,
565 typename... TailOps, typename StateMap>
566struct MIRExecutor<
567 std::tuple<LetClause<VarToBind, CppExpr<CapturedVars, Func>>, TailOps...>,
568 StateMap> {
569
570 template <CSemiNaiveDatabase DB>
571 static void execute(DB &db, const StateMap &state) {
572 // 1. Get the runtime values for the expression's dependencies
573 // We can now use 'CapturedVars' directly
574 auto dep_vals = GetPrefixValues<CapturedVars, StateMap>::get(state);
575 // 2. Call the lambda
576 // We can now use 'Func' directly
577 auto new_val = std::apply(Func, dep_vals);
578 // 3. Insert the new variable binding into the state map
579 auto new_state =
580 hana::insert(state, hana::make_pair(hana::type_c<VarToBind>, new_val));
581 // 4. Recurse with the new state
582 MIRExecutor<std::tuple<TailOps...>, decltype(new_state)>::execute(
583 db, new_state);
584 }
585};
