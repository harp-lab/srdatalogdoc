196template <typename StateMap, typename Var>
197concept CStateMap =
198 CVar<Var> &&
199 requires(const StateMap &state) {
200 { state[boost::hana::type_c<Var>] };
201 {
202 boost::hana::contains(state, boost::hana::type_c<Var>)
203 } -> std::convertible_to<bool>;
204 { boost::hana::keys(state) };
205 } &&
206 ColumnElement<std::remove_cvref_t<
207 decltype(std::declval<const StateMap &>()[boost::hana::type_c<Var>])>>;

222template <typename StateMap>
223concept CStateMapBasic = requires(const StateMap &state) {
224 // Must support key iteration (returns hana::tuple of keys)
225 { boost::hana::keys(state) };
226
227 // Must support size query (returns hana::size_t or similar)
228 { boost::hana::length(state) };
229};

256template <typename Term, CStateMapBasic StateMap>
257constexpr auto get_term_value(const StateMap &state) {
258 if constexpr (CVar<Term>) {
259 // Case 1: Variable - get value from state map using type as key
260 return state[hana::type_c<Term>];
261 } else if constexpr (CConst<Term>) {
262 // Case 2: Constant - use compile-time constant value
263 return Term::value;
264 } else if constexpr (CCppExpr<Term>) {
265 // Case 3: C++ Expression - would evaluate expression
266 static_assert(!std::is_same_v<Term, Term>,
267 "CppExpr not yet supported in head position");
268 return 0; // unreachable
269 } else {
270 // Case 4: Unknown term type - compile error
271 static_assert(!std::is_same_v<Term, Term>, "Unknown term type");
272 return 0; // unreachable
273 }
274}

297template <CVar VarToFind, CStateMap<VarToFind> StateMap>
298struct GetValueFromState {
299 static decltype(auto) get(const StateMap &state) {
300 // Create a type key for the variable
301 constexpr auto var_key = hana::type_c<VarToFind>;
302
303 // Lookup in the map (compile-time check that key exists)
304 constexpr auto has_key = hana::contains(state, var_key);
305 static_assert(decltype(has_key)::value,
306 "Variable not found in bound state. Check that the variable "
307 "is bound before use.");
308
309 // Get the value from the map (O(1) compile-time lookup)
310 return state[var_key];
311 }
312};

548template <CVar VarToBind, tmp::CTuple Sources, typename... TailOps,

550struct MIRExecutor<std::tuple<mir::ColumnJoin<VarToBind, Sources>, TailOps...>,
551 StateMap> {
552
553 template <CSemiNaiveDatabase DB>
554 static void execute(DB &db, const StateMap &state) {
555 // std::cout << "ColumnJoin: " <<
556 // get_type_name<std::tuple<mir::ColumnJoin<VarToBind, Sources>>>() <<
557 // std::endl;
558 // 1. Generate all iterators for the intersection
559 // auto iter_tuple = GenerateIterators<Sources, DB, StateMap>::get(db,
560 // state);
561 auto iter_tuple = [&]<std::size_t... Is>(std::index_sequence<Is...>) {
562 return std::make_tuple(GenerateIterator<std::tuple_element_t<Is, Sources>,
563 DB, StateMap>::get(db, state)...);
564 }(std::make_index_sequence<std::tuple_size_v<Sources>>{});
565
566 // 2. Call column_intersect(iter1, iter2, ...)
567 auto intersection = column_intersect(iter_tuple);
568
569 // std::cout << "ColumnJoin: " << var_to_string<VarToBind>()
570 // << ", intersection size check..." << std::endl;
571
572 // 3. This is the 'for' loop
573 for (const auto &new_val : intersection) {
574 // std::cout << "new_val: " << var_to_string<VarToBind>() << " = " <<
575 // new_val
576 // << std::endl;
577
578 // 4. Insert the new variable binding into the state map
579 // Use hana::type_c<VarToBind> as the key and new_val as the value
580 auto new_state = hana::insert(
581 state, hana::make_pair(hana::type_c<VarToBind>, new_val));
582
583 // 5. Recurse on the *rest* of the plan with the new state
584 // std::cout << "TailOps type: " <<
585 // get_type_name<std::tuple<TailOps...>>() << std::endl;
586 MIRExecutor<std::tuple<TailOps...>, decltype(new_state)>::execute(

588 }
