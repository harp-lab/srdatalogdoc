25 /**
26  * @brief Context structure to hold reusable buffers for sorting operations
27  *
28  * This context is slightly simpler than Highway's SortContext because we don't need to
29  * manually push the index into the context, sort_by_key works well on GPU.
30  *
31  * The context maintains device-side buffers that are reused across multiple
32  * sorting operations to avoid repeated allocations.
33  */
34 template <typename I = uint32_t>
35 struct SortContext {
36  DeviceArray<I> permutation; ///< Permutation buffer (indices)
37 
38  /**
39  * @brief Resize permutation buffer to accommodate n elements
40  * @param n Number of elements to sort
41  */
42  void resize_permutation(size_t n) {
43  permutation.resize(n);
44  }
45 
46  /**
47  * @brief Get pointer to sorted indices after sorting
48  * @param _n Just for match what's in sort.h (unused)
49  * @return Pointer to sorted indices on device
50  */
51  const I* get_sorted_indices([[maybe_unused]] size_t _n) const {
52  return permutation.data();
53  }
54 
55  /**
56  * @brief Get temporary buffer for keys of type KeyType
57  * @tparam KeyType Key type
58  * @param n Number of elements needed
59  * @return Reference to DeviceArray<KeyType> with at least n elements
60  *
61  */
62  template <typename KeyType>
63  DeviceArray<KeyType>& get_temp_keys(size_t n) {
64  // Each KeyType gets its own thread-local buffer via template instantiation
65  // This is type-safe, simple, and efficient - no type erasure needed
66  static thread_local DeviceArray<KeyType> temp;
67  if (temp.size() < n) {
68  temp.resize(n);
69  }
70  return temp;
71  }
72 };

74 /**
75  * @brief Update permutation by sorting keys
76  * @tparam K Key type
77  * @tparam I Index type
78  * @param keys Pointer to key data on device
79  * @param n Number of elements
80  * @param permutation Pointer to permutation indices on device
81  * @param ctx SortContext to manage temporary buffers
82  */
83 template <typename K, typename I = uint32_t>
84 void update_permutation(K* keys, size_t n, I* permutation, SortContext<I>& ctx) {
85  // Get temporary storage for keys from context
86  auto& temp = ctx.template get_temp_keys<K>(n);
87 
88  // Permute the keys with the current reordering
89  thrust::gather(thrust::device, thrust::device_ptr<I>(permutation),
90  thrust::device_ptr<I>(permutation) + n, thrust::device_ptr<K>(keys), temp.begin());
91 
92  // Stable sort the permuted keys and update the permutation
93  thrust::stable_sort_by_key(rmm::exec_policy(), temp.begin(), temp.begin() + n,
94  thrust::device_ptr<I>(permutation));
95 }

97 /**
98  * @brief Apply permutation to keys
99  * @tparam K Key type
100  * @tparam I Index type
101  * @param keys Pointer to key data on device (will be permuted in-place)
102  * @param n Number of elements
103  * @param permutation Pointer to permutation indices on device
104  * @param ctx SortContext to manage temporary buffers
105  */
106 template <typename K, typename I = uint32_t>
107 void apply_permutation(K* keys, size_t n, I* permutation, SortContext<I>& ctx) {
108  // Get temporary storage for keys from context
109  auto& temp = ctx.template get_temp_keys<K>(n);
110 
111  // Copy keys to temporary buffer
112  thrust::copy(thrust::device, thrust::device_ptr<K>(keys), thrust::device_ptr<K>(keys) + n,
113  temp.begin());
114 
115  // Permute the keys
116  thrust::gather(thrust::device, thrust::device_ptr<I>(permutation),
117  thrust::device_ptr<I>(permutation) + n, temp.begin(), thrust::device_ptr<K>(keys));
118 }

120 /**
121  * @brief Stable lexicographic sort implementation (internal) with context - GPU version
122  *
123  * GPU-accelerated version using Thrust. See hwy_lexsort::stable_lex_sort_impl in sort.h
124  * for detailed documentation.
125  *
126  * @tparam I Index type (default: uint32_t)
127  * @tparam Ts... Column data types
128  * @param n Number of rows to sort
129  * @param order Span of column indices specifying sort priority
130  * @param ctx Reference to SortContext for buffer reuse and result storage
131  * @param reorder_data If true, reorder the column data in place using sorted indices
132  * @param cols... Pointers to column data arrays (one per column type in Ts...)
133  */
134 template <typename I = uint32_t, typename... Ts>
135 void stable_lex_sort_impl(size_t n, std::span<const int> order, SortContext<I>& ctx,
136  bool reorder_data, Ts*... cols) {
137  if (n == 0) {
138  return;
139  }
140 
141  ctx.resize_permutation(n);
142 
143  // Initialize permutation to [0, 1, 2, ..., n-1]
144  I* permutation = const_cast<I*>(ctx.permutation.data());
145  thrust::sequence(thrust::device, thrust::device_ptr<I>(permutation),
146  thrust::device_ptr<I>(permutation) + n);
147 
148  // Store column pointers in a tuple for indexed access
149  std::tuple<Ts*...> col_tuple(cols...);
150  constexpr size_t num_cols = sizeof...(Ts);
151 
152  // Sort from least significant column to most significant column
153  // Process order in reverse (least significant first)
154  for (int col_idx : std::ranges::reverse_view(order)) {
155  if (col_idx < 0 || static_cast<size_t>(col_idx) >= num_cols) {
156  continue; // Skip invalid column indices
157  }
158 
159  // Extract the column pointer by index and call update_permutation
160  [&]<size_t... Is>(std::index_sequence<Is...>) {
161  ((col_idx == static_cast<int>(Is) &&
162  (update_permutation(std::get<Is>(col_tuple), n, permutation, ctx), true)),
163  ...);
164  }(std::make_index_sequence<num_cols>{});
165  }
166 
167  // Optionally apply permutation to columns (for in-place reordering)
168  if (reorder_data) {
169  [&]<size_t... Is>(std::index_sequence<Is...>) {
170  (apply_permutation(std::get<Is>(col_tuple), n, permutation, ctx), ...);
171  }(std::make_index_sequence<num_cols>{});
172  }
173 }

189 template <typename I = uint32_t, typename... Ts>
190 void stable_lex_sort_impl(I* ids_out, size_t n, std::span<const int> order, bool reorder_data,
