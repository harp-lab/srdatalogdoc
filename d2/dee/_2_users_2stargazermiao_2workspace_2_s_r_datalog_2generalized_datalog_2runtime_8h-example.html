Extract incremental relations from a tuple of rules.

Extract incremental relations from a tuple of rules. This trait extracts all relation schemas that appear in the head of the given rules. These are the relations that will be updated during rule execution and need delta tracking for semi-naive evaluation.

// Look up each Var in the state map and return as std::tuple directly
return std::make_tuple(state[hana::type_c<Vars>]...);

template <typename Term, typename StateMap>
constexpr auto get_term_value(const StateMap &state) {
if constexpr (CVar<Term>) {
// Case 1: Variable - get value from state map using type as key
return state[hana::type_c<Term>];
} else if constexpr (CConst<Term>) {
// Case 2: Constant - use compile-time constant value
return Term::value;
} else if constexpr (CCppExpr<Term>) {
// Case 3: C++ Expression - would evaluate expression
static_assert(!std::is_same_v<Term, Term>,
"CppExpr not yet supported in head position");
return 0; // unreachable
} else {
// Case 4: Unknown term type - compile error
static_assert(!std::is_same_v<Term, Term>, "Unknown term type");
return 0; // unreachable
}
}
} // namespace state

template <CVar VarToFind, typename StateMap> struct GetValueFromState {
static decltype(auto) get(const StateMap &state) {
// Create a type key for the variable
constexpr auto var_key = hana::type_c<VarToFind>;
// Lookup in the map (compile-time check that key exists)
constexpr auto has_key = hana::contains(state, var_key);
static_assert(decltype(has_key)::value,
"Variable not found in bound state. Check that the variable "
"is bound before use.");
// Get the value from the map (O(1) compile-time lookup)
return state[var_key];
}
};

template <mir::CColumnSource Source, CSemiNaiveDatabase DB, typename StateMap>
struct GenerateIterator;
template <CRelationSchema Schema, int... Cols, std::size_t Ver,
tmp::CTuple PrefixVars, CSemiNaiveDatabase DB, typename StateMap>
struct GenerateIterator<
mir::ColumnSource<
IndexSpecT<Schema, std::integer_sequence<int, Cols...>, Ver>,
PrefixVars>,
DB, StateMap> {
static auto get(DB &db, const StateMap &state) {
// 1. Get the runtime relation (e.g., 'r') from the DB
// Version is extracted from IndexSpec
auto &rel = get_relation_by_schema<Schema, Ver, DB>(db);
// 2. lazy build index, generate if really needed
auto &index = rel.ensure_index({{Cols...}});
// 3. Get the root iterator (equiv. to INDEX_ROOT)
auto root_iter = index.root();
// 4. Get the runtime prefix values from state map (e.g., {5, 10})
auto prefix_vals = GetPrefixValues<PrefixVars, StateMap>::get(state);
// print the prefix
// std::cout << "prefix_vals: ";
// std::apply(
// [&](auto... vals) {
// std::cout << "(";
// int n = 0;
// ((std::cout << (n++ ? ", " : "") << vals), ...);
// std::cout << ") ";
// },
// prefix_vals);
// std::cout << std::endl;
// print_state(state);
// 5. Apply the prefix (equiv. to INDEX_AT)
// We use std::apply to call prefix(iter, val1, val2...)
// Compiler need support for inline lambda for apply
auto prefixed_iter = std::apply(
[&](auto... vals) { return root_iter.prefix({vals...}); }, prefix_vals);
return prefixed_iter;
}
};

template <typename MIRPlan, typename StateMap> struct MIRExecutor;
// --- Base Case: No more operations ---
template <typename StateMap> struct MIRExecutor<std::tuple<>, StateMap> {
template <CSemiNaiveDatabase DB> static void execute(DB &, const StateMap &) {
// Done
}
};
// --- Recursive Case: Process the HeadOp ---
template <typename HeadOp, typename... TailOps, typename StateMap>
struct MIRExecutor<std::tuple<HeadOp, TailOps...>, StateMap> {
// This is the recursive call for the *rest* of the plan
template <CSemiNaiveDatabase DB>
static void execute_tail(DB &db, const StateMap &state) {
MIRExecutor<std::tuple<TailOps...>, StateMap>::execute(db, state);
}
};
// --- Specialization for ColJoin ---
template <CVar VarToBind, tmp::CTuple Sources, typename... TailOps,
typename StateMap>
struct MIRExecutor<std::tuple<mir::ColumnJoin<VarToBind, Sources>, TailOps...>,
StateMap> {
template <CSemiNaiveDatabase DB>
static void execute(DB &db, const StateMap &state) {
// 1. Generate all iterators for the intersection
// auto iter_tuple = GenerateIterators<Sources, DB, StateMap>::get(db,
// state);
auto iter_tuple = [&]<std::size_t... Is>(std::index_sequence<Is...>) {
return std::make_tuple(GenerateIterator<std::tuple_element_t<Is, Sources>,
DB, StateMap>::get(db, state)...);
}(std::make_index_sequence<std::tuple_size_v<Sources>>{});
// 2. Call hash_intersect(iter1, iter2, ...)
auto intersection = hash_intersect(iter_tuple);
// std::cout << "ColumnJoin: " << var_to_string<VarToBind>()
// << ", intersection size check..." << std::endl;
// 3. This is the 'for' loop
for (const auto &new_val : intersection) {
// std::cout << "new_val: " << var_to_string<VarToBind>() << " = " <<
// new_val
// << std::endl;
// 4. Insert the new variable binding into the state map
// Use hana::type_c<VarToBind> as the key and new_val as the value
auto new_state = hana::insert(
state, hana::make_pair(hana::type_c<VarToBind>, new_val));
// 5. Recurse on the *rest* of the plan with the new state
// std::cout << "TailOps type: " <<
// get_type_name<std::tuple<TailOps...>>() << std::endl;
MIRExecutor<std::tuple<TailOps...>, decltype(new_state)>::execute(
db, new_state);
}
}
};
// --- Specialization for LetClause ---
template <CVar VarToBind, tmp::CTuple CapturedVars, auto Func,
typename... TailOps, typename StateMap>
struct MIRExecutor<
std::tuple<LetClause<VarToBind, CppExpr<CapturedVars, Func>>, TailOps...>,
StateMap> {
template <CSemiNaiveDatabase DB>
static void execute(DB &db, const StateMap &state) {
// 1. Get the runtime values for the expression's dependencies
// We can now use 'CapturedVars' directly
auto dep_vals = GetPrefixValues<CapturedVars, StateMap>::get(state);
// 2. Call the lambda
// We can now use 'Func' directly
auto new_val = std::apply(Func, dep_vals);
// 3. Insert the new variable binding into the state map
auto new_state =
hana::insert(state, hana::make_pair(hana::type_c<VarToBind>, new_val));
// 4. Recurse with the new state
MIRExecutor<std::tuple<TailOps...>, decltype(new_state)>::execute(
db, new_state);
}
};
// --- Specialization for DestinationRelation ---
template <CRelationSchema Schema, tmp::CTuple Terms, std::size_t Ver,
typename... TailOps, typename StateMap>
struct MIRExecutor<
std::tuple<mir::DestinationRelation<Schema, Terms, Ver>, TailOps...>,
StateMap> {
template <CSemiNaiveDatabase DB>
static void execute(DB &db, const StateMap &state) {
// 1. Get the runtime relation
auto &rel = get_relation_by_schema<Schema, Ver>(db);
// 2. Get the runtime values for all terms (Vars and Consts)
auto payload_vals = GetPayloadValues<Terms, StateMap>::get(state);
// 3. Call push_row( {val1, val2...}, semiring_val )`
// (Assuming a default semiring value for simplicity)
auto sr_val = Schema::semiring_type::one();
std::apply(
[&](auto... vals) {
// check if the values already exist in the relation
// get the prefix
// TODO: this API need to be improved
const auto &canonical_index =
rel.ensure_index(make_canonical_index_spec<Schema>());
if (lookup_prefix(canonical_index, Prefix<decltype(vals)...>{vals...})
.valid()) {
// the values already exist in the relation
rel.push_row({vals...}, sr_val);
}
},
payload_vals);
// std::cout << "pushed row to relation : ";
// std::apply(
// [&](auto... vals) {
// std::cout << "(";
// int n = 0;
// ((std::cout << (n++ ? ", " : "") << vals), ...);
// std::cout << ") ";
// },
// payload_vals);
// std::cout << std::endl;
// 4. Recurse on the rest of the plan (e.g., another 'Head' clause)
MIRExecutor<std::tuple<TailOps...>, StateMap>::execute(db, state);
}
};

template <CRelationSchema Schema, AST::CSemiNaiveDatabase DB>
std::size_t compute_delta(DB &db) {
// get the relation from the database
auto &relation_full = get_relation_by_schema<Schema, FULL_VER>(db);
auto &relation_delta = get_relation_by_schema<Schema, DELTA_VER>(db);
auto &relation_newt = get_relation_by_schema<Schema, NEW_VER>(db);
// ensure canonical index is built
std::swap(relation_delta, relation_newt);
// clean the delta relation
// insert the delta relation into the full relation via copy
relation_full.reserve(relation_delta.size());
relation_newt.clear();
relation_delta.ensure_index(make_canonical_index_spec<Schema>());
relation_full.concat(relation_delta);
relation_full.ensure_index(make_canonical_index_spec<Schema>());
// TODO: let all the other build by demand?
// relation_full.build_all_indexes();
// relation_delta.build_all_indexes();
return relation_delta.size();
}
template <typename RelationList, AST::CSemiNaiveDatabase DB>
bool compute_delta_for_ruleset(DB &db) {
std::size_t total_delta = 0;
mp::mp_for_each<RelationList>([&]<typename Schema>(Schema) {
total_delta += compute_delta<Schema>(db);
});
return total_delta > 0;
}
template <CRelationSchema Schema, AST::CSemiNaiveDatabase DB>
void load_full_to_delta(DB &db) {
auto &relation_full = get_relation_by_schema<Schema, FULL_VER>(db);
auto &relation_delta = get_relation_by_schema<Schema, DELTA_VER>(db);
relation_full.clone_into(relation_delta);
relation_delta.ensure_index(make_canonical_index_spec<Schema>());
relation_delta.build_all_indexes();
}

template <CQuery Q, AST::CSemiNaiveDatabase DB> void execute_query(DB &db) {
if constexpr (CFixpoint<Q>) {
execute_rules_for_fixpoint<Q>(db);
} else if constexpr (CNonIterativeRuleSets<Q>) {
execute_rules_for_non_iterative_rule_sets<Q>(db);
} else {
static_assert(false, "Unknown query type");
}
}
