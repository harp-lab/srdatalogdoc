41// template helper to dispatch by column index
42// It is very common pattern when you construct lexical order on tuple type
43// and you want to custom the order of comparison.
44template <class Rel, std::size_t Arity = Rel::arity>
45inline void dispatch_by_column(std::size_t k, auto &&fn) {
46 [&]<std::size_t... I>(std::index_sequence<I...>) {
47 bool found = false;
48#ifdef __clang__
49#pragma clang diagnostic push
50#pragma clang diagnostic ignored "-Wunused-value"
51#elif defined(__GNUC__)
52#pragma GCC diagnostic push
53#pragma GCC diagnostic ignored "-Wunused-value"
54#endif
55 ((found ||
56 (k == I ? (found = true, fn.template operator()<I>(), false) : false)),
57 ...);
58#ifdef __clang__
59#pragma clang diagnostic pop
60#elif defined(__GNUC__)
61#pragma GCC diagnostic pop
62#endif
63 (void)found; // suppress unused warning if all branches handled
64 }(std::make_index_sequence<Arity>{});
65}

135// Build a ProjectedView on selected indices <Idx...>, with optional row filter.
136template <Semiring SR, ColumnElement... AttrTs>
137template <int... Idx>
138inline auto
139Relation<SR, AttrTs...>::project(const std::span<const RowId> &row_ids) const {
140 const std::size_t logical_size =
141 (row_ids.empty()) ? this->size() : row_ids.size();
142 auto base_indices = std::ranges::views::iota(0u, logical_size);
143 auto make_col_view = [base_indices, row_ids](const auto *col_ptr) {
144 auto indirection_fn = [col_ptr, row_ids](std::size_t i) -> const auto & {
145 const std::size_t phys_idx = row_ids.empty() ? i : row_ids[i];
146 return (*col_ptr)[phys_idx];
147 };
148
149 return std::ranges::views::transform(base_indices, indirection_fn);
150 };
151
152 auto ann_view = [&]() {
153 auto indirection_fn = [this,
154 row_ids](std::size_t i) -> const AnnotationVal & {
155 const std::size_t phys_idx = row_ids.empty() ? i : row_ids[i];
156 return this->ann_[phys_idx];
157 };
158 return std::ranges::views::transform(base_indices, indirection_fn);
159 }();
160
161 return std::ranges::views::zip(make_col_view(&this->column<Idx>())...,
162 ann_view);
163}

168template <Semiring SR, ColumnElement... AttrTs>
169struct Relation<SR, AttrTs...>::HashTrieIndex::Impl {
170 using RowId = typename Relation<SR, AttrTs...>::RowId;
171
172 struct Node {
173 // non-leaf nodes
174 // std::unordered_map<std::size_t, uint32_t> children;
175 Map<std::size_t, uint32_t> children;
176 // key kache for the non-leaf node values
177 // std::vector<std::size_t> keys_cache;
178 Vector<std::size_t> keys_cache;
179
180 // when comes to the leaf value, the row ids
181 Vector<RowId> postings;
182
183 uint32_t depth{0}; // root
184 bool is_leaf() const noexcept {
185 return !postings.empty() || children.empty();
186 }
187 std::size_t degree() const noexcept { return keys_cache.size(); }
188 bool contains(std::size_t k) const noexcept {
189 return children.find(k) != children.end();
190 }
191 };
192
193 // vector of nodes
194 Vector<Node> nodes;
195 // the degree of the current node index
196 uint32_t arity{0};
197
198 Impl() {
199 nodes.reserve(1024);
200 nodes.push_back(Node{});
201 } // root
202
203 Impl(boost::container::pmr::memory_resource *resource) {
204 nodes = Vector<Node>(resource);
205 nodes.reserve(1024);
206 nodes.push_back(Node{});
207 }
208 uint32_t root_id() const noexcept { return 0; }
209
210 uint32_t ensure_child(uint32_t nid, std::size_t key, uint32_t child_depth) {
211 auto &n = nodes[nid];
212 auto it = n.children.find(key);
213 if (it != n.children.end())
214 return it->second;
215 const uint32_t cid = static_cast<uint32_t>(nodes.size());
216 nodes.push_back(Node{});
217 nodes.back().depth = child_depth;
218 n.children.emplace(key, cid);
219 return cid;
220 }
221
222 uint32_t find_child(uint32_t nid, std::size_t key) const {
223 const auto &n = nodes[nid];
224 auto it = n.children.find(key);
225 return (it == n.children.end()) ? UINT32_MAX : it->second;
226 }
227
228 void finalize_keys_cache() {
229 // inflate the key cache for all non-leaf node
230 // I am using `|` on ranges to chain the views together
231 for (auto &n : nodes) {
232 n.keys_cache =
233 n.children | std::ranges::views::keys | SRDatalog::to<Vector>();
234 // at here, we are not aiming the sorted
235 }
236 }
237
238 // approximate computation of the memory
239 std::size_t bytes_used() const {
240 std::size_t b = sizeof(*this) + nodes.capacity() * sizeof(Node);
241 for (const auto &n : nodes) {
242 b += n.children.size() *
243 (sizeof(std::size_t) + sizeof(uint32_t) + sizeof(void *) * 2);
244 b += n.keys_cache.capacity() * sizeof(std::size_t);
245 b += n.postings.capacity() * sizeof(RowId);
246 }
247 return b;
248 }
249};

403template <Semiring SR, ColumnElement... AttrTs>
404void Relation<SR, AttrTs...>::HashTrieIndex::build(Relation &owner,
405 const IndexSpec &spec,
406 memory_resource *resource) {
407 // encode if needed (we have multiple indices, so could be already encoded)
408 if (owner.interned_size() != owner.size()) {
409 auto old_size = owner.interned_size();
410 owner.resize_interned_columns(owner.size());
411 // encode the rows
412 mp::mp_for_each<mp::mp_iota_c<arity>>([&](auto I) {
413 auto &interned_col =
414 const_cast<Relation &>(owner).template interned_column<I>();
415 for (std::size_t r = old_size; r < owner.size(); ++r) {
416 interned_col[r] = encode_to_size_t(owner.template column<I>()[r]);
417 }
418 });
419 }
420 if (size() == owner.interned_size()) {
421 return;
422 }
423 if (size() > owner.interned_size()) {
424 throw std::runtime_error(
425 "build: relation size is greater than the interned size");
426 }
427 // construct the span of the encoded rows from trie size to owner size
428 auto encoded = owner.encoded_rows(size(), owner.interned_size());
429 // Convert const spans to non-const spans (safe since build_from_encoded only
430 // reads)
431 std::array<std::span<std::size_t>, arity> non_const_encoded;
432 [&]<std::size_t... Is>(std::index_sequence<Is...>) {
433 ((non_const_encoded[Is] = std::span<std::size_t>(
434 const_cast<std::size_t *>(encoded[Is].data()), encoded[Is].size())),
435 ...);
436 }(std::make_index_sequence<arity>{});
437 this->build_from_encoded(spec, non_const_encoded, resource);
438}

533template <Semiring SR, ColumnElement... AttrTs>
534void Relation<SR, AttrTs...>::HashTrieIndex::clone_from(
535 const HashTrieIndex &other, memory_resource *resource) {
536 // If source is empty, clear this index
537 if (!other.p_ || other.p_->nodes.empty()) {
538 if (resource) {
539 p_ = std::make_shared<Impl>(resource);
540 } else {
541 p_ = std::make_shared<Impl>();
542 }
543 size_ = 0;
544 return;
545 }
546
547 // Determine memory resource: use provided, or get from source, or use default
548 memory_resource *target_resource = resource;
549 if (!target_resource) {
550 // Try to get resource from source's nodes vector
551 // PMR containers expose get_allocator().resource()
552 target_resource = other.p_->nodes.get_allocator().resource();
553 if (!target_resource) {
554 target_resource = boost::container::pmr::get_default_resource();
555 }
556 }
557
558 // Create new Impl with target resource
559 p_ = std::make_shared<Impl>(target_resource);
560 p_->arity = other.p_->arity;
561 size_ = other.size_;
562
563 // Reserve space for nodes
564 p_->nodes.reserve(other.p_->nodes.size());
565
566 // Deep copy all nodes
567 for (const auto &src_node : other.p_->nodes) {
568 typename Impl::Node dst_node;
569 dst_node.depth = src_node.depth;
570
571 // Deep copy children map - construct with explicit resource
572 // Use the range constructor with allocator
573 dst_node.children = Map<std::size_t, uint32_t>(
574 src_node.children.begin(), src_node.children.end(),
575 src_node.children.size(),
576 boost::container::pmr::polymorphic_allocator<
577 std::pair<const std::size_t, uint32_t>>(target_resource));
578
579 // Deep copy keys_cache vector
580 dst_node.keys_cache = Vector<std::size_t>(target_resource);
581 dst_node.keys_cache.assign(src_node.keys_cache.begin(),
582 src_node.keys_cache.end());
583
584 // Deep copy postings vector
585 dst_node.postings = Vector<RowId>(target_resource);
586 dst_node.postings.assign(src_node.postings.begin(),
587 src_node.postings.end());
588
589 p_->nodes.push_back(std::move(dst_node));
590 }
591}
