42// template helper to dispatch by column index
43// It is very common pattern when you construct lexical order on tuple type
44// and you want to custom the order of comparison.
45template <class Rel, std::size_t Arity = Rel::arity>
46inline void dispatch_by_column(std::size_t k, auto &&fn) {
47 [&]<std::size_t... I>(std::index_sequence<I...>) {
48 bool found = false;
49#ifdef __clang__
50#pragma clang diagnostic push
51#pragma clang diagnostic ignored "-Wunused-value"
52#elif defined(__GNUC__)
53#pragma GCC diagnostic push
54#pragma GCC diagnostic ignored "-Wunused-value"
55#endif
56 ((found ||
57 (k == I ? (found = true, fn.template operator()<I>(), false) : false)),
58 ...);
59#ifdef __clang__
60#pragma clang diagnostic pop
61#elif defined(__GNUC__)
62#pragma GCC diagnostic pop
63#endif
64 (void)found; // suppress unused warning if all branches handled
65 }(std::make_index_sequence<Arity>{});
66}

140// Build a ProjectedView on selected indices <Idx...>, with optional row filter.
141template <Semiring SR, ColumnElementTuple AttrTuple>
142template <int... Idx>
143inline auto
144Relation<SR, AttrTuple>::project(const std::span<const RowId> &row_ids) const {
145 const std::size_t logical_size =
146 (row_ids.empty()) ? this->size() : row_ids.size();
147 auto base_indices = std::ranges::views::iota(0u, logical_size);
148 auto make_col_view = [base_indices, row_ids](const auto *col_ptr) {
149 auto indirection_fn = [col_ptr, row_ids](std::size_t i) -> const auto & {
150 const std::size_t phys_idx = row_ids.empty() ? i : row_ids[i];
151 return (*col_ptr)[phys_idx];
152 };
153
154 return std::ranges::views::transform(base_indices, indirection_fn);
155 };
156
157 auto ann_view = [&]() {
158 auto indirection_fn = [this,
159 row_ids](std::size_t i) -> const AnnotationVal & {
160 const std::size_t phys_idx = row_ids.empty() ? i : row_ids[i];
161 return this->ann_[phys_idx];
162 };
163 return std::ranges::views::transform(base_indices, indirection_fn);
164 }();
165
166 return std::ranges::views::zip(make_col_view(&this->column<Idx>())...,
167 ann_view);
168}

173template <Semiring SR, ColumnElementTuple AttrTuple>
174struct Relation<SR, AttrTuple>::HashTrieIndex::Impl {
175 using RowId = typename Relation<SR, AttrTuple>::RowId;
176
177 struct Node {
178 // non-leaf nodes
179 // std::unordered_map<std::size_t, uint32_t> children;
180 Map<std::size_t, uint32_t> children;
181 // key kache for the non-leaf node values
182 // std::vector<std::size_t> keys_cache;
183 Vector<std::size_t> keys_cache;
184
185 // when comes to the leaf value, the row ids
186 Vector<RowId> postings;
187
188 uint32_t depth{0}; // root
189 bool is_leaf() const noexcept {
190 return !postings.empty() || children.empty();
191 }
192 std::size_t degree() const noexcept { return keys_cache.size(); }
193 bool contains(std::size_t k) const noexcept {
194 return children.find(k) != children.end();
195 }
196 };
197
198 // vector of nodes
199 Vector<Node> nodes;
200 // the degree of the current node index
201 uint32_t arity{0};
202
203 Impl() {
204 nodes.reserve(1024);
205 nodes.push_back(Node{});
206 } // root
207
208 Impl(boost::container::pmr::memory_resource *resource) {
209 nodes = Vector<Node>(resource);
210 nodes.reserve(1024);
211 nodes.push_back(Node{});
212 }
213 uint32_t root_id() const noexcept { return 0; }
214
215 uint32_t ensure_child(uint32_t nid, std::size_t key, uint32_t child_depth) {
216 auto &n = nodes[nid];
217 auto it = n.children.find(key);
218 if (it != n.children.end())
219 return it->second;
220 const uint32_t cid = static_cast<uint32_t>(nodes.size());
221 nodes.push_back(Node{});
222 nodes.back().depth = child_depth;
223 n.children.emplace(key, cid);
224 return cid;
225 }
226
227 uint32_t find_child(uint32_t nid, std::size_t key) const {
228 const auto &n = nodes[nid];
229 auto it = n.children.find(key);
230 return (it == n.children.end()) ? UINT32_MAX : it->second;
231 }
232
233 void finalize_keys_cache() {
234 // inflate the key cache for all non-leaf node
235 // I am using `|` on ranges to chain the views together
236 for (auto &n : nodes) {
237 n.keys_cache =
238 n.children | std::ranges::views::keys | SRDatalog::to<Vector>();
239 // at here, we are not aiming the sorted
240 }
241 }
242
243 // approximate computation of the memory
244 std::size_t bytes_used() const {
245 std::size_t b = sizeof(*this) + nodes.capacity() * sizeof(Node);
246 for (const auto &n : nodes) {
247 b += n.children.size() *
248 (sizeof(std::size_t) + sizeof(uint32_t) + sizeof(void *) * 2);
249 b += n.keys_cache.capacity() * sizeof(std::size_t);
250 b += n.postings.capacity() * sizeof(RowId);
251 }
252 return b;
253 }
254
255 void clear() {
256 nodes.clear();
257 nodes.reserve(1024);
258 nodes.push_back(Node{});
259 }
260};

415void Relation<SR, AttrTuple>::HashTrieIndex::build(Relation &owner,
416 const IndexSpec &spec,
417 memory_resource *resource) {

459// build the indexing from pre-encoded rows
460template <Semiring SR, ColumnElementTuple AttrTuple>
461inline void Relation<SR, AttrTuple>::HashTrieIndex::build_from_encoded(
462 const IndexSpec &spec,
463 const std::array<std::span<const std::size_t>, arity> &encoded_rows,
464 memory_resource *resource) {
465 // constexpr std::size_t N = std::tuple_size_v<AttrTuple>;
466 auto row_size = encoded_rows[0].size();
467 if (row_size == 0) {
468 return; // Nothing to do
469 }
470 // Check if this is first build or incremental
471 bool needs_init = !p_ || p_->nodes.empty() || size_ == 0;
472
473 if (needs_init) {
474 // Initialize fresh trie
475 p_ = std::make_shared<Impl>(resource);
476 p_->nodes.clear();
477 p_->nodes.reserve(1024);
478 p_->nodes.push_back(typename Impl::Node{});
479 p_->arity = static_cast<uint32_t>(spec.cols.size());
480 size_ = 0;
481 } else {
482 // Validate for incremental build
483 if (p_->arity != spec.cols.size()) {
484 throw std::runtime_error("build_from_encoded: arity mismatch");
485 }
486 }
487
488 uint32_t root = p_->root_id();
489 std::size_t base_row_id = size_; // Current index size
490
491 // Insert all rows (works for both full and incremental)
492 for (std::size_t row_id = 0; row_id < row_size; ++row_id) {
493 uint32_t nid = root;
494 uint32_t depth = 0;
495
496 // Navigate/create path through trie
497 for (std::size_t col_idx : spec.cols) {
498 std::size_t key = encoded_rows[col_idx][row_id];
499 const uint32_t cid = p_->ensure_child(nid, key, depth + 1);
500 nid = cid;
501 ++depth;
502 }
503
504 // Append absolute row_id to leaf
505 p_->nodes[nid].postings.push_back(static_cast<RowId>(base_row_id + row_id));
506 }
507
508 // Regenerate key caches for all nodes
509 p_->finalize_keys_cache();
510
511 // Update size
512 size_ += row_size;
513}

551template <Semiring SR, ColumnElementTuple AttrTuple>
552void Relation<SR, AttrTuple>::HashTrieIndex::clone_from(
553 const HashTrieIndex &other, memory_resource *resource) {
554 // If source is empty, clear this index
555 if (!other.p_ || other.p_->nodes.empty()) {
556 if (resource) {
557 p_ = std::make_shared<Impl>(resource);
558 } else {
559 p_ = std::make_shared<Impl>();
560 }
561 size_ = 0;
562 return;
563 }
564
565 // Determine memory resource: use provided, or get from source, or use default
566 memory_resource *target_resource = resource;
567 if (!target_resource) {
568 // Try to get resource from source's nodes vector
569 // PMR containers expose get_allocator().resource()
570 target_resource = other.p_->nodes.get_allocator().resource();
571 if (!target_resource) {
572 target_resource = boost::container::pmr::get_default_resource();
573 }
574 }
575
576 // Create new Impl with target resource
577 p_ = std::make_shared<Impl>(target_resource);
578 p_->arity = other.p_->arity;
579 size_ = other.size_;
580
581 // Reserve space for nodes
582 p_->nodes.reserve(other.p_->nodes.size());
583
584 // Deep copy all nodes
585 for (const auto &src_node : other.p_->nodes) {
586 typename Impl::Node dst_node;
587 dst_node.depth = src_node.depth;
588
589 // Deep copy children map - construct with explicit resource
590 // Use the range constructor with allocator
591 dst_node.children = Map<std::size_t, uint32_t>(
592 src_node.children.begin(), src_node.children.end(),
593 src_node.children.size(),
594 boost::container::pmr::polymorphic_allocator<
595 std::pair<const std::size_t, uint32_t>>(target_resource));
596
597 // Deep copy keys_cache vector
598 dst_node.keys_cache = Vector<std::size_t>(target_resource);
599 dst_node.keys_cache.assign(src_node.keys_cache.begin(),
600 src_node.keys_cache.end());
601
602 // Deep copy postings vector
603 dst_node.postings = Vector<RowId>(target_resource);
604 dst_node.postings.assign(src_node.postings.begin(),
605 src_node.postings.end());
606
607 p_->nodes.push_back(std::move(dst_node));
608 }
609}
