31 /// @brief Concept for NodeHandle types from index implementations.
32 /// @details Detects types that provide the NodeHandle interface for navigating
33 /// trie structures. This enables better IDE support and type checking
34 /// in generic code.
35 /// @note **C++20 feature**: Uses C++20 concepts with `requires` expressions to
36 /// check interface requirements, including `std::ranges::range` concept
37 /// for range-based operations.
38 /// @tparam T The type to check
39 template <typename T>
40 concept CNodeHandle = requires(const T& handle) {
41  { handle.valid() } -> std::convertible_to<bool>;
42  // Allow both CPU-style is_leaf() and GPU-style is_leaf(view)
43  (
44  requires {
45  { handle.is_leaf() } -> std::convertible_to<bool>;
46  } ||
47  requires(const typename T::View& view) {
48  { handle.is_leaf(view) } -> std::convertible_to<bool>;
49  });
50  { handle.depth() } -> std::convertible_to<uint32_t>;
51  // Check for range-like operations (begin/end) instead of std::ranges::range
52  // This works for both std::span, cuda::std::span, and other range types
53  // Allow both CPU-style values() and GPU-style values(view)
54  (
55  requires {
56  requires requires(decltype(handle.values()) r) {
57  { r.begin() };
58  { r.end() };
59  };
60  } ||
61  requires(const typename T::View& view) {
62  requires requires(decltype(handle.values(view)) r) {
63  { r.begin() };
64  { r.end() };
65  };
66  });
67  // Allow both CPU-style rows() and GPU-style rows(view)
68  (
69  requires {
70  requires requires(decltype(handle.rows()) r) {
71  { r.begin() };
72  { r.end() };
73  };
74  } ||
75  requires(const typename T::View& view) {
76  requires requires(decltype(handle.rows(view)) r) {
77  { r.begin() };
78  { r.end() };
79  };
80  });
81  // prefix and contains_value should accept the value type from the values() range
82  // For spans, use element_type; for other ranges, use range_value_t
83  // Handle both CPU-style (values()) and GPU-style (values(view))
84  requires(
85  requires(const T& h) {
86  // Try element_type first (for spans), then fall back to range_value_t
87  requires(
88  requires { typename std::remove_cvref_t<decltype(h.values())>::element_type; } ||
89  requires { typename std::ranges::range_value_t<decltype(h.values())>; });
90  } ||
91  requires(const T& h, const typename T::View& view) {
92  // GPU-style: get value type from values(view)
93  requires(
94  requires { typename std::remove_cvref_t<decltype(h.values(view))>::element_type; } ||
95  requires { typename std::ranges::range_value_t<decltype(h.values(view))>; });
96  });
97  // Check prefix and contains_value with the appropriate value type
98  // Allow both CPU-style prefix(key) and GPU-style template prefix(key, tile, view)
99  // For GPU handles with template prefix, we can't fully validate in a concept
100  // so we relax the check for device handles
101  requires(
102  // CPU-style: prefix(key) returns same type
103  requires(const T& h, typename std::remove_cvref_t<decltype(h.values())>::element_type key) {
104  { h.prefix(key) } -> std::same_as<T>;
105  { h.contains_value(key) } -> std::convertible_to<bool>;
106  } ||
107  requires(const T& h, std::ranges::range_value_t<decltype(h.values())> key) {
108  { h.prefix(key) } -> std::same_as<T>;
109  { h.contains_value(key) } -> std::convertible_to<bool>;
110  } ||
111  // Alternative GPU-style check: has View type and prefix takes view
112  requires(T& h, const typename T::View& view) {
113  typename T::View;
114  // Just check that View type exists, actual prefix validation is in CDeviceHandle
115  });
116  // degree() can return uint32_t or std::size_t
117  { handle.degree() } -> std::convertible_to<std::size_t>;
118  // unique_count returns the number of unique values (size of values() view)
119  // Can return uint32_t or std::size_t
120  { handle.unique_count() } -> std::convertible_to<std::size_t>;
121 };

133 /// @brief Concept for index-like types with trie navigation interface.
134 /// @note **C++20 feature**: Uses nested `requires` clauses (requires CNodeHandle)
135 /// and `requires` expressions to check for required types and operations.
136 /// @details This concept defines the interface that any index type must satisfy to be used
137 /// with Relation. It checks for:
138 /// - Required nested types (NodeHandle, RowId, RowRange, ValueRange)
139 /// - Required query methods (root(), bytes_used(), size(), clear(), empty(),
140 /// prefix_lookup())
141 /// - Required build methods (build() for constructing the index)
142 /// - Required merge method (merge() for merging another index into this one)
143 /// - NodeHandle must satisfy CNodeHandle concept
144 /// @note This is in the main namespace to allow for future index implementations beyond
145 /// HashTrieIndex
146 template <typename T>
147 concept IndexLike =
148  requires(const T& idx, T& non_const_idx, memory_resource* resource) {
149  typename T::NodeHandle;
150  typename T::RowId;
151  typename T::RowRange;
152  typename T::ValueRange;
153  // ValueRange should be a range (span or view) of some value type
154  // For views, we check range_value_t instead of element_type
155  requires std::ranges::range<typename T::ValueRange>;
156  requires std::is_same_v < std::ranges::range_value_t<typename T::ValueRange>,
157  std::conditional_t < requires {typename T::ValueRange::element_type;
158  },
159  typename T::ValueRange::element_type,
160  std::ranges::range_value_t < typename T::ValueRange >>>
161  ;
162 requires CNodeHandle<typename T::NodeHandle>;
163 {idx.root()}->std::same_as<typename T::NodeHandle>;
164 {idx.bytes_used()}->std::convertible_to<std::size_t>;
165 {idx.size()}->std::convertible_to<std::size_t>;
166 {non_const_idx.clear()}->std::same_as<void>;
167 {idx.empty()}->std::convertible_to<bool>;
168 // Check that the Index type has a variadic static intersect method
169 // This is used for set intersection of columns in WCOJ
170 // Check with 2 arguments (minimum for intersection)
171 requires requires(const typename T::NodeHandle& h1, const typename T::NodeHandle& h2) {
172  { T::intersect(h1, h2) } -> std::ranges::view;
173 };
174 // Also check with 3 arguments to verify variadic support
175 requires requires(const typename T::NodeHandle& h1, const typename T::NodeHandle& h2,
176  const typename T::NodeHandle& h3) {
177  { T::intersect(h1, h2, h3) } -> std::ranges::view;
178 };
179 // Check prefix_lookup with different key sizes
180 // This is tricky, but I am not sure any other way to check if a class has a template method in
181 // C++20
182 requires requires { idx.template prefix_lookup<1>(std::array<std::size_t, 1>{}); };
183 // Check that build_from_encoded method exists (concrete signature, easier to check)
184 // This is the primary build method that accepts pre-encoded data
185 // Relation handles encoding and calls this method - this keeps Index independent
186 // Note: We check with a fixed arity (e.g., 2) to verify the method signature exists
187 // The actual arity will be checked at template instantiation time
188 // The encoded_rows type should match ValueRange (which is std::span<const ValueType>)
189 requires requires(
190  T& idx_mut, const IndexSpec& spec,
191  const std::array<std::span<const typename T::ValueRange::value_type>, 2>& encoded_rows,
192  memory_resource* res) {
193  { idx_mut.build_from_encoded(spec, encoded_rows, res) } -> std::same_as<void>;
194 };
195 // Check that merge method exists
196 // Merges another index into this index, applying a row ID offset to the merged postings
197 requires requires(T& idx_mut, const T& other_idx, std::size_t row_id_offset) {
198  { idx_mut.merge(other_idx, row_id_offset) } -> std::same_as<void>;
199 };

230 /**
231  * @brief Specification for a multi-column index on a relation.
232  *
233  * @details IndexSpec defines which columns of a relation should be indexed
234  * together. The columns form a composite key where values are concatenated
235  * left-to-right. This is used to create HashTrieIndex structures for efficient
236  * prefix lookups and joins.
237  *
238  * @section indexspec_structure Structure
239  *
240  * An IndexSpec contains:
241  * - `cols`: A vector of column indices (0-based) that form the index key
242  * - The order matters: columns are indexed in the order specified
243  * - Duplicate column indices are allowed but generally not useful
244  *
245  * @section indexspec_usage Usage
246  *
247  * @code{.cpp}
248  * // Create an index on the first column only
249  * IndexSpec spec1;
250  * spec1.cols = {0};
251  *
252  * // Create an index on columns 0 and 1 (composite key)
253  * IndexSpec spec2;
254  * spec2.cols = {0, 1};
255  *
256  * // Create an index on columns 1, 0 (different order = different index)
257  * IndexSpec spec3;
258  * spec3.cols = {1, 0}; // Note: spec2 != spec3
259  *
260  * // Use with Relation::ensure_index()
261  * const auto& idx = rel.ensure_index(spec2);
262  * @endcode
263  *
264  * @section indexspec_key_format Key Format
265  *
266  * The index key is formed by concatenating encoded values from the specified
267  * columns in order:
268  * - For IndexSpec with cols = {0, 1}, the key is (col0_value, col1_value)
269  * - Prefix lookups work on prefixes of this key (e.g., just col0_value)
270  * - Full key lookups require all components
271  *
272  * @section indexspec_examples Examples
273  *
274  * For a relation R(x: int, y: int, z: int):
275  * - `{0}` indexes on column x only
276  * - `{0, 1}` indexes on (x, y) composite key
277  * - `{1, 2}` indexes on (y, z) composite key
278  * - `{0, 1, 2}` indexes on all columns (full key)
279  *
280  * @section indexspec_comparison Comparison
281  *
282  * IndexSpec supports equality comparison:
283  * - Two IndexSpecs are equal if their `cols` vectors are identical
284  * - Order matters: {0, 1} != {1, 0}
285  * - Can be used as keys in std::map and std::set
286  *
287  * @note Column indices must be valid (0 <= index < relation arity) when used
288  * to build an index. Invalid indices will cause errors during index
289  * construction.
290  *
291  * @see HashTrieIndex for the index structure created from this specification
292  * @see Relation::ensure_index for building indexes from specifications
293  * @see IndexSpecHash for use in unordered containers
294  */

395 /**
396  * @brief Hash function for IndexSpec to enable use in unordered containers.
397  *
398  * @details IndexSpecHash provides a custom hash function that allows IndexSpec
399  * to be used as a key in std::unordered_map and std::unordered_set. The hash
400  * combines the size of the column vector and all column indices using a
401  * FNV-inspired hash with golden ratio mixing.
402  *
403  * @section indexspechash_algorithm Algorithm
404  *
405  * The hash function:
406  * - Starts with an FNV-like seed constant
407  * - Mixes in the vector size
408  * - Mixes in each column index with bit rotations and golden ratio constant
409  * - Provides good distribution for typical index specifications
410  *
411  * @note This hash function is deterministic and satisfies the requirements
412  * for std::hash: if two IndexSpecs are equal (operator==), they must
413  * have the same hash value.
414  *
415  * @see IndexSpec for the type being hashed
416  * @see std::unordered_map for usage in hash tables
417  */
418 struct IndexSpecHash {
419  /// @brief Compute hash value for an IndexSpec.
420  /// @param s The IndexSpec to hash
421  /// @return Hash value suitable for use in unordered containers
422  /// @details Uses FNV-inspired hashing with golden ratio mixing for good
423  /// distribution. The hash combines the vector size and all column
424  /// indices.
425  std::size_t operator()(const IndexSpec& s) const noexcept {
426  // Start from a large odd constant; mix size and each element (like FNV +
427  // golden ratio salt)
428  std::size_t h = 1469598103934665603ull; // FNV-ish seed
429  h ^= s.cols.size() + 0x9e3779b97f4a7c15ull + (h << 6) + (h >> 2);
430  for (int c : s.cols) {
431  h ^= std::hash<int>{}(c) + 0x9e3779b97f4a7c15ull + (h << 6) + (h >> 2);
432  }
433  return h;
434  }
435 };
