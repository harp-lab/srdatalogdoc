221struct IndexSpecHash {
228 std::size_t operator()(const IndexSpec &s) const noexcept {
229 // Start from a large odd constant; mix size and each element (like FNV +
230 // golden ratio salt)
231 std::size_t h = 1469598103934665603ull; // FNV-ish seed
232 h ^= s.cols.size() + 0x9e3779b97f4a7c15ull + (h << 6) + (h >> 2);
233 for (int c : s.cols) {
234 h ^= std::hash<int>{}(c) + 0x9e3779b97f4a7c15ull + (h << 6) + (h >> 2);
235 }
236 return h;
237 }
238};

491template <typename T> class DictionaryCodec {
492public:

500 explicit DictionaryCodec() : next_id_(0) {}
501
509 std::size_t encode(const T &value) {
510 if (auto it = forward_.find(value); it != forward_.end()) {
511 return it->second;
512 }
513 std::size_t id = next_id_++;
514 forward_.emplace(value, id);
515 inverse_.emplace(id, value);
516 return id;
517 }
518
525 const T &decode(std::size_t id) const {
526 auto it = inverse_.find(id);
527 if (it == inverse_.end()) {
528 throw std::runtime_error("Invalid encoded value for dictionary");
529 }
530 return it->second;
531 }

537 bool contains(std::size_t id) const { return inverse_.contains(id); }
538
543 std::size_t size() const { return forward_.size(); }
544
548 void clear() {
549 forward_.clear();
550 inverse_.clear();
551 next_id_ = 0;
552 }

554private:
555 DictMap forward_;
556 InverseMap inverse_;

1534 ((all_contain = all_contain && handles.contains_value(key)), ...);
1535 },
1536 other_handles);
