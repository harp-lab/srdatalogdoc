116// Custom hash so IndexSpec can be used as key in unordered_map/unordered_set
117struct IndexSpecHash {
118 std::size_t operator()(const IndexSpec &s) const noexcept {
119 // Start from a large odd constant; mix size and each element (like FNV +
120 // golden ratio salt)
121 std::size_t h = 1469598103934665603ull; // FNV-ish seed
122 h ^= s.cols.size() + 0x9e3779b97f4a7c15ull + (h << 6) + (h >> 2);
123 for (int c : s.cols) {
124 h ^= std::hash<int>{}(c) + 0x9e3779b97f4a7c15ull + (h << 6) + (h >> 2);
125 }
126 return h;
127 }
128};

205template <typename T> class DictionaryCodec {
206public:
207 using DictMap = boost::unordered::pmr::unordered_map<T, std::size_t>;
208 using InverseMap = boost::unordered::pmr::unordered_map<std::size_t, T>;
209
210 explicit DictionaryCodec() : next_id_(0) {}
211
212 std::size_t encode(const T &value) {
213 if (auto it = forward_.find(value); it != forward_.end()) {
214 return it->second;
215 }
216 std::size_t id = next_id_++;
217 forward_.emplace(value, id);
218 inverse_.emplace(id, value);
219 return id;
220 }
221
222 const T &decode(std::size_t id) const {
223 auto it = inverse_.find(id);
224 if (it == inverse_.end()) {
225 throw std::runtime_error("Invalid encoded value for dictionary");
226 }
227 return it->second;
228 }
229
230 bool contains(std::size_t id) const { return inverse_.contains(id); }
231
232 std::size_t size() const { return forward_.size(); }
233
234 void clear() {
235 forward_.clear();
236 inverse_.clear();
237 next_id_ = 0;
238 }
239
240private:
241 DictMap forward_;
242 InverseMap inverse_;
243 std::size_t next_id_;
244};

822template <typename NodeHandleT, typename... OtherNodeHandleTs>
823static inline auto hash_intersect(const NodeHandleT &a,
824 const OtherNodeHandleTs &...others) noexcept {
825
826 // Pack the "other" handles into a tuple of references to avoid copying
827 auto other_handles = std::make_tuple(std::cref(others)...);
828
829 // The predicate checks if 'key' is contained in *all* other handles
830 auto predicate = [other_handles](const auto &key) {
831 bool all_contain = true;
832
833 // Use std::apply to iterate the tuple of handles (C++17)
834 // This is fully unrolled at compile time.
835 std::apply(
836 [&all_contain, &key](const auto &...handles) {
837 ((all_contain = all_contain && handles.contains_value(key)), ...);
838 },
839 other_handles);
840
841 return all_contain;
842 };
843
844 // This is the core C++20 feature:
845 // It returns a *lazy* view. No computation is done here.
846 // The predicate is only called as the view is iterated.
847 return std::ranges::filter_view(a.values(), predicate);
848}
