219struct IndexSpecHash {
226 std::size_t operator()(const IndexSpec &s) const noexcept {
227 // Start from a large odd constant; mix size and each element (like FNV +
228 // golden ratio salt)
229 std::size_t h = 1469598103934665603ull; // FNV-ish seed
230 h ^= s.cols.size() + 0x9e3779b97f4a7c15ull + (h << 6) + (h >> 2);
231 for (int c : s.cols) {
232 h ^= std::hash<int>{}(c) + 0x9e3779b97f4a7c15ull + (h << 6) + (h >> 2);
233 }
234 return h;
235 }
236};

1437template <typename NodeHandleT, typename... OtherNodeHandleTs>
1438static inline auto hash_intersect(const NodeHandleT &a,
1439 const OtherNodeHandleTs &...others) noexcept {
1440
1441 // Pack the "other" handles into a tuple of references to avoid copying
1442 auto other_handles = std::make_tuple(std::cref(others)...);
1443
1444 // The predicate checks if 'key' is contained in *all* other handles
1445 auto predicate = [other_handles](const auto &key) {
1446 bool all_contain = true;
1447
1448 // Use std::apply to iterate the tuple of handles (C++17)
1449 // This is fully unrolled at compile time.
1450 std::apply(
1451 [&all_contain, &key](const auto &...handles) {
1452 ((all_contain = all_contain && handles.contains_value(key)), ...);
1453 },
1454 other_handles);
1455
1456 return all_contain;
1457 };
1458
1459 // This is the core C++20 feature:
1460 // It returns a *lazy* view. No computation is done here.
1461 // The predicate is only called as the view is iterated.
1462 return std::ranges::filter_view(a.values(), predicate);
1463}
