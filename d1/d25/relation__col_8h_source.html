227struct IndexSpecHash {
234 std::size_t operator()(const IndexSpec &s) const noexcept {
235 // Start from a large odd constant; mix size and each element (like FNV +
236 // golden ratio salt)
237 std::size_t h = 1469598103934665603ull; // FNV-ish seed
238 h ^= s.cols.size() + 0x9e3779b97f4a7c15ull + (h << 6) + (h >> 2);
239 for (int c : s.cols) {
240 h ^= std::hash<int>{}(c) + 0x9e3779b97f4a7c15ull + (h << 6) + (h >> 2);
241 }
242 return h;
243 }
244};

1121 const OtherNodeHandleTs &...others) noexcept {
1122 // Pack the "other" handles into a tuple of references to avoid copying
1123 auto other_handles = std::make_tuple(std::cref(others)...);

1125 // The predicate checks if 'key' is contained in *all* other handles
1126 auto predicate = [other_handles](const auto &key) {
1127 bool all_contain = true;

1129 // Use std::apply to iterate the tuple of handles (C++17)
1130 // This is fully unrolled at compile time.

1133 ((all_contain = all_contain && handles.contains_value(key)), ...);
1134 },
1135 other_handles);

1138 };
