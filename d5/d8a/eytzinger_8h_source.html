35 // --- 1. Internal Helper: Eytzinger Index (L3 Cache Resident) ---
36 // This manages the "Block Separators". It is cache-oblivious.
37 template <typename T>
38 class EytzingerMap {
39  public:
40  Vector<T> data;
41  Vector<size_t> heap_to_block; // Mapping from Eytzinger heap index to original block index
42 
43  void build(const Vector<T>& sorted_separators) {
44  size_t n = sorted_separators.size();
45  data.resize(n);
46  heap_to_block.resize(n);
47  if (n == 0)
48  return;
49  size_t idx = 0;
50  size_t block_idx = 0;
51  build_recursive(sorted_separators, idx, block_idx, 1);
52  }
53 
54  // Returns the Block ID (0-based) - the original block index, not the heap index
55  [[nodiscard]] size_t search(T key) const noexcept {
56  if (data.empty()) [[unlikely]]
57  return 0;
58 
59  size_t k = 1;
60  size_t n = data.size();
61 
62  // HPC Optimization: Prefetch future tree levels
63  // 64 bytes (cache line) / sizeof(T) gives the lookahead distance
64  constexpr size_t k_prefetch_dist = 64 / sizeof(T);
65 
66  while (k <= n) {
67 #if defined(__GNUC__) || defined(__clang__)
68  __builtin_prefetch(&data[(k * k_prefetch_dist) - 1]);
69 #endif
70  // Branchless descent
71  k = 2 * k + (data[k - 1] < key);
72  }
73 
74  // C++20 Decode: Backtrack to find the lower bound leaf
75  k >>= (std::countr_one(k) + 1);
76  size_t heap_idx = k - 1;
77 
78  // Map heap index to original block index
79  if (heap_idx < heap_to_block.size()) {
80  return heap_to_block[heap_idx];
81  }
82  return heap_idx; // Fallback if mapping is invalid
83  }
84 
85  void clear() {
86  data.clear();
87  heap_to_block.clear();
88  }
89 
90  private:
91  void build_recursive(const Vector<T>& src, size_t& i, size_t& block_idx, size_t k) {
92  if (k > src.size())
93  return;
94  build_recursive(src, i, block_idx, 2 * k);
95  size_t heap_pos = k - 1;
96  data[heap_pos] = src[i];
97  heap_to_block[heap_pos] = block_idx; // Store mapping: heap position -> block index
98  i++;
99  block_idx++;
100  build_recursive(src, i, block_idx, 2 * k + 1);
101  }
102 };

105 template <typename T, std::size_t BlockSize = 64>
106 class AcceleratedArrayIterator {
107  public:
108  using iterator_category = std::forward_iterator_tag;
109  using value_type = T;
110  using difference_type = std::ptrdiff_t;
111  using pointer = const T*;
112  using reference = const T&;
113

115 
116  // Constructor
117  AcceleratedArrayIterator(const T* base, const T* ptr, const T* end, const EytzingerMap<T>* idx)
118  : base_(base), ptr_(ptr), end_(end), idx_(idx) {}
119 
120  reference operator*() const {
121  return *ptr_;
122  }
123  pointer operator->() const {
124  return ptr_;
125  }
126 
127  // 1. Advance (Linear)
128  // Use the exact same logic as SortedArrayIndex to ensure consistency
129  AcceleratedArrayIterator& operator++() {
130  if (ptr_ < end_) {
131  // Use scalar loop or your trusted skip_duplicates
132  ptr_ = SRDatalog::skip_duplicates(ptr_, end_);
133  }
134  return *this;
135  }
136 
137  AcceleratedArrayIterator operator++(int) {
138  AcceleratedArrayIterator tmp = *this;
139  ++(*this);
140  return tmp;
141  }
142 
143  void seek(const T& target) {
144  if (ptr_ >= end_)
145  return;
146  if (*ptr_ >= target)
147  return;
148 
149  // 1. Fallback (No Index)
150  if (idx_ == nullptr || idx_->data.empty()) {
151  size_t len = end_ - ptr_;
152  size_t offset = SRDatalog::search::adaptive_lower_bound(ptr_, len, target);
153  ptr_ += offset;
154  return;
155  }
156 
157  // 2. Eytzinger Lookup
158  size_t block_idx = idx_->search(target);
159  if (block_idx >= idx_->data.size())
160  block_idx = idx_->data.size();
161 
162  // Safety: Backtrack one block to handle boundary duplicates
163  if (block_idx > 0)
164  block_idx--;
165 
166  // 3. Calculate Jump
167  const T* block_start_ptr = base_ + (block_idx * BlockSize);
168 
169  // Forward Only Rule
170  const T* search_start = (block_start_ptr > ptr_) ? block_start_ptr : ptr_;
171 
172  if (search_start >= end_) {
173  ptr_ = end_;
174  return;
175  }
176 
177  // 4. THE CRITICAL FIX: Window Clamping
178  // We know the target is in Block[k] or Block[k+1] (due to backtrack).
179  // We do NOT need to search to 'end_'.
180  // Clamp length to 256 items (4 blocks). This guarantees SIMD usage
181  // (assuming adaptive_lower_bound threshold is > 64).
182 
183  size_t dist_to_end = end_ - search_start;
184  size_t scan_len = std::min(dist_to_end, size_t{256});
185 
186  // asm volatile("nop; nop; nop; nop; nop");
187  // This will now use SIMD Linear Scan because 256 is "small enough"
188  // (or close enough to the threshold) to avoid the overhead of Binary Search setup.
189  // If your adaptive threshold is strictly 128, change 256 to 128.
190  size_t offset = SRDatalog::search::adaptive_lower_bound(search_start, scan_len, target);
191 
192  ptr_ = search_start + offset;
193 
194  // 5. Paranoia Safety / Soft Fallback
195  // If we scanned the 256-item window and didn't find the target (hit the end of window),
196  // AND we haven't hit the true end, it means Eytzinger was wrong (unlikely)
197  // or duplicates are extremely long. We resume searching from here.
198  if (offset == scan_len && ptr_ < end_ && *ptr_ < target) {
199  size_t remaining = end_ - ptr_;
200  ptr_ += SRDatalog::search::adaptive_lower_bound(ptr_, remaining, target);
201  }
202  }
203 
204  bool operator==(const AcceleratedArrayIterator& other) const {
205  return ptr_ == other.ptr_;
206  }
207  bool operator!=(const AcceleratedArrayIterator& other) const {
208  return ptr_ != other.ptr_;
209  }
210 
211  private:
212  const T* base_{nullptr};
213  const T* ptr_{nullptr};
214  const T* end_{nullptr};
215  const EytzingerMap<T>* idx_{nullptr};
216 };
