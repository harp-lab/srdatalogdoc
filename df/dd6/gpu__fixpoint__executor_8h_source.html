122  using IndexSpec = SRDatalog::IndexSpec;

125  }(ColSeq{});
126 
127  // Force rebuild by calling ensure_index with build=true
128  // This will rebuild the index from relation data if the relation has data
129  // For NEW_VER after InsertFromRelation (merge), we need to reconstruct the relation from index
130  // to match tc_lower_device_benchmark behavior. The merge operation updates the index, but
131  // the relation data might be out of sync. Reconstructing ensures consistency before swap.
132  bool new_tuples = false;
133  if constexpr (ver == NEW_VER) {
134  // Ensure index exists first
135  auto& new_idx = rel.ensure_index(runtime_spec, true);
136  new_tuples = new_idx.root().degree() > 0;
137  // Reconstruct relation from index
138  // this is mainly because we might need rebuild other indexes, this can avoid deduplication
139  // multiple times
140  rel.reconstruct_from_index(runtime_spec);
141  } else if constexpr (ver == FULL_VER) {
142  // For FULL_VER, if the relation was just reconstructed from index in ComputeDeltaAndMerge,
143  // the index should already be correct. Just ensure it exists without forcing rebuild.
144  // However, if this is the first time (iteration 0) or if the index might be out of sync,
145  // we should rebuild. For now, we always rebuild to be safe, but the relation should
146  // already have the correct data from ComputeDeltaAndMerge's reconstruct_from_index.
147  rel.ensure_index(runtime_spec, true);
148  } else {
149  // For other versions (DELTA_VER), rebuild index from relation data
150  rel.ensure_index(runtime_spec, true);
151  }
152 
153  auto res = GPUFixpointExecutor<std::tuple<TailOps...>, RelationsBundle, SchemaTuple>::execute(
154  relations, iteration, max_output);
155  return res || new_tuples;

525  // CheckSize is used in ComputeDeltaOps, which already determines the canonical index
526  // at compile time. The canonical index is the first IndexSpec found in AllIndexSpecs
527  // for this schema. We extract it from the InsertFromRelation in TailOps that matches
528  // this schema, which uses IndexForInsert derived from the canonical index.
529  using TailOpsTuple = std::tuple<TailOps...>;
530  using FoundInsert = mp::mp_find_if<TailOpsTuple, IsInsertForSchema>;
531  static_assert(FoundInsert::value < sizeof...(TailOps),
532  "CheckSize: InsertFromRelation for this schema must exist in TailOps");
533 
534  using InsertOp = mp::mp_at_c<TailOpsTuple, FoundInsert::value>;
535  using IndexSpecT = typename InsertOp::index_spec_type;
536  using ColSeq = typename IndexSpecT::column_indexes_type;
537  auto runtime_canonical_spec = []<int... Cols>(std::integer_sequence<int, Cols...>) {
538  return SRDatalog::IndexSpec{{Cols...}};
539  }(ColSeq{});
540 
541  // Get the canonical index and check its size (this is the accurate size)
542  // The canonical index should already be built on NEW_VER by RebuildNewVerOps
543  // Force rebuild to ensure index reflects relation data
544  r.ensure_index(runtime_canonical_spec, true);
545 
546  std::size_t canonical_size = 0;
547  try {
548  auto& canonical_idx = r.get_index(runtime_canonical_spec);
549  canonical_size = canonical_idx.root().degree();
550  } catch (...) {
551  // Index doesn't exist, use relation size as fallback
552  canonical_size = r.size();
553  }
554 
555  if (canonical_size == 0) {
556  return false; // Halt execution
557  }
558 
559  // Print NEW size for NEW_VER
560  if constexpr (Ver == NEW_VER) {
561  std::string schema_name = SRDatalog::AST::tstring_to_string<typename Schema::name_type>();
562  std::cout << "Iteration " << iteration << ": " << schema_name
563  << " (NEW) size: " << canonical_size << std::endl;
564  }
565  return GPUFixpointExecutor<std::tuple<TailOps...>, RelationsBundle, SchemaTuple>::execute(
566  relations, iteration, max_output);
567  }

570 // 7. InsertFromRelation (Merge/Copy)
571 /**
572  * @brief Specialization for InsertFromRelation: Merges one relation into another.
573  *
574  * Performed using `DeviceSortedArrayIndex::merge`.
575  * Used for merging NEW tuples into FULL or DELTA relations.
576  */
577 template <typename Schema, std::size_t FromVer, std::size_t ToVer, typename IndexSpec,
578  typename... TailOps, typename RelationsBundle, typename SchemaTuple>
579 struct GPUFixpointExecutor<
580  std::tuple<mir::InsertFromRelation<Schema, FromVer, ToVer, IndexSpec>, TailOps...>,
581  RelationsBundle, SchemaTuple> {
582  static bool execute(RelationsBundle& relations, std::size_t iteration, std::size_t max_output) {
583  // In iteration 0, redirect DELTA_VER reads to FULL_VER (but not writes)
584  auto& from = [&]() -> auto& {
585  if constexpr (FromVer == DELTA_VER) {
586  if (iteration == 0) {
587  return detail::get_relation<Schema, FULL_VER, RelationsBundle, SchemaTuple>(relations);
588  }
589  }
590  return detail::get_relation<Schema, FromVer, RelationsBundle, SchemaTuple>(relations);
591  }();
592 
593  // Don't redirect writes - always use the actual target version
594  auto& to = detail::get_relation<Schema, ToVer, RelationsBundle, SchemaTuple>(relations);
595 
596  using ColSeq = typename IndexSpec::column_indexes_type;
597  auto runtime_spec = []<int... Cols>(std::integer_sequence<int, Cols...>) {
598  return SRDatalog::IndexSpec{{Cols...}};
599  }(ColSeq{});
600 
601  // We need the index on FROM relation.
602  auto& from_idx = from.ensure_index(runtime_spec);
603 
604  if (iteration == 0) {
605  to.ensure_index(runtime_spec);
606  } else {
607  try {
608  // Verify index exists (will throw if it doesn't, which shouldn't happen)
609  auto& test_idx = to.get_index(runtime_spec);
610  (void)test_idx;
611  } catch (...) {
612  // Index doesn't exist (shouldn't happen after iteration 0, but handle it)
613  to.ensure_index(runtime_spec);
614  }
615  }
616  // Get the index (now guaranteed to exist)
617  auto& to_idx = to.get_index(runtime_spec);
618 
619  // Merge
620  // Note: merge invalidates handles!
621  std::size_t row_offset =
622  to_idx.size(); // Get size BEFORE merge (matches manual version line 392)
623  std::size_t from_size = from_idx.size();
624  std::size_t to_size_before = to_idx.size();
625  // DeviceSortedArrayIndex::merge takes (other, offset)
626  to_idx.merge(from_idx, row_offset);
627  std::size_t to_size_after = to_idx.size();
628 
629  // Debug: Print merge info to verify it's working
630  std::string schema_name = SRDatalog::AST::tstring_to_string<typename Schema::name_type>();
631  std::cout << "InsertFromRelation iter " << iteration << ": " << schema_name << " merge "
632  << runtime_spec.to_string() << ": from_size=" << from_size
633  << ", to_before=" << to_size_before << ", to_after=" << to_size_after << std::endl;
634 
635  // After merge, the index is correct. For device relations, is_dirty() doesn't use
636  // rows_processed_ so we don't need to update it. The index size is the source of truth.
637 
638  bool result =
639  GPUFixpointExecutor<std::tuple<TailOps...>, RelationsBundle, SchemaTuple>::execute(
640  relations, iteration, max_output);
641  return result;
642  }

645 // 9. ComputeDeltaAndMerge
646 template <typename Schema, std::size_t FromVer, std::size_t ToVer, std::size_t DeltaVer,
647  typename IndexSpecT, typename... TailOps, typename RelationsBundle, typename SchemaTuple>
648 struct GPUFixpointExecutor<
649  std::tuple<mir::ComputeDeltaAndMerge<Schema, FromVer, ToVer, DeltaVer, IndexSpecT>, TailOps...>,
650  RelationsBundle, SchemaTuple> {
651  static bool execute(RelationsBundle& relations, std::size_t iteration, std::size_t max_output) {
652  using namespace SRDatalog::GPU;
653  // Retrieve relations
654  // From = NEW, To = FULL, Delta = DELTA
655  auto& new_rel = detail::get_relation<Schema, FromVer, RelationsBundle, SchemaTuple>(relations);
656  auto& full_rel = detail::get_relation<Schema, ToVer, RelationsBundle, SchemaTuple>(relations);
657  auto& delta_rel =
658  detail::get_relation<Schema, DeltaVer, RelationsBundle, SchemaTuple>(relations);
659 
660  // Clear delta relation at the start
661  delta_rel.clear();
662 
663  using ColSeq = typename IndexSpecT::column_indexes_type;
664  auto runtime_spec = []<int... Cols>(std::integer_sequence<int, Cols...>) {
665  return SRDatalog::IndexSpec{{Cols...}};
666  }(ColSeq{});
667 
668  // Ensure indexes exists
669  new_rel.ensure_index(runtime_spec);
670  auto& new_idx = new_rel.get_index(runtime_spec);
671 
672  if (iteration == 0) {
673  full_rel.ensure_index(runtime_spec);
674  } else {
675  // Full index should exist
676  full_rel.ensure_index(runtime_spec); // Safe check
677  }
678  auto& full_idx = full_rel.get_index(runtime_spec);
679 
680  // Temporary index for Delta (result of diff) - managed by delta_rel
681  delta_rel.ensure_index(runtime_spec);
682  auto& delta_idx = delta_rel.get_index(runtime_spec);
683 
684  // 1. Compute Diff (New - Full) -> delta_idx
685  // Side effect: merged results updated in full_idx (provenance)
686  new_idx.set_difference_update(full_idx, delta_idx);
687 
688  // 2. Merge Diff into Full
689  if (!delta_idx.empty()) {
690  std::size_t row_offset = full_idx.size();
691  full_idx.merge(delta_idx, row_offset);
692  }
693 
694  // 3. Populate DELTA relation with Diff
695  // delta_rel.clear(); // Already populated or cleared at start
696  // Access internals to reconstruct
697  delta_idx.reconstruct_to_relation(runtime_spec, delta_rel.unsafe_interned_columns(),
698  delta_rel.provenance());
699 
700  // 4. Clear NEW relation
701  new_rel.clear();
702 
703  // Debug logging
704  std::string schema_name = SRDatalog::AST::tstring_to_string<typename Schema::name_type>();
705  std::cout << "ComputeDeltaAndMerge iter " << iteration << ": " << schema_name
706  << " (DELTA) size: " << delta_rel.size() << std::endl;
707 
708  // Recurse
709  return GPUFixpointExecutor<std::tuple<TailOps...>, RelationsBundle, SchemaTuple>::execute(
710  relations, iteration, max_output);
711  }
