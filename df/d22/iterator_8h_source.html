41  * @details Looks up the iterator from the iterator tuple using the IndexSpecT.

44  *
45  * @tparam Schema The relation schema
46  * @tparam Cols The column indexes for the index
47  * @tparam Ver The version number
48  * @tparam PrefixVars The tuple of prefix variables (unused here, kept for compatibility)
49  * @tparam IteratorPosMap Compile-time hana::map mapping IndexSpecT types to iterator positions
50  * @tparam IteratorTuple Runtime tuple of iterators
51  */
52 template <CRelationSchema Schema, int... Cols, std::size_t Ver, tmp::CTuple PrefixVars,
53  typename IteratorPosMap, typename IteratorTuple>
54 struct GenerateIterator<
55  mir::ColumnSource<IndexSpecT<Schema, std::integer_sequence<int, Cols...>, Ver>, PrefixVars>,
56  IteratorPosMap, IteratorTuple> {
57  static auto get(const IteratorTuple& iterators) {
58  // 1. Get the IndexSpecT type for this ColumnSource
59  using Idx = IndexSpecT<Schema, std::integer_sequence<int, Cols...>, Ver>;
60 
61  // 2. Look up the iterator position from the compile-time map
62  // decltype preserves the reference qualifiers (in this case, an rvalue reference &&). In C++,
63  // you cannot access a static member (like ::value) directly on a reference type (e.g.,
64  // T&&::value is invalid syntax). std::decay_t removes the reference and/or cv-qualifiers,
65  // so the result is a plain type.
66  constexpr auto iter_pos =
67  std::decay_t<decltype(hana::at_key(IteratorPosMap{}, hana::type_c<Idx>))>::value;
68 
69  // 3. Get and return the iterator from the iterator tuple
70  // The iterator is already updated (prefixed) during pipeline execution
71  return std::get<iter_pos>(iterators);
72  }
