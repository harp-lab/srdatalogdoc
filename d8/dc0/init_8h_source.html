32  * @param device_id The CUDA device ID to initialize (default: 0)

36  */

39 
40  // Ensure device is set for the calling thread (cudaSetDevice is thread-local)
41  cudaError_t err = cudaSetDevice(device_id);
42  if (err != cudaSuccess) {
43  throw std::runtime_error("Failed to set CUDA device " + std::to_string(device_id) + ": " +
44  std::string(cudaGetErrorString(err)));
45  }
46 
47  std::call_once(init_flag, [device_id]() {
48  int device_count = 0;
49  cudaError_t err = cudaGetDeviceCount(&device_count);
50  if (err != cudaSuccess || device_count == 0) {
51  throw std::runtime_error("No CUDA devices found or CUDA error: " +
52  std::string(cudaGetErrorString(err)));
53  }
54 
55  if (device_id < 0 || device_id >= device_count) {
56  throw std::runtime_error("Invalid CUDA device ID: " + std::to_string(device_id) +
57  " (available devices: " + std::to_string(device_count) + ")");
58  }
59

62  if (err == cudaSuccess) {

64  << prop.minor << ")" << std::endl;
