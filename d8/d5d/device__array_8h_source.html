217  void push_back(const T& value) {
218  // Need to resize to accommodate new element
219  const size_type old_size = size();

247  * @brief Append all elements from another array

250  void concat(const DeviceArray& other) {
251  if (other.empty()) {
252  return;

255  const size_type new_size = old_size + other.size();
256  resize(new_size, true);

362  /**
363  * @brief Print the first N elements to an output stream
364  * @param n Number of elements to print (default: all elements)
365  * @param os Output stream (default: std::cout)
366  */
367  void print_first(size_type n = std::numeric_limits<size_type>::max(),
368  std::ostream& os = std::cout) const {
369  if (empty() || data() == nullptr) {
370  os << "DeviceArray (empty)\n";
371  return;
372  }
373 
374  const size_type print_count = std::min(n, size());
375  std::vector<T> host_data(print_count);
376 
377  cudaError_t err =
378  cudaMemcpy(host_data.data(), data(), print_count * sizeof(T), cudaMemcpyDeviceToHost);
379  if (err != cudaSuccess) {
380  os << "DeviceArray: Failed to copy data from device: " << cudaGetErrorString(err) << "\n";
381  return;
382  }
383 
384  os << "DeviceArray (size=" << size() << ", capacity=" << capacity() << ", showing first "
385  << print_count << "):\n";
386  for (size_type i = 0; i < print_count; ++i) {
387  os << " [" << i << "] = " << host_data[i];
388  if constexpr (std::is_integral_v<T>) {
389  os << " (0x" << std::hex << static_cast<std::uint64_t>(host_data[i]) << std::dec << ")";
390  }
391  os << "\n";
392  }
393  if (print_count < size()) {
394  os << " ... (" << (size() - print_count) << " more elements)\n";
395  }
396  }
