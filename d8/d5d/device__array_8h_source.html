26 /**
27  * @brief Thread-safe singleton that provides a global GPU arena allocator
28  * @note CUDA must be initialized before this is called (call init_cuda() first)
29  */
30 inline umpire::Allocator& get_gpu_arena_allocator() {
31  static umpire::Allocator arena = []() {
32  // Ensure CUDA is initialized before accessing Umpire DEVICE allocator
33  // Umpire's DEVICE allocator requires an active CUDA context
34  int current_device = -1;
35  cudaError_t err = cudaGetDevice(&current_device);
36  if (err != cudaSuccess) {
37  throw std::runtime_error(
38  "get_gpu_arena_allocator: CUDA not initialized. Call init_cuda() first. Error: " +
39  std::string(cudaGetErrorString(err)));
40  }
41 
42  auto& rm = umpire::ResourceManager::getInstance();
43  auto base_allocator = rm.getAllocator("DEVICE");
44  return rm.makeAllocator<umpire::strategy::DynamicPoolList>("gpu_arena", base_allocator);
45  }();
46  return arena;
47 }

178  * @param new_size New logical size
179  * @param preserve_data If true, copy existing data when reallocating (default: false)
