25 /// @brief Group-cooperative lower_bound (works with arbitrary group sizes)
26 /// @param tile Cooperative group (can be warp, sub-warp, or even sequential)
27 /// @return First index where data[i] >= key, or len if all < key
28 template <typename T, typename IndexType = uint32_t, typename Group,
29  T sentinel = cuda::std::numeric_limits<T>::max()>
30 __device__ GPU_FORCE_INLINE IndexType group_lower_bound(const T* data, IndexType len, T key,
31  Group tile) {
32  const IndexType group_size = static_cast<IndexType>(tile.size());
33  IndexType lo = 0;
34  IndexType hi = len;
35 
36  // Binary search until range fits in the group
37  while (hi - lo > group_size) {
38  IndexType mid = lo + ((hi - lo) >> 1);
39  if (data[mid] < key) {
40  lo = mid + 1;
41  } else {
42  hi = mid;
43  }
44  }
45 
46  // Parallel scan of final <= group_size elements
47  // For sequential case (group_size == 1), this becomes a simple loop
48  if (group_size == 1) {
49  // Sequential case: simple linear search
50  IndexType idx = lo;
51  while (idx < hi && data[idx] < key) {
52  ++idx;
53  }
54  return idx;
55  }
56 
57  // Parallel case: use cooperative group primitives
58  IndexType lane = static_cast<IndexType>(tile.thread_rank());
59  IndexType idx = lo + lane;
60  T val = (idx < hi) ? data[idx] : sentinel;
61  unsigned long long mask = tile.ballot(val < key);
62 
63  // Count number of set bits in mask (population count)
64  // For groups <= 32, use __popc on lower 32 bits; for larger groups, use __popcll
65  IndexType count;
66  if (group_size <= 32) {
67  count = static_cast<IndexType>(__popc(static_cast<uint32_t>(mask)));
68  } else {
69  count = static_cast<IndexType>(__popcll(mask));
70  }
71 
72  return lo + count;
73 }

144 /// @brief Group-cooperative equal_range (works with arbitrary group sizes)
145 /// @param tile Cooperative group (can be warp, sub-warp, or even sequential)
146 /// @return {lower_bound, upper_bound} pair
147 template <typename T, typename IndexType = uint32_t, typename Group>
148 __device__ GPU_FORCE_INLINE thrust::pair<IndexType, IndexType> group_equal_range(const T* data,
149  IndexType len,
150  T key,
151  Group tile) {
152  IndexType lb = group_lower_bound<T, IndexType>(data, len, key, tile);
153  if (lb == len) {
154  return {len, len};
155  }
156  if (data[lb] != key) {
157  return {lb, lb};
158  }
159  IndexType ub = group_upper_bound<T, IndexType>(data + lb, len - lb, key, tile);
160  return {lb, lb + ub};
161 }
