71 /**
72  * @brief Get a single value from tuple-based state for a given variable
73  *
74  * @details Uses compile-time position map to look up variable position,
75  * then accesses the tuple at runtime. All Hana operations are compile-time.
76  *
77  * @tparam VarPosMap Compile-time hana::map mapping variable types to positions
78  * @tparam StateTuple Runtime state as std::tuple<Value1, Value2, ...>
79  * @tparam Var The variable type to extract
80  * @return The value of the variable from the state tuple
81  */
82 template <typename VarPosMap, typename StateTuple, CVar Var>
83 inline auto get_value_from_tuple(const StateTuple& state) {
84  // Get position from compile-time map, then access tuple at runtime
85  // All Hana operations (at_key, type_c) are evaluated at compile-time
86  constexpr auto pos = std::decay_t<decltype(hana::at_key(VarPosMap{}, hana::type_c<Var>))>::value;
87  return std::get<pos>(state);
88 }

205 /**
206  * @brief Get value for a single term (Var, Const, or CppExpr) from tuple-based state
207  *
208  * @details Handles different term types:
209  * - Var: extracts from StateTuple using VarPosMap
210  * - Const: returns compile-time constant value
211  * - CppExpr: evaluates expression with captured variables (not yet fully supported)
212  *
213  * @tparam VarPosMap Compile-time hana::map mapping variable types to positions
214  * @tparam StateTuple Runtime state as std::tuple<Value1, Value2, ...>
215  * @tparam Term The term type (Var, Const, or CppExpr)
216  */
217 template <typename VarPosMap, typename StateTuple, typename Term>
218 constexpr auto get_term_value_tuple(const StateTuple& state) {
219  if constexpr (CVar<Term>) {
220  // Case 1: Variable - get value from state tuple
221  return get_value_from_tuple<VarPosMap, StateTuple, Term>(state);
222  } else if constexpr (CConst<Term>) {
223  // Case 2: Constant - use compile-time constant value
224  return Term::value;
225  } else if constexpr (CCppExpr<Term>) {
226  // Case 3: C++ Expression - would evaluate expression
227  static_assert(!std::is_same_v<Term, Term>, "CppExpr not yet supported in head position");
228  return std::size_t(0); // unreachable
229  } else {
230  // Case 4: Unknown term type - compile error
231  static_assert(!std::is_same_v<Term, Term>, "Unknown term type");
232  return std::size_t(0); // unreachable
233  }
234 }

254 /**
255  * @brief Set a value at a specific position in the tuple-based state
256  *
257  * @details Sets the value for a variable at its compile-time determined position.
258  * The state tuple must be pre-allocated with the correct size based on the join plan.
259  * All variable positions are known at compile time, so the tuple type is fixed.
260  *
261  * @tparam VarPosMap Compile-time hana::map mapping variable types to positions
262  * @tparam StateTuple Runtime state as std::tuple<Value1, Value2, ...> (fixed size)
263  * @tparam Var The variable type to set
264  * @tparam Value The value type
265  * @param state The runtime state tuple (passed by reference, will be modified)
266  * @param value The value to set
267  */
268 template <typename VarPosMap, typename StateTuple, CVar Var, typename Value>
269 inline void set_value_in_tuple(StateTuple& state, const Value& value) {
270  // Get position from compile-time map
271  constexpr auto pos = std::decay_t<decltype(hana::at_key(VarPosMap{}, hana::type_c<Var>))>::value;
272  static_assert(pos < std::tuple_size_v<StateTuple>,
273  "Variable position must be within tuple bounds. Tuple size must match join plan.");
274  // Set value at the position
275  std::get<pos>(state) = value;
276 }

321  * @brief Extracts a type that maps to a specific position in a position map.
322  *
323  * @details Finds the type that maps to the given position by iterating
324  * through all keys in the map and checking their position values.
325  * Works with any compile-time hana::map that maps types to positions
326  * (e.g., VarPosMap, IteratorPosMap).
327  *
328  * @tparam PosMap Compile-time hana::map mapping types to positions
329  * @tparam Pos The position to find
330  */
331 template <typename PosMap, std::size_t Pos>
332 struct FindTypeAtPos {
333  // Helper to check if a key maps to position Pos
334  template <typename Key>
335  struct KeyMatchesPos {
336  using AtKeyResult = decltype(hana::at_key(PosMap{}, Key{}));
337  static constexpr bool value = std::remove_reference_t<AtKeyResult>::value == Pos;
338  };

401  * @brief Gets a mutable reference to an iterator from IteratorTuple using IteratorPosMap.
402  *
403  * @details Looks up the iterator position from IteratorPosMap at compile-time,
404  * then returns a mutable reference to the iterator at that position in IteratorTuple.
405  * This allows the caller to modify the iterator in-place (e.g., apply prefixes).
406  *
407  * @tparam IteratorPosMap Compile-time hana::map mapping IndexSpecT types to iterator positions
408  * @tparam IteratorTuple Runtime tuple of iterators
409  * @tparam IndexSpecT The IndexSpecT type to look up
410  * @param iterators The runtime iterator tuple (passed by reference)
411  * @return Mutable reference to the iterator at the position specified by IteratorPosMap
412  */
413 template <typename IteratorPosMap, typename IteratorTuple, typename IndexSpecT>
414 inline auto& get_iterator_ref(IteratorTuple& iterators) {
415  // decay for rvalue reference &&
416  constexpr auto iter_pos =
417  std::decay_t<decltype(hana::at_key(IteratorPosMap{}, hana::type_c<IndexSpecT>))>::value;
418  return std::get<iter_pos>(iterators);

421 /**
422  * @brief Updates an iterator in IteratorTuple by replacing it with itself prefixed with a value.
423  *
424  * @details Looks up the iterator mapped to the given Index and replaces it with itself prefixed
425  * with the provided value. This is useful for updating iterators when variable bindings change.
426  * Handles both single values and tuples of values.
427  *
428  * @tparam IteratorPosMap Compile-time hana::map mapping IndexSpecT types to iterator positions
429  * @tparam IteratorTuple Runtime tuple of iterators
430  * @tparam Index The IndexSpecT type whose mapped iterator will be updated
431  * @param iterators The runtime iterator tuple (passed by reference, will be modified)
432  * @param prefix_val The prefix value(s) to apply (can be a single value or a tuple)
433  */
434 template <typename IteratorPosMap, typename IteratorTuple, CIndexSpec Index>
435 inline void update_iterator_with_prefixed(IteratorTuple& iterators, const auto& prefix_val) {
436  // Get mutable reference to the iterator mapped to Index
437  auto& iter = get_iterator_ref<IteratorPosMap, IteratorTuple, Index>(iterators);
438 
439  // Check if prefix_val is a tuple type and handle accordingly
440  if constexpr (tmp::is_tuple_v<std::decay_t<decltype(prefix_val)>>) {
441  // Apply prefix with multiple values from tuple
442  std::apply([&](const auto&... vals) { iter = iter.prefix(vals...); }, prefix_val);
443  } else {
444  // Single value case
445  iter = iter.prefix(prefix_val);
446  }
447 }
