1 /**
2  * @file ast.h
3  * @brief Abstract Syntax Tree (AST) for Datalog using Template Metaprogramming.
4  *
5  * @details This file defines the compile-time AST structures for representing
6  * Datalog programs in C++. The AST uses template metaprogramming to encode
7  * programs as types, enabling:
8  *
9  * - Type-safe program representation
10  * - Compile-time analysis and transformation
11  * - Zero-overhead runtime execution
12  * - Integration with C++ type system
13  *
14  * The AST includes:
15  * - Basic constructs: Var, Const, CppExpr, Clause
16  * - Rule structures: Rule, Fixpoint, NonIterativeRuleSets
17  * - Program structure: Program, Database
18  * - MIR (Mid-level IR): ColumnSource, ColumnJoin, DestinationRelation
19  * - Template metaprogramming utilities for AST analysis
20  *
21  * All AST nodes are compile-time types, not runtime values. The actual
22  * execution happens in runtime.h which interprets MIR operations.
23  *
24  * @note **C++20/23 Features Used**: This file extensively uses C++20 concepts
25  * for type constraints, `auto` non-type template parameters for lambda
26  * expressions, lambda in unevaluated contexts, fold expressions, and
27  * `requires` clauses for template constraints.
28  */

1000  * @details Represents a database with three versions of each relation:
1001  * - full: The complete relation (all facts derived so far)

1014 struct SemiNaiveDatabase {
1015  /// @brief The database schema type
1016  using schema_type = DBType;
1017  /// @brief The tuple type of all relation types in the database
1018  using data_type = typename DatabaseToRelationTypes<DBType>::type;
1019  /// @brief The full relations (all facts)

1021  /// @brief The delta relations (previous iteration)

1023  /// @brief The new relations (current iteration)
