52#define DEFINE_TYPE_PREDICATE_IMPL(PredName, VarName, ConceptName, \
53 TEMPLATE_PARAMS, TEMPLATE_ARGS) \
54 template <typename T> struct PredName : std::false_type {}; \
55 template <TEMPLATE_PARAMS> \
56 struct PredName<VarName<TEMPLATE_ARGS>> : std::true_type {}; \
57 template <typename T> \
58 inline constexpr bool PredName##_v = PredName<T>::value; \
59 template <typename T> \
60 concept ConceptName = PredName<T>::value;

62// Macro to define type predicate boilerplate
63// Usage: DEFINE_TYPE_PREDICATE(is_xxx, Xxx, CXxx, (template_params),
64// (template_args)) For simple cases: DEFINE_TYPE_PREDICATE(is_var, Var, CVar,
65// (CTString Name), (Name)) For parameter packs:
66// DEFINE_TYPE_PREDICATE(is_tstring, TString, CTString, (char... chars),
67// (chars...)) For multiple params: DEFINE_TYPE_PREDICATE(is_xxx, Xxx, CXxx, (T1
68// p1, T2 p2), (p1, p2)) Note: Parentheses in arguments are stripped by the
69// preprocessor when used in template contexts
70#define STRIP_PARENS(...) __VA_ARGS__
71#define DEFINE_TYPE_PREDICATE(PredName, VarName, ConceptName, TEMPLATE_PARAMS, \
72 TEMPLATE_ARGS) \
73 DEFINE_TYPE_PREDICATE_IMPL(PredName, VarName, ConceptName, \
74 STRIP_PARENS TEMPLATE_PARAMS, \
75 STRIP_PARENS TEMPLATE_ARGS)
