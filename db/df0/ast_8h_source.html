57#define DEFINE_TYPE_PREDICATE_IMPL(PredName, VarName, ConceptName, \
58 TEMPLATE_PARAMS, TEMPLATE_ARGS) \
59 template <typename T> struct PredName : std::false_type {}; \
60 template <TEMPLATE_PARAMS> \
61 struct PredName<VarName<TEMPLATE_ARGS>> : std::true_type {}; \
62 template <typename T> \
63 inline constexpr bool PredName##_v = PredName<T>::value; \
64 template <typename T> \
65 concept ConceptName = PredName<T>::value;

67// Macro to define type predicate boilerplate
68// Usage: DEFINE_TYPE_PREDICATE(is_xxx, Xxx, CXxx, (template_params),
69// (template_args)) For simple cases: DEFINE_TYPE_PREDICATE(is_var, Var, CVar,
70// (CTString Name), (Name)) For parameter packs:
71// DEFINE_TYPE_PREDICATE(is_tstring, TString, CTString, (char... chars),
72// (chars...)) For multiple params: DEFINE_TYPE_PREDICATE(is_xxx, Xxx, CXxx, (T1
73// p1, T2 p2), (p1, p2)) Note: Parentheses in arguments are stripped by the
74// preprocessor when used in template contexts
75#define STRIP_PARENS(...) __VA_ARGS__
76#define DEFINE_TYPE_PREDICATE(PredName, VarName, ConceptName, TEMPLATE_PARAMS, \
77 TEMPLATE_ARGS) \

79 STRIP_PARENS TEMPLATE_PARAMS, \
80 STRIP_PARENS TEMPLATE_ARGS)
