100 __device__ _CCCL_FORCEINLINE void binary_search_iteration_custom(It data, Size& begin, Size& end,
101  T key, int shift, Comp comp) {
102  IntT scale = (1 << shift) - 1;
103  Size mid = (begin + scale * end) >> shift;
104 
105  T key2 = data[mid];
106  bool pred = UpperBound ? !comp(key, key2) : comp(key2, key);
107  if (pred) {
108  begin = mid + 1;
109  } else {
110  end = mid;
111  }
112 }

125 __device__ _CCCL_FORCEINLINE Size biased_binary_search_custom(It data, Size count, T key,
126  IntT levels, Comp comp) {
127  Size begin = 0;

149 template <bool UpperBound, class Size, class It1, class It2, class Comp>
150 __device__ _CCCL_FORCEINLINE Size merge_path_custom(It1 a, Size aCount, It2 b, Size bCount,
151  Size diag, Comp comp) {
152  // using T = thrust::detail::it_value_t<It1>;
153  using T = patch::it_value_t<It1>;
154 
155  Size begin = ::cuda::std::max<Size>(0, diag - bCount);
156  Size end = ::cuda::std::min<Size>(diag, aCount);
157 
158  while (begin < end) {
159  Size mid = (begin + end) >> 1;
160  T aKey = a[mid];
161  T bKey = b[diag - 1 - mid];
162  bool pred = UpperBound ? comp(aKey, bKey) : !comp(bKey, aKey);
163  if (pred) {
164  begin = mid + 1;
165  } else {
166  end = mid;
167  }
168  }
169  return begin;
170 }

172 template <class It1, class It2, class Size, class Size2, class CompareOp>
173 __device__ _CCCL_FORCEINLINE ::cuda::std::pair<Size, Size> balanced_path_custom(
174  It1 keys1, It2 keys2, Size num_keys1, Size num_keys2, Size diag, Size2 levels,
175  CompareOp compare_op) {
176  // using T = thrust::detail::it_value_t<It1>;
177  using T = patch::it_value_t<It1>;
178 
179  Size index1 = merge_path_custom<false>(keys1, num_keys1, keys2, num_keys2, diag, compare_op);
180  Size index2 = diag - index1;
181 
182  bool star = false;
183  if (index2 < num_keys2) {
184  T x = keys2[index2];
185 
186  // Search for the beginning of the duplicate run in both A and B.
187  Size start1 = biased_binary_search_custom<false>(keys1, index1, x, levels, compare_op);
188  Size start2 = biased_binary_search_custom<false>(keys2, index2, x, levels, compare_op);
189 
190  // The distance between x's merge path and its lower_bound is its rank.
191  // We add up the a and b ranks and evenly distribute them to
192  // get a stairstep path.
193  Size run1 = index1 - start1;
194  Size run2 = index2 - start2;
195  Size total_run = run1 + run2;
196 
197  // Attempt to advance b and regress a.
198  Size advance2 = max<Size>(total_run >> 1, total_run - run1);
199  Size end2 = min<Size>(num_keys2, start2 + advance2 + 1);
200 
201  Size run_end2 =
202  index2 + binary_search_custom<true>(keys2 + index2, end2 - index2, x, compare_op);
203  run2 = run_end2 - start2;
204 
205  advance2 = min<Size>(advance2, run2);
206  Size advance1 = total_run - advance2;
207 
208  bool round_up = (advance1 == advance2 + 1) && (advance2 < run2);
209  if (round_up) {
210  star = true;
211  }
212 
213  index1 = start1 + advance1;
214  }
215  return ::cuda::std::make_pair(index1, (diag - index1) + star);
216 } // func balanced_path_custom

385  template <bool IS_FULL_TILE, class T, class It1, class It2>
386  __device__ _CCCL_FORCEINLINE void gmem_to_reg(T (&output)[ITEMS_PER_THREAD], It1 input1,
387  It2 input2, int count1, int count2) {
388  if (IS_FULL_TILE) {
389 // _CCCL_PRAGMA_UNROLL_FULL()
390 #pragma unroll
391  for (int ITEM = 0; ITEM < ITEMS_PER_THREAD - 1; ++ITEM) {
392  int idx = BLOCK_THREADS * ITEM + threadIdx.x;
393  output[ITEM] =
394  (idx < count1) ? static_cast<T>(input1[idx]) : static_cast<T>(input2[idx - count1]);
395  }
396 
397  // last ITEM might be a conditional load even for full tiles
398  // please check first before attempting to load.
399  int ITEM = ITEMS_PER_THREAD - 1;
400  int idx = BLOCK_THREADS * ITEM + threadIdx.x;
401  if (idx < count1 + count2) {
402  output[ITEM] =
403  (idx < count1) ? static_cast<T>(input1[idx]) : static_cast<T>(input2[idx - count1]);
404  }
405  } else {
406 // _CCCL_PRAGMA_UNROLL_FULL()
407 #pragma unroll
408  for (int ITEM = 0; ITEM < ITEMS_PER_THREAD; ++ITEM) {
409  int idx = BLOCK_THREADS * ITEM + threadIdx.x;

411  output[ITEM] =
