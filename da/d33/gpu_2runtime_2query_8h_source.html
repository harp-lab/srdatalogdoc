41 // Helper to copy data from Host DB to Device Bundle
42 // This functor iterates over Schemas/Relations and performs the copy
43 template <typename DB, typename DeviceTuple, std::size_t Ver>
44 struct Initializer {
45  static void execute(DB& host_db, DeviceTuple& device_tuple) {
46  using SchemaTuple = typename DB::schema_type;
47  constexpr std::size_t N = std::tuple_size_v<SchemaTuple>;
48 
49  mp::mp_for_each<mp::mp_iota_c<N>>([&](auto I) {
50  using Schema = std::tuple_element_t<I, SchemaTuple>;
51 
52  // Get Host Relation - access directly via tuple indexing to avoid get_relation_by_schema
53  // visibility issues We know I is the index of Schema in SchemaTuple, so we can use it
54  // directly
55  auto& host_rel = [&]() -> auto& {
56  if constexpr (Ver == FULL_VER) {
57  return std::get<I>(host_db.full);
58  } else if constexpr (Ver == DELTA_VER) {
59  return std::get<I>(host_db.delta);
60  } else if constexpr (Ver == NEW_VER) {
61  return std::get<I>(host_db.newt);
62  }
63  }();
64 
65  // Get Device Relation from tuple
66  auto& device_rel = std::get<I>(device_tuple);
67 
68  // Convert and Assign
69  // We use the to_device static method which returns a Device Relation
70  // Verify logic: Relation::to_device returns a new instance.
71  // TaggedRelation has move ctor/assignment from Base Relation.
72  device_rel = DeviceRelationType<Schema>::to_device(host_rel);
73 
74  // Set version (important for executor lookup usually? No, executor looks up by location in
75  // separate bundles usually, but setting version is good practice)
76  device_rel.set_version(Ver);
77  // Copy name for debugging
78  device_rel.set_name(host_rel.name() + (Ver == FULL_VER
79  ? "_Full_GPU"
80  : (Ver == DELTA_VER ? "_Delta_GPU" : "_New_GPU")));
81 
82  std::cerr << "[DEBUG] Initializer: Copied " << host_rel.name()
83  << " to device. Size: " << device_rel.size() << std::endl;
84  if (device_rel.size() > 0) {
85  // Verify content by copying back
86  // Call to_host on host relation type, not device relation type
87  auto verify_host = HostRelationType<Schema>::to_host(device_rel);
88  // to_host only populates interned columns. We must reconstruct original columns to print
89  // them.
90  verify_host.reconstruct_columns_from_interned();
91 
92  std::cerr << "[DEBUG] Initializer: Content of " << host_rel.name()
93  << " on device (first 5):" << std::endl;
94  size_t num_verify = std::min((size_t)5, verify_host.size());
95 
96  for (size_t i = 0; i < num_verify; ++i) {
97  std::cerr << " Row " << i << ": (";
98  bool first = true;
99  mp::mp_for_each<mp::mp_iota_c<Schema::Arity>>([&](auto I) {
100  if (!first)
101  std::cerr << ", ";
102  std::cerr << verify_host.template column<I>()[i];
103  first = false;
104  });
105  std::cerr << ")" << std::endl;
106  }
107  }
108  });
109  }
110 };

116 template <AST::CFixpoint FP>
117 struct GPUQueryExecutor {
118  // 1. Compile Plan
119  using Plan = typename SRDatalog::CompileFixpointToPlan<FP>::type;
120  using Instructions = typename Plan::instructions_type;
121 
122  template <typename DB>
123  static void execute(DB& host_db, std::size_t max_iterations = 100) {
124  LOG_INFO << "Starting GPU Query Execution...";
125 
126  // 2. Define Device Storage Types
127  using SchemaTuple = typename DB::schema_type;
128 
129  using DeviceFullTuple = typename BuildDeviceRelationTuple<SchemaTuple>::type;
130  using DeviceDeltaTuple = typename BuildDeviceRelationTuple<SchemaTuple>::type;
131  using DeviceNewtTuple = typename BuildDeviceRelationTuple<SchemaTuple>::type;
132 
133  using RelationsBundle = std::tuple<DeviceFullTuple, DeviceDeltaTuple, DeviceNewtTuple>;
134 
135  // 3. Instantiate Device Relations
136  DeviceFullTuple full_tuple;
137  DeviceDeltaTuple delta_tuple;
138  DeviceNewtTuple newt_tuple;
139 
140  // 4. Initialize/Copy Data from Host
141  LOG_INFO << "Copying data to GPU...";
142  Initializer<DB, DeviceFullTuple, FULL_VER>::execute(host_db, full_tuple);
143  Initializer<DB, DeviceDeltaTuple, DELTA_VER>::execute(host_db, delta_tuple);
144  Initializer<DB, DeviceNewtTuple, NEW_VER>::execute(host_db, newt_tuple);
145 
146  // 5. Build Bundle
147  // We move the tuples into the bundle.
148  // Note: The bundle holds the relations by value (move).
149  RelationsBundle bundle =
150  std::make_tuple(std::move(full_tuple), std::move(delta_tuple), std::move(newt_tuple));
151 
152  // 6. Execute Fixpoint on GPU
153  // LOG_INFO << "Executing GPU Fixpoint Plan: " << get_type_name<Instructions>();
154  GPUFixpointExecutor<Instructions, RelationsBundle, SchemaTuple>::execute(bundle, 0,
155  max_iterations);
156 
157  cudaDeviceSynchronize();
158  LOG_INFO << "GPU Execution Complete.";
159 
160  // 7. Copy Results Back to Host (Optional / On Demand)
161  // For now, let's copy FULL relations back to verify correctness in tests.
162  // We need a Helper for this too.
163  ResultReflector<DB, DeviceFullTuple, FULL_VER>::execute(host_db, std::get<0>(bundle));
164  }

166  // Helper to reflect results back to host
167  template <typename DB, typename DeviceTuple, std::size_t Ver>
168  struct ResultReflector {
169  static void execute(DB& host_db, DeviceTuple& device_tuple) {
170  using SchemaTuple = typename DB::schema_type;
171  constexpr std::size_t N = std::tuple_size_v<SchemaTuple>;
172 
173  mp::mp_for_each<mp::mp_iota_c<N>>([&](auto I) {
174  using Schema = std::tuple_element_t<I, SchemaTuple>;
175 
176  auto& device_rel = std::get<I>(device_tuple);
177 
178  // Convert back to host

180  auto host_rel_copy = HostRelationType<Schema>::to_host(device_rel);
181  host_rel_copy.reconstruct_columns_from_interned();
182 
183  // Replace host relation? Or merge?
184  // Usually we want to replace the content of FULL relations with the result.
185  // Access directly via tuple indexing to avoid get_relation_by_schema visibility issues
186  auto& host_rel = [&]() -> auto& {
