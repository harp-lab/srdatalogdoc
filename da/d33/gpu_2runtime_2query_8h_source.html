128  /**
129  * @brief Executes the fixpoint loop on the GPU.
130  *
131  * @details Iterates up to `max_iterations`, calling `GPUFixpointExecutor`
132  * for each iteration. Synchronizes the device after completion.
133  *
134  * @tparam SchemaTuple The database schema blueprint (CDatabase type)
135  * @param device_db The Device Database to operate on (GPU memory)
136  * @param max_iterations Maximum number of iterations to run
137  */
138  template <AST::CDatabase SchemaTuple>
139  static void execute_kernel(DeviceDB<SchemaTuple>& device_db, std::size_t max_iterations = 100) {
140 
141  // Tuple indices match Version macros (FULL_VER=0, DELTA_VER=1, NEW_VER=2)
142  // (See detail::get_relation in gpu_fixpoint_executor.h)
143 
144  // Build ALL required indexes before entering fixpoint loop
145  // Extract all IndexSpecs from Instructions (ExecutePipeline, RebuildIndex, InsertFromRelation,
146  // etc.)
147  auto bundle_tuple = std::tie(device_db.full, device_db.delta, device_db.newt);
148 
149  // Helper to extract IndexSpec from a single instruction
150  auto build_indexes_from_instruction = [&]<typename Inst>(Inst) {
151  if constexpr (mir::is_rebuild_index_v<Inst>) {
152  // RebuildIndex<IndexSpecT>
153  using IndexSpecT = typename Inst::index_spec_type;
154  using Schema = typename IndexSpecT::schema_type;
155  constexpr std::size_t ver = IndexSpecT::kVersion;
156  using ColSeq = typename IndexSpecT::column_indexes_type;
157  auto runtime_spec = []<int... Cols>(std::integer_sequence<int, Cols...>) {
158  return SRDatalog::IndexSpec{{Cols...}};
159  }(ColSeq{});
160  auto& rel = GPU::detail::get_relation<Schema, ver, decltype(bundle_tuple), SchemaTuple>(
161  bundle_tuple);
162  rel.ensure_index(runtime_spec);
163  } else if constexpr (mir::is_insert_from_relation_v<Inst>) {
164  // InsertFromRelation<Schema, FromVer, ToVer, IndexSpec>
165  using IndexSpecT = typename Inst::index_spec_type;
166  using Schema = typename Inst::schema_type;
167  constexpr std::size_t from_ver = Inst::FromVersion;
168  constexpr std::size_t to_ver = Inst::ToVersion;
169  using ColSeq = typename IndexSpecT::column_indexes_type;
170  auto runtime_spec = []<int... Cols>(std::integer_sequence<int, Cols...>) {
171  return SRDatalog::IndexSpec{{Cols...}};
172  }(ColSeq{});
173  // Build index on both FROM and TO relations
174  auto& from_rel =
175  GPU::detail::get_relation<Schema, from_ver, decltype(bundle_tuple), SchemaTuple>(
176  bundle_tuple);
177  from_rel.ensure_index(runtime_spec);
178  auto& to_rel =
179  GPU::detail::get_relation<Schema, to_ver, decltype(bundle_tuple), SchemaTuple>(
180  bundle_tuple);
181  to_rel.ensure_index(runtime_spec);
182  } else if constexpr (mir::is_execute_pipeline_v<Inst>) {
183  // ExecutePipeline<Pipeline> - extract IndexSpecs from Pipeline's MIROps
184  using Pipeline = typename Inst::pipeline_type;
185  using MIROps = typename Pipeline::mir_ops_type;
186  using ExtractedIndexSpecs = typename mir::ExtractIndexSpecsFromMIR<MIROps>::type;
187  mp::mp_for_each<ExtractedIndexSpecs>([&]<typename IndexSpecT>(IndexSpecT) {
188  using Schema = typename IndexSpecT::schema_type;
189  constexpr std::size_t ver = IndexSpecT::kVersion;
190  using ColSeq = typename IndexSpecT::column_indexes_type;
191  auto runtime_spec = []<int... Cols>(std::integer_sequence<int, Cols...>) {
192  return SRDatalog::IndexSpec{{Cols...}};
193  }(ColSeq{});
194  auto& rel = GPU::detail::get_relation<Schema, ver, decltype(bundle_tuple), SchemaTuple>(
195  bundle_tuple);
196  rel.ensure_index(runtime_spec);
197  });
198  }
199  };
200 
201  // Build all indexes from all instructions
202  mp::mp_for_each<Instructions>(build_indexes_from_instruction);
203 
204  // Verify if CartesianJoin is present in any pipeline
205  mp::mp_for_each<Instructions>([](auto inst) {
206  using Inst = decltype(inst);
207  if constexpr (mir::is_execute_pipeline_v<Inst>) {
208  using Pipeline = typename Inst::pipeline_type;
209  using MIROps = typename Pipeline::mir_ops_type;
210  if constexpr (mir::gpu_opt::HasCartesianJoin<MIROps>::value) {
211  std::cout << "[GPU Opt] Pipeline has CartesianJoin optimization enabled." << std::endl;
212  } else {
213  std::cout
214  << "[GPU Opt] Pipeline does NOT have CartesianJoin. This may cause load imbalance."
215  << std::endl;
216  }
217  }
218  });
219 
220  for (std::size_t i = 0; i < max_iterations; ++i) {
221  // Create tuple of references to pass to executor
222  std::cout << "Iteration " << i << std::endl;
223  auto bundle_tuple = std::tie(device_db.full, device_db.delta, device_db.newt);
224  bool changed =
225  GPUFixpointExecutor<Instructions, decltype(bundle_tuple), SchemaTuple>::execute(
226  bundle_tuple, i, max_iterations);
227  if (i > 0 && !changed) {
228  break;
229  }
230  }
231  cudaDeviceSynchronize();
232  }

273  /**
274  * @brief Full execution pipeline: Prepare -> Kernel -> Reflect.
275  *
276  * @details Convenience method to run the entire GPU query process.
277  *
278  * @tparam HostDB The Host Database type
279  * @param host_db The Host Database (input and output, CPU memory)
280  * @param max_iterations Maximum iterations for the fixpoint loop

282  template <typename HostDB>
283  static void execute(HostDB& host_db, std::size_t max_iterations = 100) {

285  using SchemaTuple = typename HostDB::schema_type;
286  execute_kernel<SchemaTuple>(device_db, max_iterations);

290  /**
291  * @brief Helper struct to iterate over relations and copy them back to host.
292  *
293  * @tparam DB Host Database type
294  * @tparam DeviceTuple Tuple of device relations
295  * @tparam Ver Version tag (FULL_VER, DELTA_VER, etc.)
296  */
297  template <typename DB, typename DeviceTuple, std::size_t Ver>
298  struct ResultReflector {
299  static void execute(DB& host_db, DeviceTuple& device_tuple) {
300  using SchemaTuple = typename DB::schema_type;
301  constexpr std::size_t N = std::tuple_size_v<SchemaTuple>;
302 
303  mp::mp_for_each<mp::mp_iota_c<N>>([&](auto I) {
304  using Schema = std::tuple_element_t<I, SchemaTuple>;
305 
306  auto& device_rel = std::get<I>(device_tuple);
307 
308  // Convert back to host
309  // Call to_host on host relation type, not device relation type
310  auto host_rel_copy = HostRelationType<Schema>::to_host(device_rel);
311  host_rel_copy.reconstruct_columns_from_interned();
312 
313  // Replace host relation? Or merge?
314  // Usually we want to replace the content of FULL relations with the result.
315  // Access directly via tuple indexing to avoid get_relation_by_schema visibility issues
316  auto& host_rel = [&]() -> auto& {
317  if constexpr (Ver == FULL_VER) {
318  return std::get<I>(host_db.full);
319  } else if constexpr (Ver == DELTA_VER) {
320  return std::get<I>(host_db.delta);
321  } else if constexpr (Ver == NEW_VER) {
322  return std::get<I>(host_db.newt);
323  }
324  }();
325  // Replace host relation content
326  host_rel.clear();
327  host_rel.concat(host_rel_copy);
328  });
329  }
330  };
331 };
